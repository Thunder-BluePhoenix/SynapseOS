### 1. Natural Language Prompts
Generate data using simple English descriptions:

```python

generate_with_ai("Create 10 employees for a tech startup")

# Example 2: Detailed scenario
generate_with_ai("""
    Create a B2B SaaS company with:

    - 50 customers (small, medium, large businesses)
    - 200 support tickets (realistic issues)
    - Make sure employee skills match their roles
    - Ensure customer industries are tech-focused

# Example 3: Specific relationships

    Create 5 sales orders where:
    - Customers are from USA and Canada
    - Order values range from $10k to $100k
    - Products are software licenses
    - Include realistic payment terms
""")
```

AI understands relationships between fields:

```python
# Traditional random generation (bad):
{

# AI-powered generation (good):
{
    "customer_name": "Rajesh Kumar",
    "country": "India",            # ✅ Matches name
    "email": "rajesh.kumar@techcorp.in",  # ✅ Indian domain
    "phone": "+91-9876543210",    # ✅ Indian format

```

## 🏗️ Implementation Architecture

### Phase 1: Basic AI Integration

```python
# f_docgenesis/core/ai_engine.py

import frappe
import json
from typing import Dict, List, Any, Optional

class AIDataGenerator:
    """AI-powered data generation engine"""

    def __init__(self, provider: str = 'openai'):
        """
        Initialize AI engine

        Args:
            provider: 'openai', 'anthropic', or 'local'
        """
        self.provider = provider
        self.api_key = frappe.conf.get(f'{provider}_api_key')
        self.model = self._get_model()

    def _get_model(self) -> str:
        """Get appropriate model based on provider"""
        models = {
            'openai': 'gpt-4-turbo',
            'anthropic': 'claude-3-5-sonnet',
            'local': 'llama-3-70b'
        }
        return models.get(self.provider)

    def generate_from_prompt(self, prompt: str, doctype: str,
                            count: int = 10) -> List[Dict]:
        """
        Generate data from natural language prompt

        Args:
            prompt: Natural language description
            doctype: Target Frappe doctype
            count: Number of records to generate

        Returns:
            List of document dictionaries
        """
        # Get doctype schema
        from f_docgenesis.core.field_parser import DoctypeFieldParser
        parser = DoctypeFieldParser(doctype)
        schema = self._build_schema_description(parser)

        # Build AI prompt
        ai_prompt = self._build_ai_prompt(prompt, schema, count)

        # Call AI API
        response = self._call_ai_api(ai_prompt)

        # Parse and validate response
        documents = self._parse_ai_response(response, doctype)

        return documents

    def _build_schema_description(self, parser) -> str:
        """Build human-readable schema description"""
        fields = parser.get_all_fields()

        schema_parts = [f"DocType: {parser.doctype}\n\nFields:"]

        for field in fields:
            field_desc = f"- {field['fieldname']} ({field['fieldtype']})"
            if field.get('reqd'):
                field_desc += " [Required]"
            if field.get('options'):
                field_desc += f" Options: {field['options']}"
            schema_parts.append(field_desc)

        return "\n".join(schema_parts)

    def _build_ai_prompt(self, user_prompt: str, schema: str,
                        count: int) -> str:
        """Build complete AI prompt"""
        return f"""You are a data generation expert for Frappe/ERPNext.

User Request: {user_prompt}

Target Schema:
{schema}

Task: Generate {count} realistic records that match the schema and fulfill the user's request.

Requirements:
1. Ensure all required fields are filled
2. Make data contextually appropriate (e.g., if name is "John", email should be john@...)
3. Maintain realistic relationships between fields
4. Use appropriate formats (dates, emails, phones)
5. Generate diverse but realistic data

Output Format: Valid JSON array of objects, each matching the schema.

Example:
[
    {{
        "field1": "value1",
        "field2": "value2"
    }},
    ...
]

Generate the data:"""

    def _call_ai_api(self, prompt: str) -> str:
        """Call AI API based on provider"""
        if self.provider == 'openai':
            return self._call_openai(prompt)
        elif self.provider == 'anthropic':
            return self._call_anthropic(prompt)
        elif self.provider == 'local':
            return self._call_local_llm(prompt)

    def _call_openai(self, prompt: str) -> str:
        """Call OpenAI API"""
        import openai

        openai.api_key = self.api_key

        response = openai.ChatCompletion.create(
            model=self.model,
            messages=[
                {"role": "system", "content": "You are a data generation expert."},
                {"role": "user", "content": prompt}
            ],
            temperature=0.7,
            max_tokens=4000
        )

        return response.choices[0].message.content

    def _call_anthropic(self, prompt: str) -> str:
        """Call Anthropic Claude API"""
        import anthropic

        client = anthropic.Anthropic(api_key=self.api_key)

        message = client.messages.create(
            model=self.model,
            max_tokens=4000,
            messages=[
                {"role": "user", "content": prompt}
            ]
        )

        return message.content[0].text

    def _call_local_llm(self, prompt: str) -> str:
        """Call local LLM (Ollama, etc.)"""
        import requests

        response = requests.post(
            'http://localhost:11434/api/generate',
            json={
                'model': self.model,
                'prompt': prompt,
                'stream': False
            }
        )

        return response.json()['response']

    def _parse_ai_response(self, response: str, doctype: str) -> List[Dict]:
        """Parse AI response into document dictionaries"""
        # Extract JSON from response (might have markdown code blocks)
        json_start = response.find('[')
        json_end = response.rfind(']') + 1

        if json_start == -1 or json_end == 0:
            raise ValueError("No valid JSON array found in AI response")

        json_str = response[json_start:json_end]
        documents = json.loads(json_str)

        # Validate each document
        validated_docs = []
        for doc_data in documents:
            doc_data['doctype'] = doctype
            validated_docs.append(doc_data)

        return validated_docs


# Scenario generators
class ScenarioGenerator:
    """Pre-built business scenario generators"""

    SCENARIOS = {
        'ecommerce_black_friday': {
            'description': 'E-commerce Black Friday sale scenario',
            'doctypes': ['Customer', 'Sales Order', 'Item', 'Support Ticket'],
            'prompt_template': """
                Generate a Black Friday e-commerce scenario with:
                - {customer_count} customers (realistic demographics)
                - {order_count} orders (3x normal volume)
                - Peak times: midnight launch, morning rush
                - Popular categories: {categories}
                - Discounts: {discount_range}
                - Increased support tickets (returns, payment issues)
            """
        },

        'startup_seed_round': {
            'description': 'Tech startup after seed funding',
            'doctypes': ['Employee', 'Department', 'Project', 'Task'],
            'prompt_template': """
                Generate a tech startup scenario:
                - {employee_count} employees (founders, early team)
                - Departments: Engineering, Product, Sales
                - Active projects: MVP development, customer acquisition
                - Realistic skill distribution
                - Early-stage tasks and priorities
            """
        },

        'hospital_emergency': {
            'description': 'Hospital emergency department',
            'doctypes': ['Patient', 'Appointment', 'Medical Record'],
            'prompt_template': """
                Generate hospital emergency scenario:
                - {patient_count} patients
                - Various severity levels (critical, urgent, standard)
                - Realistic medical conditions
                - Appropriate wait times
                - Staff assignments
            """
        },

        'manufacturing_production': {
            'description': 'Manufacturing production run',
            'doctypes': ['Work Order', 'Item', 'Stock Entry', 'Quality Inspection'],
            'prompt_template': """
                Generate manufacturing scenario:
                - {product_count} products being manufactured
                - Raw materials consumption
                - Quality checks
                - Production schedules
                - Realistic yield rates
            """
        }
    }

    def generate_scenario(self, scenario_name: str,
                         parameters: Dict[str, Any]) -> Dict[str, Any]:
        """Generate a complete business scenario"""
        if scenario_name not in self.SCENARIOS:
            raise ValueError(f"Unknown scenario: {scenario_name}")

        scenario = self.SCENARIOS[scenario_name]
        prompt = scenario['prompt_template'].format(**parameters)

        ai_gen = AIDataGenerator()

        results = {}
        for doctype in scenario['doctypes']:
            docs = ai_gen.generate_from_prompt(prompt, doctype)
            results[doctype] = docs

        return results
```

### Phase 2: API Endpoints

```python
# f_docgenesis/api/ai_generation.py

import frappe
from typing import Dict, List, Any
import json

from f_docgenesis.core.ai_engine import AIDataGenerator, ScenarioGenerator


@frappe.whitelist()
def generate_with_ai(prompt: str, doctype: str, count: int = 10,
                     create_records: bool = True) -> Dict[str, Any]:
    """
    Generate data using AI from natural language prompt

    Args:
        prompt: Natural language description
        doctype: Target doctype
        count: Number of records
        create_records: Actually create the records (vs preview)

    Returns:
        Generated data and stats
    """
    try:
        count = int(count)
        create_records = bool(create_records)

        # Check permissions
        if not frappe.has_permission(doctype, 'create'):
            frappe.throw(f"Insufficient permissions to create {doctype}")

        # Generate with AI
        ai_gen = AIDataGenerator()
        documents = ai_gen.generate_from_prompt(prompt, doctype, count)

        created_records = []

        if create_records:
            # Create the records
            for doc_data in documents:
                try:
                    doc = frappe.get_doc(doc_data)
                    doc.insert(ignore_permissions=True)

                    # Mark as AI-generated
                    if hasattr(doc, 'custom_fdocgenesis_flag'):
                        doc.custom_fdocgenesis_flag = 1
                    if hasattr(doc, 'custom_fdocgenesis_ai_generated'):
                        doc.custom_fdocgenesis_ai_generated = 1
                    doc.save(ignore_permissions=True)

                    created_records.append(doc.name)
                except Exception as e:
                    frappe.log_error(f"Error creating {doctype}: {str(e)}")
                    continue

            frappe.db.commit()

        return {
            'success': True,
            'prompt': prompt,
            'doctype': doctype,
            'requested_count': count,
            'generated_count': len(documents),
            'created_count': len(created_records) if create_records else 0,
            'preview': documents if not create_records else None,
            'created_records': created_records if create_records else None
        }

    except Exception as e:
        frappe.log_error(f"AI generation error: {str(e)}")
        return {
            'success': False,
            'error': str(e)
        }


@frappe.whitelist()
def generate_scenario(scenario_name: str, parameters: str) -> Dict[str, Any]:
    """
    Generate a pre-built business scenario

    Args:
        scenario_name: Name of the scenario
        parameters: JSON string of scenario parameters

    Returns:
        Generated scenario data
    """
    try:
        if isinstance(parameters, str):
            params = json.loads(parameters)
        else:
            params = parameters

        scenario_gen = ScenarioGenerator()
        results = scenario_gen.generate_scenario(scenario_name, params)

        return {
            'success': True,
            'scenario': scenario_name,
            'results': results
        }

    except Exception as e:
        frappe.log_error(f"Scenario generation error: {str(e)}")
        return {
            'success': False,
            'error': str(e)
        }


@frappe.whitelist()
def list_scenarios() -> Dict[str, Any]:
    """List available pre-built scenarios"""
    scenario_gen = ScenarioGenerator()

    scenarios = []
    for name, config in scenario_gen.SCENARIOS.items():
        scenarios.append({
            'name': name,
            'description': config['description'],
            'doctypes': config['doctypes']
        })

    return {
        'success': True,
        'scenarios': scenarios
    }
```

# AI-Powered Data Generation - Setup Guide

Complete guide to set up and use AI-powered data generation in F-DocGenesis.

---

## 🚀 Quick Start

### Option 1: OpenAI (Recommended)

1. **Get API Key:**
   - Go to https://platform.openai.com/api-keys
   - Create new API key
   - Copy the key

2. **Configure in Frappe:**
   ```bash
   # Add to site_config.json
   bench --site your-site.local set-config openai_api_key "sk-your-key-here"

   # Optional: Set custom model (default: gpt-4-turbo-preview)
   bench --site your-site.local set-config openai_model "gpt-4-turbo-preview"
   ```

3. **Test:**
   ```javascript
   frappe.call({
       method: 'f_docgenesis.api.ai_generation.check_ai_config',
       callback: function(r) {
           console.log('AI Config:', r.message);
       }
   });
   ```

### Option 2: Anthropic Claude

1. **Get API Key:**
   - Go to https://console.anthropic.com/
   - Create API key
   - Copy the key

2. **Configure:**
   ```bash
   bench --site your-site.local set-config anthropic_api_key "sk-ant-your-key-here"
   bench --site your-site.local set-config anthropic_model "claude-3-5-sonnet-20241022"
   ```

### Option 3: Local LLM (Free, requires GPU)

1. **Install Ollama:**
   ```bash
   # Linux/Mac
   curl https://ollama.ai/install.sh | sh

   # Start Ollama
   ollama serve
   ```

2. **Pull Model:**
   ```bash
   # Recommended model (requires ~40GB RAM)
   ollama pull llama3:70b

   # Or smaller model (requires ~8GB RAM)
   ollama pull llama3:8b
   ```

3. **Configure:**
   ```bash
   bench --site your-site.local set-config local_llm_enabled true
   bench --site your-site.local set-config local_llm_model "llama3:70b"
   bench --site your-site.local set-config ollama_url "http://localhost:11434"
   ```

```javascript
    let d = new frappe.ui.Dialog({
        title: __('AI Data Generation'),
        fields: [
            {
                fieldname: 'prompt',
                fieldtype: 'Small Text',

            },
            {
                fieldname: 'provider',
                fieldtype: 'Select',
                label: __('AI Provider'),

            },
            {
                fieldname: 'preview',

            }
        ],
        primary_action_label: __('Generate'),
        primary_action: function(values) {
            let provider = values.provider === 'Auto' ? null : values.provider.toLowerCase().replace(' ', '_');

            frappe.call({
                method: 'f_docgenesis.api.ai_generation.generate_with_ai',
                args: {
                    prompt: values.prompt,
                    doctype: doctype,

                    create_records: !values.preview,

                },
```

### Practice 3: Error Handling

Handle generation errors gracefully:

```python
from f_docgenesis.api import generate
import frappe

def safe_generate(doctype, count, **kwargs):
    """Generate with error handling"""
    try:
        result = generate(doctype, count, **kwargs)

        if result.get('success'):
            print(f"✅ Generated {result['records_created']} {doctype} records")
            return result
        else:
            print(f"❌ Generation failed: {result.get('error')}")
            return None

    except Exception as e:
        print(f"❌ Exception during generation: {str(e)}")
        frappe.log_error(f"Generation failed for {doctype}", str(e))
        return None

# Usage
result = safe_generate('Customer', count=10)
if result:
    print(f"Session ID: {result['session_id']}")
```

### Practice 4: Batch with Checkpoints

Generate large datasets with checkpoints:

```python
from f_docgenesis.api import generate
import frappe

def batch_generate_with_checkpoint(doctype, total_count, batch_size=100):
    """Generate in batches with commits"""
    created = []

    for batch in range(0, total_count, batch_size):
        count = min(batch_size, total_count - batch)

        print(f"Batch {batch//batch_size + 1}: Generating {count} records...")

        result = generate(doctype, count=count)

        if result.get('success'):
            created.extend(result.get('records', []))
            frappe.db.commit()  # Checkpoint
            print(f"  ✓ Created {len(created)} total records")
        else:
            print(f"  ✗ Batch failed, stopping")
            break

    return created

# Generate 500 customers in batches of 100
customers = batch_generate_with_checkpoint('Customer', total_count=500, batch_size=100)
print(f"Total created: {len(customers)}")
```

## 🐛 Known Issues & Limitations

1. **AI Generation:**
   - Requires API key configuration
   - May occasionally produce invalid JSON
   - Rate limits apply to cloud providers

2. **Deep Copy:**
   - Max depth configurable (default: 5)
   - Large document trees may be slow
   - Circular references handled but logged

   - Foreign key constraints may prevent some deletions
   - Rollback window is 24 hours
   - Deleted data not recoverable after commit

#### Dependency Resolver (`f_docgenesis/core/dependency_resolver.py`)
- **380+ lines** of graph-based dependency resolution
- **Key Features**:
  - Builds complete dependency graphs
  - Detects and prevents circular dependencies
  - Configurable recursion depth (default: 10)
  - System doctype filtering
  - Generation order calculation
  - Complexity metrics

**Usage Example**:
```python

# Build dependency graph
graph = resolve_dependencies('Sales Invoice', max_depth=5)
summary = graph.get_summary()
print(f"Total nodes: {summary['total_nodes']}")

# Check circular dependencies
has_circular, path = check_circular('Sales Invoice')
if has_circular:
    print(f"Circular: {' → '.join(path)}")
```

# F-DocGenesis Upgrade Ideas

This document contains feature ideas and enhancements for F-DocGenesis.

## ✅ Implemented Features

### 1. Deep Copy with Link Traversal (NEW!)
**Status:** Implemented
**Files:** `core/deep_copy_engine.py`, `api/copy_paste.py`

**Features:**
- **DocView Button:** Copy document with all linked dependencies
- **Recursive Link Traversal:** Automatically copies linked docs up to configurable depth
- **Smart Duplicate Detection:** When pasting, detects duplicates and reuses existing docs
- **Link Field Resolution:** Maintains relationships by updating link fields appropriately
- **Multi-Select ListView:** Copy multiple documents at once with all their dependencies

**Usage:**
```javascript
// In DocView - Copy single document with links
frappe.call({
    method: 'f_docgenesis.api.copy_paste.deep_copy_document',
    args: {
        doctype: 'Employee',
        name: 'EMP-001',
        max_depth: 5
    },
    callback: function(r) {
        if (r.message.success) {
            localStorage.setItem('fdocgenesis_clipboard', JSON.stringify(r.message));
            frappe.show_alert('Copied with linked documents!');
        }
    }
});

// In ListView - Copy multiple documents
frappe.call({
    method: 'f_docgenesis.api.copy_paste.deep_copy_multiple',
    args: {
        documents: JSON.stringify([
            {doctype: 'Employee', name: 'EMP-001'},
            {doctype: 'Employee', name: 'EMP-002'}
        ]),
        max_depth: 5
    },
    callback: function(r) {
        if (r.message.success) {
            localStorage.setItem('fdocgenesis_clipboard', JSON.stringify(r.message));
            frappe.show_alert(`Copied ${r.message.metadata.total_documents} documents!`);
        }
    }
});

// Paste documents
let clipboard = JSON.parse(localStorage.getItem('fdocgenesis_clipboard'));
frappe.call({
    method: 'f_docgenesis.api.copy_paste.paste_documents',
    args: {
        clipboard_data: JSON.stringify(clipboard),
        skip_duplicates: true
    },
    callback: function(r) {
        if (r.message.success) {
            frappe.show_alert(`Created ${r.message.created_count} documents, skipped ${r.message.duplicate_skipped} duplicates`);
        }
    }
});
```

## 3. Doctype Specifications

### 3.1 FDG Generation Session

**Purpose**: Tracks each data generation operation

| Field | Type | Description | Mandatory | Default |
|-------|------|-------------|-----------|---------|
| name | Data | Auto-generated session ID | Yes | Auto |
| doctype | Link | Target doctype for generation | Yes | - |
| count | Int | Number of main records to generate | Yes | 10 |
| profile | Select | Generation profile (Minimal/Standard/Rich/Custom) | Yes | Standard |
| max_depth | Int | Maximum dependency depth | No | 10 |
| dry_run | Check | Preview mode without creating | No | 0 |
| background | Check | Run as background job | No | 0 |
| status | Select | pending/running/completed/failed | Yes | pending |
| config | JSON | Full generation configuration | Yes | {} |
| template | Link | FDG Template (if used) | No | - |
| created_by | Link | User who initiated | Yes | Auto |
| created_at | Datetime | Session creation time | Yes | Auto |
| started_at | Datetime | Actual generation start | No | - |
| completed_at | Datetime | Generation completion | No | - |
| records_created | Int | Count of main records created | No | 0 |
| total_documents | Int | Total docs including linked | No | 0 |
| generation_time | Float | Time taken in seconds | No | 0 |
| error_message | Text | Error details if failed | No | - |
| linked_summary | JSON | Summary of linked records | No | {} |

**Indexes**:
- doctype
- created_by
- created_at
- status

**Permissions**:
- Create: Users with "Create" permission on target doctype
- Read: Creator + System Manager
- Write: Creator (only status updates)
- Delete: System Manager only

## 4. Relationships

### 4.1 Primary Relationships

```
FDG Generation Session (1) ──→ (Many) FDG Generated Record
FDG Generation Session (1) ──→ (Many) FDG Generation Log
FDG Generation Session (1) ──→ (Many) FDG Field Config
FDG Generation Session (1) ──→ (1) FDG Deletion Session (optional)

FDG Template (1) ──→ (Many) FDG Template Version
FDG Template (1) ──→ (Many) FDG Field Config
FDG Template (1) ──→ (Many) FDG Generation Session (via reference)

FDG Deletion Session (1) ──→ (Many) FDG Deleted Record

FDG Generated Record (Many) ──→ (1) Actual Frappe Document (via doctype + docname)
```

### 4.2 Cascade Behavior

**On Session Deletion**:
- Keep logs for audit (configurable)
- Keep generated records (not deleted automatically)
- Mark as orphaned in cleanup job

**On Template Deletion**:
- Keep versions for history
- Sessions retain template config in JSON

**On User Deletion**:
- Reassign to Administrator
- Maintain audit trail integrity

## 6. Data Integrity Constraints

### 6.1 Unique Constraints

- Template name (globally unique)
- Generated Record (doctype + docname combination)
- Template Version (template + version_number)

### 6.2 Foreign Key Constraints

While Frappe doesn't enforce FK at DB level, logical constraints:

- session in Generated Record must exist
- template in Generation Session must exist if set
- doctype must be valid Frappe DocType

### 6.3 Business Rules

1. **Session Status Transitions**:
   - pending → running → completed/failed
   - No reverse transitions allowed

2. **Deletion Rules**:
   - Can only delete if can_rollback = 1
   - Rollback window expires after configured hours

3. **Template Versioning**:
   - Version increments on each save
   - Old versions are immutable

4. **Generated Records**:
   - Must have custom_fdocgenesis_flag = 1
   - depth_level must be >= 0

## 10. Example Data Flows

### 10.1 Generation Flow

```
1. User creates FDG Generation Session
   ├─ doctype: "Sales Invoice"
   ├─ count: 10
   ├─ profile: "Rich"
   └─ config: {...}

2. System creates FDG Field Config (child)
   ├─ field_name: "customer"
   ├─ mode: "hybrid"
   ├─ existing_count: 2
   └─ new_count: 3

3. Generation starts
   ├─ Status: pending → running
   └─ started_at: now()

4. For each record created:
   └─ Insert into FDG Generated Record
      ├─ doctype: "Sales Invoice"
      ├─ docname: "SINV-2025-00001"
      ├─ depth_level: 0
      └─ is_primary: 1

5. For each linked record:
   └─ Insert into FDG Generated Record
      ├─ doctype: "Customer"
      ├─ docname: "CUST-2025-00001"
      ├─ parent_record: "SINV-2025-00001"
      ├─ depth_level: 1
      └─ is_primary: 0

6. Log each step:
   └─ Insert into FDG Generation Log
      ├─ level: "info"
      └─ message: "Created Customer: CUST-2025-00001"

7. Track performance:
   └─ Insert into FDG Performance Metric
      ├─ operation: "insert"
      ├─ execution_time: 0.23
      └─ record_count: 10

8. Session completes:
   ├─ Status: running → completed
   ├─ completed_at: now()
   ├─ records_created: 10
   └─ total_documents: 51

9. Audit trail:
   └─ Insert into FDG Audit Log
      ├─ operation: "generate"
      ├─ status: "success"
      └─ duration: 2.3
```

### 10.2 Deletion Flow

```
1. User initiates deletion
   └─ Create FDG Deletion Session
      ├─ generation_session: "fdg-20251115-abc123"
      └─ dry_run: 0

2. Fetch all generated records:
   └─ SELECT FROM `tabFDG Generated Record`
      WHERE session = 'fdg-20251115-abc123'
      ORDER BY depth_level DESC

3. For each record (reverse order):
   ├─ Fetch full document data
   ├─ Store in FDG Deleted Record
   ├─ Delete actual document
   └─ Log in FDG Generation Log

4. Update Deletion Session:
   ├─ deleted_count: 51
   ├─ can_rollback: 1
   └─ status: completed

5. Schedule cleanup:
   └─ Delete FDG Deleted Record after 24 hours
```

## 12. Backup & Recovery

### 12.1 Backup Strategy

**Critical Data**:
- FDG Template (always backup)
- FDG Settings (always backup)
- FDG Custom Handler (always backup)

**Optional Data**:
- FDG Generation Session (historical data)
- FDG Audit Log (compliance requirement)

**Temporary Data** (no backup needed):
- FDG Deleted Record (24-hour TTL)
- FDG Generation Log (cleanup after 30 days)

### 12.2 Recovery Scenarios

**Lost Session**:
- Can recreate from template
- No data loss for generated documents (they exist independently)

**Lost Template**:
- Check Template Version history
- Export/Import mechanism available

**Accidental Deletion**:
- Use FDG Deleted Record within rollback window
- Rollback function restores documents

- failed: Error occurred

- success: Operation completed

# 📘 FULL UPDATED PRD — F-DocGenesis

**Universal Recursive Demo-Data Generator for Frappe Applications**
Version: 2.1
Owner: Thunder BluePhoenix
Status: Ready for Development
Last Updated: 2025-11-15

---

## 1. Product Overview

### 1.1 Summary

F-DocGenesis is a Frappe application that automatically generates demo/test data for any doctype. It recursively resolves all link dependencies and can create multiple new linked documents even when valid existing master records exist.

**Core Capabilities:**
- Deep dependency generation with configurable depth
- Multiple new linked docs per field
- Optional use of existing docs
- Safe recursive deletion with rollback
- Sandbox/QA environment support
- CI-ready data seeding with batch processing
- Template-based generation profiles
- Real-time validation and conflict resolution

The goal is to accelerate development, testing, and demo environments across any Frappe ecosystem while maintaining data integrity and business rule compliance.

---

## 2. Goals & Objectives

### ✔ Primary Goals

1. **Flexible Data Generation**: Create demo documents for any doctype with configurable depth and complexity
2. **Multi-Mode Linking**: Allow selection between:
   - Use Existing Records
   - Generate New Records
   - Hybrid (mix existing + new)
3. **Multiple Linked Documents**: Generate multiple linked documents per Link field
4. **Developer-Friendly Configuration**: Provide intuitive pre-generation dialog
5. **Safe Cleanup**: Enable safe deletion of generated documents with rollback capability

### ✔ Secondary Goals

1. **CI/CD Integration**: Enable automated seeding in CI pipelines
2. **Performance Optimization**: Batch processing and background jobs for large datasets
3. **Reusability**: Template profiles for common generation scenarios
4. **Visibility**: Comprehensive logging and audit trails
5. **Validation**: Ensure generated data meets business rules and constraints

### ✔ Enhanced Goals (v2.1)

1. **Template Management**: Save, export, and share generation configurations
2. **Dry Run Mode**: Preview generation without creating records
3. **Incremental Generation**: Add to existing demo data sets
4. **Custom Handlers**: Allow developers to register custom field generators
5. **Dependency Visualization**: Show dependency graph before generation
6. **Scheduled Generation**: Periodic automated data refresh
7. **Multi-tenant Support**: Respect tenant isolation in Frappe environments

---

## 3. Target Audience

- **Frappe Developers**: Rapid test data creation
- **QA Engineers**: Comprehensive test coverage scenarios
- **DevOps Engineers**: Automated environment seeding
- **Product Managers**: Quick demo environment setup
- **Solution Integrators**: Client environment preparation
- **Training Teams**: Consistent training data sets

---

## 4. Product Scope

### 4.1 In-Scope

#### A. Doctype-based Demo Data Generator
- Developer selects any doctype
- Sets number of records to generate
- Configuration available per link field
- Template-based quick start options

#### B. Linked Field Modes

Every Link or Table field supports three modes:

**1. Use Existing Records**
- System picks existing records randomly
- Optional limit: min/max constraints
- Smart filtering based on context

**2. Generate New Records**
- Developer specifies count (1, 5, 10, custom)
- Recursive generation of dependencies
- Unique constraint handling

**3. Hybrid Mode**
- Use X existing + generate Y new
- Proportional distribution
- Configurable ratio

#### C. Recursive Link Resolution

- Resolve nested dependencies (e.g., Sales Invoice → Customer → Territory → Company)
- Prevent infinite loops using visited map
- Configurable max depth (default: 10)
- Cycle detection and handling
- Dependency graph visualization

#### D. Child Table Handling

- Developer chooses number of rows per child table
- Each child row's link fields support all 3 modes
- Can generate unique docs per row
- Bulk child record creation optimization
- Child-level validation

#### E. Controlled Deletion

- Delete demo data using metadata flag: `custom_fdocgenesis_flag = 1`
- Reverse dependency traversal
- "Delete Demo Data" button in menu
- Batch deletion with progress tracking
- **NEW**: Rollback capability for last deletion
- **NEW**: Selective deletion by generation session

#### F. Interface Support

- **Form Menu**: Menu → Generate Demo Data (in every doctype)
- **Desk Page**: "F-DocGenesis Playground" for batch operations
- **REST API**: Full CRUD operations
- **Python Functions**: `frappe.generate_demo_data()`
- **Bench CLI**: `bench generate-demo --doctype "Sales Invoice" --count 10`

#### G. Template & Profile Management (NEW)

- **Save Configurations**: Save generation settings as templates
- **Profile Library**: Pre-built profiles (Minimal, Standard, Rich, Custom)
- **Export/Import**: Share templates across environments
- **Version Control**: Track template changes
- **Team Sharing**: Collaborate on generation configs

#### H. Validation & Business Rules (NEW)

- **Pre-Generation Validation**: Check constraints before generation
- **Business Rule Compliance**: Ensure generated data is valid
- **Unique Constraint Handling**: Smart conflict resolution
- **Custom Validators**: Register custom validation functions
- **Dry Run Mode**: Preview without creating records

#### I. Performance & Scalability (NEW)

- **Batch Processing**: Generate large datasets in background
- **Queue Management**: Enqueue jobs using Frappe's background workers
- **Progress Tracking**: Real-time progress updates
- **Performance Metrics**: Track generation speed and resource usage
- **Caching**: Cache frequently used master data

#### J. Audit & Logging (NEW)

- **Comprehensive Logs**: Track all generation operations
- **Session Management**: Group generated records by session
- **Audit Trail**: Who, what, when, why
- **Export Logs**: Download generation reports
- **Analytics Dashboard**: Visualize generation patterns

#### K. Advanced Features (NEW)

- **Incremental Generation**: Add records to existing demo sets
- **Scheduled Generation**: Cron-based periodic data refresh
- **Custom Field Handlers**: Register generators for specific fieldtypes
- **Dependency Visualization**: Interactive graph of dependencies
- **Multi-tenant Support**: Respect site/company isolation

- CSV import/export for external data
- Cross-environment synchronization
- Real-time performance analytics (basic metrics only)
- Machine learning-based data generation
- Data migration tools

## 6. Functional Requirements

### FR-1: Doctype Field Parser

Extract and analyze:
- Mandatory fields
- Field types (Link, Data, Select, Table, etc.)
- Child tables and their structure
- Link dependencies and relationships
- Field options and constraints
- Hidden/internal fields (exclude from generation)
- Unique constraints and indexes
- Default values and expressions

### FR-2: Linked Field Configuration

Configuration dialog shows each Link field with:

```
Field: customer
Linked Doctype: Customer
Mode: [Use Existing | Generate New | Hybrid]
Count (New): [  ]
Count (Existing): [  ]
Filter (Optional): {"territory": "North"}
```

### FR-3: Recursive Generator

**Generation Logic**:
- If "Generate New": create defined count with full dependency resolution
- If "Use Existing": fetch from database with optional filters
- If "Hybrid": `total = existing_count + new_count`

**Recursion Controls**:
- Max depth: configurable (default 10)
- Cycle detection: visited map
- Circular reference handling
- Performance optimization for deep chains

### FR-4: Child Table Generator

For each child table:
- Specify number of child rows (min/max range)
- Child-level link fields get all 3 modes
- Bulk insert optimization
- Unique child combinations
- Child field validation

### FR-5: Faker/Random Data Engine

Generate realistic values:
- **Names**: First, last, full names
- **Contact**: Email, phone, mobile
- **Address**: Street, city, state, country, postal
- **Numbers**: Random integers, decimals, ranges
- **Dates**: Past, future, between ranges
- **Text**: Sentences, paragraphs, lorem ipsum
- **JSON**: Structured random JSON
- **Links**: Auto-resolve or random select
- **Custom Types**: Extensible for custom fieldtypes

### FR-6: Demo Data Deletion

**Capabilities**:
- Delete all records with `custom_fdocgenesis_flag = 1`
- Delete by generation session ID
- Reverse dependency traversal (delete children first)
- Dry run deletion preview
- **NEW**: Rollback last deletion (within time window)
- **NEW**: Selective deletion with filters

**Logging**:
- Doctype and docname
- Timestamp and user
- Generator settings used
- Dependencies deleted

### FR-7: Button Placement

**Form View**:
- Menu → Generate Demo Data
- Menu → Delete Demo Data
- Menu → View Generation Logs

**List View** (NEW):

### 10.1 REST API

#### Generate Demo Data

```http
POST /api/method/fdocgenesis.api.generate_demo_data
Content-Type: application/json

{
  "doctype": "Sales Invoice",
  "count": 10,
  "config": {
    "customer": {
      "mode": "hybrid",
      "existing_count": 2,
      "new_count": 3,
      "filters": {"territory": "North"}
    },
    "items": {
      "rows": 10,
      "item_code": {
        "mode": "hybrid",
        "existing_count": 4,
        "new_count": 6
      }
    }
  },
  "max_depth": 10,
  "dry_run": false,
  "background": false,
  "template_id": null
}

Response:
{
  "success": true,
  "session_id": "fdg-20251115-abc123",
  "records_created": 10,
  "total_documents": 51,
  "linked_records": {...},
  "generation_time": 2.3,
  "logs": [...]
}
```

```http
POST /api/method/fdocgenesis.api.delete_demo_data

{
  "session_id": "fdg-20251115-abc123",

  "success": true,
```

### 10.2 Python API

```python
import frappe
from fdocgenesis import generate_demo_data, delete_demo_data

# Generate demo data
result = generate_demo_data(
    doctype="Sales Invoice",
    count=10,
    config={
        "customer": {
            "mode": "hybrid",
            "existing_count": 2,
            "new_count": 3
        }

delete_demo_data(

from fdocgenesis import save_template

save_template(

from fdocgenesis import load_template

template = load_template("My Template")
```

## 13. Risks & Mitigation

### Technical Risks

| Risk | Impact | Probability | Mitigation |
|------|--------|-------------|------------|
| Infinite recursion in circular dependencies | High | Medium | Implement cycle detection and max depth limits |
| Performance degradation with deep nesting | High | High | Batch processing, caching, background jobs |
| Unique constraint violations | Medium | High | Smart conflict resolution, incremental naming |
| Database lock timeouts | Medium | Medium | Transaction batching, retry logic |
| Memory exhaustion on large datasets | High | Low | Streaming generation, pagination |

### Business Risks

| Risk | Impact | Probability | Mitigation |
|------|--------|-------------|------------|
| Low adoption rate | High | Medium | Comprehensive docs, video tutorials, community engagement |
| Production data corruption | Critical | Low | Restrict to non-production, add safety confirmations |
| Incompatibility with custom apps | Medium | Medium | Extensible architecture, custom field handlers |
| Maintenance burden | Medium | Low | Modular design, comprehensive tests |

### Security Risks

| Risk | Impact | Probability | Mitigation |
|------|--------|-------------|------------|
| Unauthorized data generation | High | Low | Permission checks, audit logging |
| Resource exhaustion (DoS) | Medium | Medium | Rate limiting, queue management |
| Sensitive data exposure in logs | Medium | Low | Sanitize logs, configurable logging levels |

## 18. Glossary

- **Doctype**: Frappe's metadata-driven document type definition
- **Link Field**: Field that references another doctype
- **Child Table**: Nested table within a document
- **Demo Data**: Test/sample data generated for development/testing
- **Session**: A single generation operation with unique ID
- **Template**: Saved configuration for reusable generation
- **Profile**: Pre-defined generation strategy (Minimal/Standard/Rich)
- **Dry Run**: Preview mode without actual record creation
- **Faker**: Library for generating realistic fake data
- **Dependency Graph**: Visual representation of linked doctypes
- **Recursive Generation**: Creating linked records automatically
- **Cycle Detection**: Preventing infinite loops in circular dependencies
- **Rollback**: Undo operation to restore previous state
