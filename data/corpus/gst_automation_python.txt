import frappe
from frappe import _
import json
import time

def log_api_request(func):
	"""Decorator to log API requests to 'API Request Log'."""
	def wrapper(*args, **kwargs):
		start_time = time.time()
		method_name = func.__name__
		user = frappe.session.user
		status = "Success"
		error_trace = None
		response_result = None

		# Capture request data (args/kwargs)
		try:
			request_data = json.dumps({"args": args, "kwargs": kwargs}, indent=2, default=str)
		except Exception:
			request_data = str({"args": args, "kwargs": kwargs})

		# Filter out 'cmd' injected by frappe.handler
		kwargs.pop("cmd", None)

		try:
			response_result = func(*args, **kwargs)
			try:
				response_data = json.dumps(response_result, indent=2, default=str)
			except Exception:
				response_data = str(response_result)
			return response_result
		except Exception as e:
			status = "Error"
			error_trace = frappe.get_traceback()
			response_data = str(e)
			raise e
		finally:
			duration = time.time() - start_time
			try:
				log = frappe.get_doc({
					"doctype": "API Request Log",
					"api_method": method_name,
					"user": user,
					"status": status,
					"duration_seconds": duration,
					"request_data": request_data,
					"response_data": response_data,
					"error_exception": error_trace
				})
				log.insert(ignore_permissions=True)
			except Exception:
				# Don't fail the API call if logging fails
				frappe.log_error("Failed to create API Request Log")

	return wrapper

@frappe.whitelist()
@log_api_request
def get_batch_detail(batch_name, page=0, page_size=20):
	"""Return paginated PAN records for a specific batch with company data."""
	page = int(page)
	page_size = int(page_size)
	start = page * page_size

	batch = frappe.get_doc("PAN Upload Batch", batch_name)
	total = len(batch.pan_details)

	# Get the slice of PAN details for this page
	detail_slice = batch.pan_details[start: start + page_size]

	records = []
	# Bulk-fetch all PAN Company Data for this page in one query
	pan_list = [d.pan for d in detail_slice]
	company_data = {}
	if pan_list:
		rows = frappe.get_all(
			"PAN Company Data",
			filters={"pan": ["in", pan_list]},
			fields=[
				"pan", "gst_number", "company_name", "business_type",
				"cin", "company_llp_name", "class_of_company",
				"company_status", "mca_company_status", "lookup_status",
				"error_message"
			],
		)
		for r in rows:
			company_data[r.pan] = r

	for detail in detail_slice:
		cd = company_data.get(detail.pan, {})
		records.append({
			"pan": detail.pan,
			"row_status": detail.row_status,
			"gst_number": cd.get("gst_number", ""),
			"company_name": cd.get("company_name", ""),
			"company_llp_name": cd.get("company_llp_name", ""),
			"cin": cd.get("cin", ""),
			"class_of_company": cd.get("class_of_company", ""),
			"company_status": cd.get("company_status", ""),
			"mca_company_status": cd.get("mca_company_status", ""),
			"lookup_status": cd.get("lookup_status", detail.row_status),
			"error_message": cd.get("error_message", ""),
		})

	return {
		"batch_name": batch_name,
		"status": batch.status,
		"total_records": total,
		"processed_records": batch.processed_records,
		"result_file": batch.result_file,
		"records": records,
		"page": page,
		"page_size": page_size,
	}

	from gst_automation.gst_automation.scraper import lookup_pan

	# Perform lookup
	result = lookup_pan(pan)

	# Update or insert PAN Company Data
	if frappe.db.exists("PAN Company Data", pan):
		doc = frappe.get_doc("PAN Company Data", pan)
		doc.update(result)
		doc.save()
	else:
		doc = frappe.new_doc("PAN Company Data")
		doc.pan = pan
		doc.update(result)
		doc.insert()

	return result

import frappe
from frappe import _
from frappe.model.document import Document
from frappe.utils.background_jobs import is_job_enqueued


class PANUploadBatch(Document):
	def after_insert(self):
		"""After the batch is saved, parse the uploaded file and populate child table."""
		self.parse_upload_file()

	def parse_upload_file(self):
		"""Read the uploaded XLSX and create child table rows."""
		from frappe.utils.xlsxutils import read_xlsx_file_from_attached_file

		rows = read_xlsx_file_from_attached_file(file_url=self.upload_file)
		if not rows:
			frappe.throw(_("The uploaded file is empty."))

		# 1. Find the header row (scan first 10 rows)
		header_row_idx = -1
		pan_col_idx = -1

		for i, row in enumerate(rows[:10]):
			# clean row values
			row_values = [str(val).strip().upper() if val else "" for val in row]
			if "PAN" in row_values:
				header_row_idx = i
				pan_col_idx = row_values.index("PAN")
				break

		if header_row_idx == -1:
			frappe.throw(_("Could not find a 'PAN' column header in the first 10 rows."))

		# 2. Extract PANs
		pan_list = []
		seen = set()

		# Iterate from the row AFTER the header
		for row in rows[header_row_idx + 1:]:
			if not row: continue  # Skip completely empty rows

			# Ensure row has enough columns
			if len(row) <= pan_col_idx: continue

			val = row[pan_col_idx]
			pan_value = str(val).strip().upper() if val else ""

			# Basic validation: length 10, alphanumeric (optional strictness)
			# For now, just check it's not empty and not "NONE"
			if pan_value and pan_value != "NONE" and pan_value not in seen:
				# strict length check? maybe just len(10)
				if len(pan_value) == 10:
					pan_list.append(pan_value)
					seen.add(pan_value)

		if not pan_list:
			frappe.throw(_("No valid PAN values found in the uploaded file."))

		self.total_records = len(pan_list)
		self.processed_records = 0

		# Clear existing if any (unlikely on insert, but safe)
		self.set("pan_details", [])

		for pan in pan_list:
			self.append("pan_details", {"pan": pan, "row_status": "Pending"})

		self.save(ignore_permissions=True)

	@frappe.whitelist()
	def start_processing(self):
		"""Enqueue the background job to process all PANs."""
		if self.status == "Processing":
			frappe.throw(_("This batch is already being processed."))

		job_id = f"pan_lookup||{self.name}"
		if not is_job_enqueued(job_id):
			self.db_set("status", "Processing")
			frappe.enqueue(
				"gst_automation.gst_automation.doctype."
				"pan_upload_batch.pan_upload_batch.process_batch",
				queue="long",
				timeout=18000,
				job_id=job_id,
				batch_name=self.name,
			)
			return True
		return False


def process_batch(batch_name):
	"""Background job: iterate PANs, scrape data, update records.

	Optimizations:
	- Skip PANs that already have 'Done' status (re-runs are fast)
	- Bulk commit every 5 records instead of per-record
	- Uses persistent HTTP sessions via scraper module
	- Resets scraper sessions on repeated failures
	"""
	from gst_automation.gst_automation.scraper import lookup_pan, reset_sessions

	batch = frappe.get_doc("PAN Upload Batch", batch_name)
	total = len(batch.pan_details)
	processed = 0
	consecutive_errors = 0

	try:
		for detail_row in batch.pan_details:
			pan = detail_row.pan

			# Skip already-completed PANs (useful for re-runs)
			if detail_row.row_status == "Done":
				processed += 1
				continue

			try:
				# Check if this PAN was already done in a previous batch
				existing_status = frappe.db.get_value(
					"PAN Company Data", pan, "lookup_status"
				)
				if existing_status == "Done":
					detail_row.db_set("pan_company_data", pan)
					detail_row.db_set("row_status", "Done")
					processed += 1
					consecutive_errors = 0
					_publish_progress(batch_name, processed, total, pan, "Done")
					continue

				result = lookup_pan(pan)
				_upsert_pan_company(pan, result)

				detail_row.db_set("pan_company_data", pan)
				detail_row.db_set(
					"row_status", result.get("lookup_status", "Error")
				)
				consecutive_errors = 0

			except Exception as e:
				frappe.log_error(
					title=f"PAN Lookup Error: {pan}",
					message=frappe.get_traceback(),
				)
				detail_row.db_set("row_status", "Error")

				# Create error record if not exists
				if not frappe.db.exists("PAN Company Data", pan):
					doc = frappe.new_doc("PAN Company Data")
					doc.pan = pan
					doc.lookup_status = "Error"
					doc.error_message = str(e)[:500]
					doc.save(ignore_permissions=True)

				consecutive_errors += 1
				# Reset sessions if too many consecutive errors
				if consecutive_errors >= 3:
					reset_sessions()
					consecutive_errors = 0

			processed += 1

			# Batch commit every 5 records for performance
			if processed % 5 == 0 or processed == total:
				batch.db_set("processed_records", processed)
				frappe.db.commit()

			_publish_progress(
				batch_name, processed, total, pan,
				detail_row.row_status or "Pending"
			)

		# Final commit
		batch.db_set("processed_records", processed)
		frappe.db.commit()

		generate_result_file(batch_name)
		batch.db_set("status", "Completed")

	except Exception:
		frappe.db.rollback()
		batch.db_set("status", "Failed")
		frappe.log_error(
			title=f"PAN Batch Processing Failed: {batch_name}",
			message=frappe.get_traceback(),
		)

	# Cleanup scraper sessions
	reset_sessions()

	frappe.publish_realtime(
		"pan_lookup_complete",
		{
			"batch_name": batch_name,
			"status": frappe.db.get_value(
				"PAN Upload Batch", batch_name, "status"
			),
		},
	)


def _upsert_pan_company(pan, result):
	"""Create or update a PAN Company Data record from scraper result."""
	if frappe.db.exists("PAN Company Data", pan):
		doc = frappe.get_doc("PAN Company Data", pan)
	else:
		doc = frappe.new_doc("PAN Company Data")
		doc.pan = pan

	doc.gst_number = result.get("gst_number", "")
	doc.company_name = result.get("company_name", "")
	doc.business_type = result.get("business_type", "")
	doc.cin = result.get("cin", "")
	doc.company_llp_name = result.get("company_llp_name", "")
	doc.class_of_company = result.get("class_of_company", "")
	doc.company_status = result.get("company_status", "")
	doc.mca_company_status = result.get("mca_company_status", "")
	doc.mca_registration_date = result.get("mca_registration_date", "")
	doc.mca_category = result.get("mca_category", "")
	doc.mca_sub_category = result.get("mca_sub_category", "")
	doc.mca_roc = result.get("mca_roc", "")
	doc.mca_authorized_capital = result.get("mca_authorized_capital", "")
	doc.mca_paidup_capital = result.get("mca_paidup_capital", "")
	doc.lookup_status = result.get("lookup_status", "Pending")
	doc.error_message = result.get("error_message", "")
	doc.save(ignore_permissions=True)


def _publish_progress(batch_name, current, total, pan, status):
	"""Send realtime progress update."""
	frappe.publish_realtime(
		"pan_lookup_progress",
		{
			"batch_name": batch_name,
			"current": current,
			"total": total,
			"pan": pan,
			"status": status,
		},
	)


def generate_result_file(batch_name):
	"""Generate XLSX result with all fields including MCA status."""
	from frappe.utils.file_manager import save_file
	from frappe.utils.xlsxutils import make_xlsx

	batch = frappe.get_doc("PAN Upload Batch", batch_name)

	headers = [
		"PAN", "GST", "Comp_Name", "CIN",
		"Company / LLP Name", "Class of Company",
		"Company Status(for efiling)", "Business Type",
		"MCA Company Status", "Registration Date",
		"Category", "Sub-Category", "ROC",
		"Authorized Capital", "Paid-up Capital",
		"Status",
	]
	data = [headers]

	# Bulk-fetch all PAN Company Data for this batch
	pan_list = [d.pan for d in batch.pan_details]
	company_map = {}
	if pan_list:
		rows = frappe.get_all(
			"PAN Company Data",
			filters={"pan": ["in", pan_list]},
			fields=[
				"pan", "gst_number", "company_name", "cin",
				"company_llp_name", "class_of_company", "company_status",
				"business_type", "mca_company_status", "mca_registration_date",
				"mca_category", "mca_sub_category", "mca_roc",
				"mca_authorized_capital", "mca_paidup_capital", "lookup_status",
			],
		)
		for r in rows:
			company_map[r.pan] = r

	for detail in batch.pan_details:
		cd = company_map.get(detail.pan)
		if cd:
			data.append([
				cd.pan,
				cd.gst_number or "",
				cd.company_name or "",
				cd.cin or "",
				cd.company_llp_name or "",
				cd.class_of_company or "",
				cd.company_status or "",
				cd.business_type or "",
				cd.mca_company_status or "",
				cd.mca_registration_date or "",
				cd.mca_category or "",
				cd.mca_sub_category or "",
				cd.mca_roc or "",
				cd.mca_authorized_capital or "",
				cd.mca_paidup_capital or "",
				cd.lookup_status or "",
			])
		else:
			data.append(
				[detail.pan] + [""] * 14 + ["Error"]
			)

	xlsx_file = make_xlsx(data, "PAN Lookup Results")
	filename = f"PAN_Results_{batch_name}.xlsx"

	# Remove old result file if exists
	if batch.result_file:
		old_file = frappe.db.get_value(
			"File", {"file_url": batch.result_file}, "name"
		)
		if old_file:
			frappe.delete_doc("File", old_file, ignore_permissions=True)

	file_doc = save_file(
		filename,
		xlsx_file.getvalue(),
		"PAN Upload Batch",
		batch_name,
		is_private=1,
	)
	batch.db_set("result_file", file_doc.file_url)

frappe.pages["pan-lookup"].on_page_load = function (wrapper) {
	var page = frappe.ui.make_app_page({
		parent: wrapper,
		title: __("PAN Lookup"),
		single_column: true,
	});

	page.main_section = $(frappe.render_template("pan_lookup")).appendTo(
		page.body.addClass("no-border")
	);

	// ── State ──
	page.batch_name = null;
	page.history_page = 0;
	page.history_page_size = 10;
	page.detail_batch = null;
	page.detail_page = 0;
	page.detail_page_size = 20;

	// ── Toolbar Buttons ──
	page.add_inner_button(__("Download Template"), function () {
		window.open(
			"/api/method/gst_automation.gst_automation.api.download_template",
			"_blank"
		);
	});

	page.add_inner_button(__("Upload PAN File"), function () {
		new frappe.ui.FileUploader({
			as_dataurl: false,
			allow_multiple: false,
			restrictions: { allowed_file_types: [".xlsx", ".xls"] },
			on_success: function (file_doc) {
				frappe.call({
					method: "gst_automation.gst_automation.api.create_batch",
					args: { file_url: file_doc.file_url },
					callback: function (r) {
						if (r.message) {
							page.batch_name = r.message.batch_name;
							_show_current_batch(page, r.message);
							_load_history(page);
						}
					},
				});
			},
		});
	});

	page.add_inner_button(__("Start Processing"), function () {
		if (!page.batch_name) {
			frappe.msgprint(__("Please upload a PAN file first."));
			return;
		}
		frappe.call({
			method: "gst_automation.gst_automation.api.start_processing",
			args: { batch_name: page.batch_name },
			callback: function (r) {
				if (r.message) {
					_set_status(page, "Processing started...", "blue");
					page.main_section.find(".progress").show();
					page.main_section.find(".pan-live-results").show();
				}
			},
		});
	});

	page.add_inner_button(__("Download Results"), function () {
		if (!page.batch_name) {
			frappe.msgprint(__("No batch to download results for."));
			return;
		}
		_download_batch(page.batch_name);
	});

	// ── History Pagination ──
	page.main_section.find(".pan-prev-page").on("click", function () {
		if (page.history_page > 0) {
			page.history_page--;
			_load_history(page);
		}
	});
	page.main_section.find(".pan-next-page").on("click", function () {
		page.history_page++;
		_load_history(page);
	});

	// ── Detail Close ──
	page.main_section.find(".pan-detail-close").on("click", function () {
		page.main_section.find(".pan-detail-section").slideUp(200);
		page.detail_batch = null;
	});

	// ── Detail Pagination ──
	page.main_section.find(".pan-detail-prev").on("click", function () {
		if (page.detail_page > 0) {
			page.detail_page--;
			_load_detail(page, page.detail_batch);
		}
	});
	page.main_section.find(".pan-detail-next").on("click", function () {
		page.detail_page++;
		_load_detail(page, page.detail_batch);
	});

	// ── Realtime ──
	frappe.realtime.on("pan_lookup_progress", function (data) {
		if (page.batch_name && data.batch_name === page.batch_name) {
			_update_progress(page, data);
		}
	});

	frappe.realtime.on("pan_lookup_complete", function (data) {
		if (page.batch_name && data.batch_name === page.batch_name) {
			var ok = data.status === "Completed";
			_set_status(
				page,
				ok
					? "Processing complete! Click 'Download Results' to get your file."
					: "Processing failed. Check Error Log for details.",
				ok ? "green" : "red"
			);
			page.main_section
				.find(".pan-progress-bar")
				.removeClass("progress-bar-animated");
			_load_history(page);
		}
	});

	// ── Load history on page load ──
	_load_history(page);
};

function _set_status(page, msg, color) {
	var el = page.main_section.find(".pan-status-message");
	el.html(msg);
	el.removeClass("text-muted text-primary text-success text-danger");
	if (color === "blue") el.addClass("text-primary");
	else if (color === "green") el.addClass("text-success");
	else if (color === "red") el.addClass("text-danger");
	else el.addClass("text-muted");
}

function _show_current_batch(page, info) {
	page.main_section
		.find(".pan-batch-info")
		.html(
			__("Batch: <strong>{0}</strong> | Total PANs: <strong>{1}</strong>", [
				info.batch_name,
				info.total_records,
			])
		);
	_set_status(page, "Batch created. Click 'Start Processing' to begin.", "blue");
	page.main_section.find(".pan-results-table tbody").empty();
	page.main_section.find(".pan-progress-bar").css("width", "0%").addClass("progress-bar-animated");

"""
PAN Lookup Scraper Module — Optimized

Key optimizations over v1:
- Persistent session with connection pooling (reuses TCP connections)
- Reduced redundant sleeps; adaptive backoff only on 403
- Single CSRF fetch per session, refreshed only on failure
- Combined GST + MCA flow with early exits
- Full MCA data extraction including Active/Inactive status,
  registration date, category, sub-category, ROC, capitals
- Optional APIClub integration for reliable CIN lookup
"""

import json

def _build_session(base_url, max_retries=2, backoff=0.5):
	"""Create a persistent session with connection pooling and auto-retry."""
	sess = requests.Session()
	retry = Retry(
		total=max_retries,
		backoff_factor=backoff,
		status_forcelist=[500, 502, 503, 504],
		allowed_methods=["GET", "POST"],
	)
	adapter = HTTPAdapter(max_retries=retry, pool_connections=5, pool_maxsize=10)
	sess.mount("https://", adapter)
	sess.mount("http://", adapter)
	return sess

def _get_findgst_session():
	"""Get or create the findgst.in session with CSRF token."""
	global _findgst_session, _findgst_csrf
	if _findgst_session is None:
		_findgst_session = _build_session("https://findgst.in")
		_findgst_session.headers.update(FINDGST_HEADERS)
		_findgst_csrf = None
	if _findgst_csrf is None:
		resp = _findgst_session.get("https://findgst.in/", timeout=20)
		soup = BeautifulSoup(resp.text, "lxml")
		token = soup.find("input", {"name": "csrfmiddlewaretoken"})
		_findgst_csrf = token["value"] if token else None
	return _findgst_session, _findgst_csrf

	"""Get or create the MCA session."""
	global _mca_session
	if _mca_session is None:

def lookup_pan(pan):
	"""
	Main entry point: given a PAN, perform the full lookup chain.

	Returns a dict with all company data fields + lookup_status.
	"""
	result = {
		"gst_number": "",
		"company_name": "",
		"business_type": "",
		"cin": "",
		"company_llp_name": "",
		"class_of_company": "",
		"company_status": "",
		"mca_company_status": "",
		"mca_registration_date": "",
		"mca_category": "",
		"mca_sub_category": "",
		"mca_roc": "",
		"mca_authorized_capital": "",
		"mca_paidup_capital": "",
		"lookup_status": "Pending",
		"error_message": "",
	}

	try:
		# Step 1: PAN → GST
		gst_info = _fetch_gst_from_pan(pan)
		if gst_info is None:
			result["lookup_status"] = "PAN Not Found"
			return result

		result["gst_number"] = gst_info.get("gst_number", "")
		result["company_name"] = gst_info.get("legal_name", "")
		result["business_type"] = gst_info.get("business_type", "")

		legal_name = result["company_name"]
		business_type = result["business_type"]

		if not legal_name:
			result["lookup_status"] = "GST Not Found"
			return result

		if not business_type:
			result["lookup_status"] = "Business Type Not Found"
			return result

		# Step 2: Partnership/Proprietorship → no CIN needed
		if business_type in ("Proprietorship", "Partnership"):
			result["company_llp_name"] = legal_name
			result["class_of_company"] = business_type
			result["lookup_status"] = "GST Found"
			return result

		# Check Configuration
		settings = frappe.get_single("GST Settings")
		use_apiclub = (
			settings.enable_apiclub
			and settings.apiclub_api_key
			and settings.cin_lookup_source in ["APIClub", "Both (Fallback)"]
		)
		mca_first = settings.cin_lookup_source != "APIClub"

		cin = "CIN not found"
		mca_data = None

		# Strategy: MCA First, fallback to APIClub
		if mca_first:
			cin = _find_cin_mca(legal_name)
			if cin != "CIN not found":
				mca_data = _get_mca_details_mca(legal_name, cin)

			if (
				(cin == "CIN not found" or not mca_data)
				and use_apiclub
				and settings.cin_lookup_source == "Both (Fallback)"
			):
				api_res = _fetch_cin_apiclub(legal_name, settings.apiclub_api_key)
				if api_res:
					mca_data = api_res
					cin = api_res.get("cin")

		# Strategy: APIClub Only/First
		elif use_apiclub:
			api_res = _fetch_cin_apiclub(legal_name, settings.apiclub_api_key)
			if api_res:
				mca_data = api_res
				cin = api_res.get("cin")
			elif settings.cin_lookup_source == "Both (Fallback)":
				cin = _find_cin_mca(legal_name)
				if cin != "CIN not found":
					mca_data = _get_mca_details_mca(legal_name, cin)

		# Final Result Population
		if mca_data:
			result.update(mca_data)
			result["lookup_status"] = "Done"
		elif cin != "CIN not found":
			result["cin"] = cin
			result["company_llp_name"] = legal_name
			result["class_of_company"] = business_type
			result["lookup_status"] = "GST Found/CIN Not Found"
		else:
			result["company_llp_name"] = legal_name
			result["class_of_company"] = business_type
			result["lookup_status"] = "GST Found/CIN Not Found"

	except Exception as e:
		result["lookup_status"] = "Error"
		result["error_message"] = str(e)[:500]

	return result


# ── Step 1+2: PAN → GST → Legal Name ──

def _fetch_gst_from_pan(pan):
	"""PAN → findgst.in → GST number + Legal Name + Business Type."""
	global _findgst_csrf

	for attempt in range(2):
		try:
			sess, csrf = _get_findgst_session()
			if not csrf:
				return None

			resp = sess.post(
				"https://findgst.in/gstin-by-pan",
				data={"gstnum": pan, "csrfmiddlewaretoken": csrf},
				timeout=20,
			)

			if resp.status_code == 403 and attempt == 0:
				_findgst_csrf = None
				continue

			soup = BeautifulSoup(resp.text, "lxml")
			gst_elem = soup.find("p", {"class": "yellow lighten-5"})
			if not gst_elem or not gst_elem.find("a"):
				return None

			gst_link = gst_elem.find("a")["href"]
			gst_number = gst_link.replace("/gst/", "")

			detail = sess.get(f"https://findgst.in{gst_link}", timeout=20)
			soup2 = BeautifulSoup(detail.text, "lxml")
			gst_dic = {}
			tbody = soup2.find("tbody")
			if tbody:
				for row in tbody.find_all("tr"):
					tds = row.find_all("td")
					if len(tds) >= 2:
						gst_dic[tds[0].text.strip()] = tds[1].text.strip()

			return {
				"gst_number": gst_number,
				"legal_name": gst_dic.get("Legal Name", ""),
				"business_type": gst_dic.get("Business Type", ""),
			}

		except requests.RequestException:
			if attempt == 0:
				_findgst_csrf = None
				time.sleep(1)
			else:
				raise

	return None


# ── Step 3: Company Name → CIN (MCA) ──

def _find_cin_mca(company_name):
	"""Look up CIN from MCA. Returns CIN string or 'CIN not found'."""
	sess = _get_mca_session()

	for attempt in range(2):
		try:
			# Try cinLookup first (faster, returns JSON)
			resp = sess.post(
				"https://www.mca.gov.in/mcafoportal/cinLookup.do",
				data=f"companyname={company_name.strip()}",
				timeout=20,
			)

			if resp.status_code == 403:
				time.sleep(3 * (attempt + 1))
				continue

			try:
				js = resp.json()
				if js.get("success") == "true" and js.get("companyList"):
					return js["companyList"][0]["companyID"]
			except (ValueError, json.JSONDecodeError):
				pass

			# Fallback: checkCompanyName (HTML table)
			return "CIN not found"

		except requests.RequestException:
			if attempt == 0:
				time.sleep(2)
			else:
				return "CIN not found"

	return "CIN not found"


# ── Step 4: CIN → Full MCA Details (MCA) ──

def _get_mca_details_mca(company_name, cin):
	"""Fetch full company details from MCA including Active/Inactive status."""
	sess = _get_mca_session()

	for attempt in range(2):
		try:
			resp = sess.post(
				"https://www.mca.gov.in/mcafoportal/companyLLPMasterData.do",
				data={
					"companyName": company_name,
					"companyID": cin,
					"displayCaptcha": "false",
				},
				timeout=25,
			)

			if resp.status_code == 403:
				time.sleep(3 * (attempt + 1))
				continue

			if resp.status_code != 200:
				return None

			soup = BeautifulSoup(resp.text, "lxml")
			div = soup.find("div", {"id": "companiesact1"})
			if not div or not div.find("table"):
				return None

			dic = {}
			for row in div.find("table").find_all("tr"):
				tds = row.find_all("td")
				if len(tds) >= 2:
					key = tds[0].text.strip()
					val = tds[1].text.strip()
					if key:
						dic[key] = val

			return {
				"cin": dic.get("CIN") or dic.get("LLPIN") or cin,
				"company_name": (
					dic.get("Company Name") or dic.get("LLP Name") or company_name
				),
				"company_status": (
					dic.get("Company Status(for efiling)")
					or dic.get("LLP Status")
					or ""
				),
				"class_of_company": dic.get("Class of Company") or "-",
				"mca_company_status": (
					dic.get("Company Status(for efiling)")
					or dic.get("LLP Status")
					or dic.get("Company Status")
					or ""
				),
				"mca_registration_date": (
					dic.get("Date of Incorporation")
					or dic.get("Date of Registration")
					or ""
				),
				"mca_category": dic.get("Category") or dic.get("Company Category") or "",
				"mca_sub_category": (
					dic.get("Sub Category")
					or dic.get("Company Sub Category")
					or ""
				),
				"mca_roc": (
					dic.get("Registrar of Companies")
					or dic.get("ROC Code")
					or ""
				),
				"mca_authorized_capital": (
					dic.get("Authorized Capital(Rs)")
					or dic.get("Authorized Capital")
					or ""
				),
				"mca_paidup_capital": (
					dic.get("Paid up Capital(Rs)")
					or dic.get("Paid up Capital")
					or ""
				),
			}

		except (requests.RequestException, AttributeError, IndexError):
			if attempt == 0:
				time.sleep(2)
			else:
				return None

	return None


# ── Step 5: APIClub Integration ──

def _fetch_cin_apiclub(company_name, api_key):
	"""Fetch CIN and company details from APIClub.in."""
	try:
		# Search API to get CIN
		url = "https://api.apiclub.in/api/v1/company_cin_data"
		payload = {"company_name": company_name}
		headers = {
			"Referer": "https://apiclub.in",
			"API-KEY": api_key,
			"Content-Type": "application/json"
		}

		# NOTE: The actual endpoint might vary. Assuming standard search.
		# If direct CIN search returns list:
		resp = requests.post(url, json=payload, headers=headers, timeout=15)
		data = resp.json()

		if data.get("status") == "success" and data.get("data"):
			# Assuming data contains the company info directly or list
			# Adjust based on actual API response structure
			# Commonly returns list of matches
			company = data["data"][0] if isinstance(data["data"], list) else data["data"]

			return {
				"cin": company.get("cin", ""),
				"company_name": company.get("company_name", company_name),
				"company_status": company.get("company_status", ""),
				"class_of_company": company.get("class_of_company", ""),
				"mca_company_status": company.get("company_status", ""),
				"mca_registration_date": company.get("date_of_registration", ""),

import os
import shutil

import frappe
from frappe.installer import update_site_config


def execute():
	# Skip if already configured
	if frappe.conf.get("chromium_binary_path"):
		return

	candidates = [
		# Linux (Frappe Cloud / Ubuntu)
		"/usr/bin/google-chrome",
		"/usr/bin/google-chrome-stable",
		"/usr/bin/chromium-browser",
		"/usr/bin/chromium",
		"/snap/bin/chromium",
		# macOS
		"/Applications/Google Chrome.app/Contents/MacOS/Google Chrome",
		"/Applications/Chromium.app/Contents/MacOS/Chromium",
		# PATH fallback (any OS)
		shutil.which("google-chrome-stable") or "",
		shutil.which("google-chrome") or "",
		shutil.which("chromium-browser") or "",
		shutil.which("chromium") or "",
	]

	path = next((p for p in candidates if p and os.path.isfile(p)), None)

	if path:
		update_site_config("chromium_binary_path", path)
	else:
		frappe.log_error(
			"set_chromium_binary_path patch: no Chrome/Chromium binary found. "
			"Install google-chrome-stable and re-run bench migrate.",
			"Chromium Not Found",
		)
