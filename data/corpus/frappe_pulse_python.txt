	return doc.as_dict()


@frappe.whitelist()
def resolve_conflict(conflict_name: str, resolution: str, note: str = ""):
	"""
	Resolve a PENDING conflict.
	`resolution` must be 'local' | 'remote' | 'merge'
	For 'merge', the caller must also pass `merged_payload` as JSON string.
	"""
	frappe.only_for("System Manager")

	if resolution not in ("local", "remote", "merge"):
		frappe.throw(_("resolution must be local, remote, or merge"))

	conflict = frappe.get_doc("Pulse Conflict", conflict_name)
	if conflict.status != "PENDING":
		frappe.throw(_("Conflict is already resolved"))

	local_snap = conflict.local_snapshot or {}
	remote_snap = conflict.remote_snapshot or {}
	doc_type = conflict.doc_type

	if resolution == "local":
		final_payload = local_snap
	elif resolution == "remote":
		final_payload = remote_snap
	else:
		# merge: use merged_payload from form_dict
		import json
		raw = frappe.local.form_dict.get("merged_payload") or "{}"
		final_payload = json.loads(raw) if isinstance(raw, str) else raw

	_apply_doc_event("UPDATE", doc_type, final_payload)

	conflict.db_set("status", "RESOLVED")
	conflict.db_set("resolved_by", frappe.session.user)
	conflict.db_set("resolved_at", frappe.utils.now())
	conflict.db_set("resolved_payload", final_payload)
	conflict.db_set("resolution_note", note)
	frappe.db.commit()

	return {"message": _("Conflict resolved successfully")}

	nodes = frappe.get_all(
		"Pulse Node",
		fields=[
			"name", "node_id", "node_name", "sync_status",
			"sync_direction", "last_synced_at", "last_error",

	)

		"is_offline": is_offline(),
		"total_nodes": len(nodes),
		"active_nodes": active,
		"error_nodes": error,
		"offline_pending": offline_pending,
		"nodes": node_list,
	}

import json

import frappe
from frappe import _

from frappe_pulse.sync.protocol import (
	apply_event,
	get_local_schema_version,
)


@frappe.whitelist(allow_guest=False)
def pull(after: str = "2000-01-01 00:00:00", node_id: str = "", limit: int = 100):
	"""
	Remote node calls this to fetch events produced after `after` timestamp.
	GET /api/method/frappe_pulse.api.sync.pull
	"""
	limit = min(int(limit), 500)

	events = frappe.get_all(
		"Pulse Event Log",
		filters={
			"timestamp": (">", after),
			"sync_status": "PENDING",
		},
		fields=[
			"event_id", "event_type", "doc_type", "doc_name",
			"node_id", "schema_version", "payload", "checksum", "timestamp",
		],
		order_by="timestamp asc",
		limit=limit,
	)

	# Deserialise payload if stored as string
	for e in events:
		if isinstance(e.get("payload"), str):
			try:
				e["payload"] = json.loads(e["payload"])
			except Exception:
				e["payload"] = {}
		e["timestamp"] = str(e["timestamp"])

	# Apply selective sync scope filter for the requesting node
	if node_id:
		from frappe_pulse.selective.scope import filter_events
		events = filter_events(events, node_id)

	# Mobile-optimized response: strip fields + MessagePack encode
	from frappe_pulse.mobile.response import encode_mobile_response, is_mobile_request, strip_payload
	if is_mobile_request():
		for e in events:
			e["payload"] = strip_payload(e.get("payload", {}))
		body, content_type = encode_mobile_response({"events": events, "count": len(events)})
		frappe.local.response["type"] = "binary"
		frappe.local.response["filename"] = "events.msgpack"
		frappe.local.response.update({"http_content_type": content_type, "filecontent": body})
		return

	return {"events": events, "count": len(events)}


@frappe.whitelist(allow_guest=False)
def receive():
	"""
	Remote node pushes a batch of events to this node.
	POST /api/method/frappe_pulse.api.sync.receive
	Body: { "events": [ <event>, ... ] }
	"""
	data = frappe.local.form_dict
	events = data.get("events") or []

	if isinstance(events, str):
		try:
			events = json.loads(events)
		except Exception:
			frappe.throw(_("Invalid events payload"), frappe.ValidationError)

	if not isinstance(events, list):
		frappe.throw(_("events must be a list"), frappe.ValidationError)

	local_schema = get_local_schema_version()
	results = {"applied": 0, "skipped": 0, "schema_mismatch": 0, "failed": 0}

	for event in events:
		outcome = apply_event(event, local_schema)
		if outcome == "applied":
			results["applied"] += 1
			# Fire push notification after successful apply — no separate backend needed
			frappe.enqueue(
				"frappe_pulse.mobile.push.notify_event",
				queue="short",
				enqueue_after_commit=True,
				event=event,
			)
		elif outcome == "skipped":
			results["skipped"] += 1
		elif outcome == "schema_mismatch":
			results["schema_mismatch"] += 1
		else:
			results["failed"] += 1

	return results

"""
Conflict detection — called from protocol.apply_event before writing
a received event when a local version of the document already exists.
"""

import uuid

import frappe

# Fields that are never meaningful to diff
_META_FIELDS = frozenset([
	"name", "owner", "creation", "modified", "modified_by",
	"docstatus", "idx", "doctype", "__islocal", "__unsaved",
])


def find_conflicting_fields(local_doc: dict, remote_payload: dict) -> dict:
	"""
	Return a dict of fieldname → {local: val, remote: val}
	for every field that differs between local and remote.
	"""
	conflicts = {}
	all_keys = set(local_doc.keys()) | set(remote_payload.keys())

	for key in all_keys:
		if key in _META_FIELDS:
			continue
		local_val = local_doc.get(key)
		remote_val = remote_payload.get(key)
		if local_val != remote_val:
			conflicts[key] = {"local": local_val, "remote": remote_val}

	return conflicts


def create_conflict_record(
	doc_type: str,
	doc_name: str,
	local_event_id: str,
	remote_event_id: str,
	local_node_id: str,
	remote_node_id: str,
	conflicting_fields: dict,
	local_snapshot: dict,
	remote_snapshot: dict,
	strategy: str,
) -> str:
	"""Create a Pulse Conflict record and return its name."""
	conflict = frappe.new_doc("Pulse Conflict")
	conflict.conflict_id = str(uuid.uuid4())
	conflict.doc_type = doc_type
	conflict.doc_name = doc_name
	conflict.local_event_id = local_event_id
	conflict.remote_event_id = remote_event_id
	conflict.local_node_id = local_node_id
	conflict.remote_node_id = remote_node_id
	conflict.conflicting_fields = conflicting_fields
	conflict.local_snapshot = local_snapshot
	conflict.remote_snapshot = remote_snapshot
	conflict.resolution_strategy = strategy
	conflict.status = "PENDING"
	conflict.insert(ignore_permissions=True)
	frappe.db.commit()
	return conflict.name


def get_strategy(doc_type: str, fieldname: str = None) -> str:
	"""
	Look up the configured conflict resolution strategy for a DocType,
	optionally for a specific field override.
	Falls back to MANUAL if no config exists.
	"""
	try:
		config = frappe.get_doc("Pulse Conflict Config", doc_type)
		if not config.enabled:
			return "MANUAL"

		if fieldname and config.field_overrides:
			overrides = config.field_overrides
			if isinstance(overrides, str):
				import json
				overrides = json.loads(overrides)
			if fieldname in overrides:
				return overrides[fieldname]

		return config.default_strategy or "MANUAL"
	except frappe.DoesNotExistError:
		return "MANUAL"

# FILE: /Users/bluephoenix/frappe-bench/bench-r2/apps/frappe_pulse/frappe_pulse/conflict/strategies.py
"""
Three conflict resolution strategies:
  A. LAST_WRITE_WINS  — latest timestamp overwrites
  B. FIELD_MERGE      — merge non-conflicting fields, flag only real conflicts
  C. MANUAL           — create Pulse Conflict record, wait for human decision
"""

import frappe

from frappe_pulse.conflict.detector import (
	create_conflict_record,
	find_conflicting_fields,
	get_strategy,
)


def resolve(
	event: dict,
	local_doc_dict: dict,
	local_event_id: str,
) -> str:
	"""
	Entry point. Determine strategy and attempt resolution.
	Returns: 'applied' | 'pending_manual' | 'error'
	"""
	doc_type = event.get("doc_type", "")
	remote_payload = event.get("payload", {})
	remote_event_id = event.get("event_id", "")
	remote_node_id = event.get("node_id", "remote")
	local_node_id = frappe.db.get_single_value("Pulse Settings", "node_id") or "local"

	conflicting_fields = find_conflicting_fields(local_doc_dict, remote_payload)

	if not conflicting_fields:
		# No real conflict — apply directly
		return "no_conflict"

	# Determine strategy for this DocType
	strategy = get_strategy(doc_type)

	if strategy == "LAST_WRITE_WINS":
		return _last_write_wins(event, local_doc_dict, conflicting_fields)

	if strategy == "FIELD_MERGE":
		return _field_merge(
			event, local_doc_dict, conflicting_fields,
			local_event_id, remote_event_id,
			local_node_id, remote_node_id,
		)

	# Default: MANUAL
	return _manual(
		event, local_doc_dict, conflicting_fields,
		local_event_id, remote_event_id,
		local_node_id, remote_node_id,
	)


# ---------------------------------------------------------------------------
# Strategy A — Last Write Wins
# ---------------------------------------------------------------------------

def _last_write_wins(event: dict, local_doc_dict: dict, conflicting_fields: dict) -> str:
	"""
	Compare modified timestamps. Latest event wins entirely.
	If timestamps are equal or missing, remote wins (prefer incoming).
	"""
	import frappe.utils

	remote_modified = event.get("payload", {}).get("modified") or ""
	local_modified = local_doc_dict.get("modified") or ""

	remote_wins = not local_modified or remote_modified >= local_modified

	if remote_wins:
		_apply_resolved(event.get("doc_type"), event.get("payload", {}))
		_mark_auto_resolved(event, conflicting_fields, local_doc_dict, "LAST_WRITE_WINS")
		return "applied"
	else:
		# Local is newer — discard remote, just log
		_mark_auto_resolved(event, conflicting_fields, local_doc_dict, "LAST_WRITE_WINS")
		return "applied"


# ---------------------------------------------------------------------------
# Strategy B — Field-Level Merge
# ---------------------------------------------------------------------------

def _field_merge(
	event: dict,
	local_doc_dict: dict,
	conflicting_fields: dict,
	local_event_id: str,
	remote_event_id: str,
	local_node_id: str,
	remote_node_id: str,
) -> str:
	"""
	Merge fields that don't truly conflict.
	Fields with identical-type changes on both sides → flag for MANUAL.
	All other fields from remote → apply.
	"""
	doc_type = event.get("doc_type", "")
	remote_payload = event.get("payload", {})

	# Financial/critical fields that must never be auto-merged
	manual_fields = _get_manual_fields(doc_type)

	hard_conflicts = {k: v for k, v in conflicting_fields.items() if k in manual_fields}
	safe_to_merge = {k: v for k, v in conflicting_fields.items() if k not in manual_fields}

	# Build merged payload: start from local, apply safe remote fields
	merged = dict(local_doc_dict)
	for fieldname in safe_to_merge:
		merged[fieldname] = remote_payload.get(fieldname)

	# Apply the merged payload
	_apply_resolved(doc_type, merged)

	if hard_conflicts:
		# Create a conflict record only for the hard fields
		create_conflict_record(
			doc_type=doc_type,
			doc_name=remote_payload.get("name", ""),
			local_event_id=local_event_id,
			remote_event_id=remote_event_id,
			local_node_id=local_node_id,
			remote_node_id=remote_node_id,
			conflicting_fields=hard_conflicts,
			local_snapshot=local_doc_dict,
			remote_snapshot=remote_payload,
			strategy="FIELD_MERGE",
		)
		return "partial_conflict"

	_mark_auto_resolved(event, conflicting_fields, local_doc_dict, "FIELD_MERGE")
	return "applied"


# ---------------------------------------------------------------------------
# Strategy C — Manual
# ---------------------------------------------------------------------------

def _manual(
	event: dict,
	local_doc_dict: dict,
	conflicting_fields: dict,
	local_event_id: str,
	remote_event_id: str,
	local_node_id: str,
	remote_node_id: str,
) -> str:
	"""
	Create a Pulse Conflict record and leave resolution to an admin.
	The document is NOT changed until the admin resolves it.
	"""
	create_conflict_record(
		doc_type=event.get("doc_type", ""),
		doc_name=event.get("payload", {}).get("name", ""),
		local_event_id=local_event_id,
		remote_event_id=remote_event_id,
		local_node_id=local_node_id,
		remote_node_id=remote_node_id,
		conflicting_fields=conflicting_fields,
		local_snapshot=local_doc_dict,
		remote_snapshot=event.get("payload", {}),
		strategy="MANUAL",
	)
	return "pending_manual"


# ---------------------------------------------------------------------------
# Helpers
# ---------------------------------------------------------------------------

def _apply_resolved(doc_type: str, payload: dict) -> None:
	"""Write the resolved payload back to Frappe."""
	from frappe_pulse.sync.protocol import _apply_doc_event
	_apply_doc_event("UPDATE", doc_type, payload)


def _mark_auto_resolved(
	event: dict,
	conflicting_fields: dict,
	local_snapshot: dict,
	strategy: str,
) -> None:
	"""Record an auto-resolved conflict for the audit trail."""
	local_node_id = frappe.db.get_single_value("Pulse Settings", "node_id") or "local"
	conflict = frappe.new_doc("Pulse Conflict")
	conflict.conflict_id = __import__("uuid").uuid4().__str__()
	conflict.doc_type = event.get("doc_type", "")

	try:

		return {line.strip() for line in raw.splitlines() if line.strip()}

def _compute_checksum(payload: dict) -> str:
	canonical = json.dumps(payload, sort_keys=True, default=str)
	try:
		import pulse_core
		return pulse_core.compute_checksum(canonical)
	except ImportError:
		# Fallback to pure Python if Rust extension not built yet
		return hashlib.sha256(canonical.encode("utf-8")).hexdigest()


def _should_skip() -> bool:
	# Skip during install, migrate, or test setup
	return bool(
		frappe.flags.in_install
		or frappe.flags.in_migrate
		or frappe.flags.in_patch
		or frappe.flags.in_import
		or frappe.flags.in_test
	)


def _capture(event_type: str, doc) -> None:
	try:
		if _should_skip():
			return
		if not _is_capture_enabled():
			return
		if doc.doctype in EXCLUDED_DOCTYPES:
			return
		if doc.doctype in _extra_excluded_doctypes():
			return

		payload = doc.as_dict()
		checksum = _compute_checksum(payload)
		event = {
			"event_id": str(uuid.uuid4()),
			"event_type": event_type,
			"doc_type": doc.doctype,
			"doc_name": doc.name,
			"node_id": _get_node_id(),
			"schema_version": _get_schema_version(),
			"payload": payload,
			"checksum": checksum,
			"timestamp": datetime.now(timezone.utc).isoformat(),
		}

		# Write-before-ack: if node is offline, persist to SQLite durable store
		from frappe_pulse.offline.detector import is_offline
		if is_offline():
			from frappe_pulse.offline.store import OfflineStore
			OfflineStore.get().enqueue(event)
		else:
			frappe.enqueue(
				"frappe_pulse.workers.event_worker.process_event",
				queue="short",
				enqueue_after_commit=True,
				**event,
			)
	except Exception:
		# Never let capture failures break the main transaction
		pass


def on_create(doc, method=None):
	_capture("CREATE", doc)


def on_update(doc, method=None):
	_capture("UPDATE", doc)


def on_submit(doc, method=None):
	_capture("SUBMIT", doc)


def on_cancel(doc, method=None):
	_capture("CANCEL", doc)


def on_delete(doc, method=None):
	_capture("DELETE", doc)

	}

# FILE: /Users/bluephoenix/frappe-bench/bench-r2/apps/frappe_pulse/frappe_pulse/mobile/push.py
"""
Push notification sender for frappe-pulse.

Fires FCM (Android + iOS via Firebase) or APNs (iOS direct) after a
sync event is successfully applied on this node.

No separate backend needed — fires directly from the Python worker
after POST/PUT (receive endpoint) succeeds.

Requires:
  pip install google-auth requests  (for FCM v1)
  pip install httpx                 (for APNs HTTP/2)
"""

import json
import logging
import time

import frappe

logger = logging.getLogger("frappe_pulse.mobile.push")

# In-memory token cache: {device_token: platform}
# Populated via the register_device API endpoint.
# In production this should be stored in a DocType; this is a fast bootstrap.
_device_tokens: dict[str, str] = {}


# ── Public API ────────────────────────────────────────────────────────────────

def notify_event(event: dict) -> None:
	"""
	Called after a sync event is successfully applied.
	Fires push notifications to all registered devices if config allows.
	"""
	try:
		cfg = _get_config()
		if not cfg or not cfg.get("enabled"):
			return
		if not _should_notify(event, cfg):
			return

		title, body = _build_message(event)
		data = {
			"event_id": event.get("event_id", ""),
			"event_type": event.get("event_type", ""),
			"doc_type": event.get("doc_type", ""),
			"doc_name": event.get("doc_name", ""),
		}

		tokens = list(_device_tokens.keys())
		if not tokens:
			return

		if cfg.get("fcm_project_id") and cfg.get("fcm_server_key"):
			_send_fcm(cfg, tokens, title, body, data)
		else:
			# APNs direct for iOS tokens
			ios_tokens = [t for t, p in _device_tokens.items() if p == "ios"]
			if ios_tokens and cfg.get("apns_key_id"):
				_send_apns(cfg, ios_tokens, title, body, data)

	except Exception as exc:
		logger.warning("push: notify_event failed: %s", exc)


def register_device(device_token: str, platform: str) -> None:
	"""Register a device token (called from the mobile API endpoint)."""
	if platform not in ("android", "ios"):
		raise ValueError(f"Unknown platform: {platform}")
	_device_tokens[device_token] = platform
	logger.info("push: registered %s token %s…", platform, device_token[:12])


def unregister_device(device_token: str) -> None:
	_device_tokens.pop(device_token, None)


# ── FCM v1 (HTTP) ────────────────────────────────────────────────────────────

def _send_fcm(cfg: dict, tokens: list[str], title: str, body: str, data: dict) -> None:
	"""Send via Firebase Cloud Messaging v1 API (one request per token batch)."""
	try:
		import google.auth
		import google.auth.transport.requests
		import google.oauth2.service_account
		import requests
	except ImportError:
		logger.warning("push: google-auth / requests not installed — FCM skipped")
		return

	try:
		sa_json = json.loads(cfg["fcm_server_key"])
		credentials = google.oauth2.service_account.Credentials.from_service_account_info(
			sa_json,
			scopes=["https://www.googleapis.com/auth/firebase.messaging"],
		)
		credentials.refresh(google.auth.transport.requests.Request())
		access_token = credentials.token
	except Exception as exc:
		logger.error("push: FCM auth failed: %s", exc)
		return

	project_id = cfg["fcm_project_id"]
	url = f"https://fcm.googleapis.com/v1/projects/{project_id}/messages:send"
	headers = {"Authorization": f"Bearer {access_token}", "Content-Type": "application/json"}

	# FCM v1 sends one message per token
	for token in tokens:
		payload = {
			"message": {
				"token": token,
				"notification": {"title": title, "body": body},
				"data": {k: str(v) for k, v in data.items()},
				"android": {"priority": "high"},
				"apns": {"headers": {"apns-priority": "10"}},
			}
		}
		try:
			resp = requests.post(url, headers=headers, json=payload, timeout=5)
			if resp.status_code != 200:
				logger.warning("push: FCM %s → %s %s", token[:12], resp.status_code, resp.text[:200])
		except Exception as exc:
			logger.warning("push: FCM send error for %s: %s", token[:12], exc)


# ── APNs HTTP/2 (direct iOS) ─────────────────────────────────────────────────

def _send_apns(cfg: dict, tokens: list[str], title: str, body: str, data: dict) -> None:
	"""Send via Apple Push Notification service HTTP/2 API."""
	try:
		import httpx
		import jwt as pyjwt
	except ImportError:
		logger.warning("push: httpx / PyJWT not installed — APNs skipped")
		return

	key_id = cfg.get("apns_key_id", "")
	team_id = cfg.get("apns_team_id", "")
	bundle_id = cfg.get("apns_bundle_id", "")
	private_key = cfg.get("apns_private_key", "")

	if not all([key_id, team_id, bundle_id, private_key]):
		logger.warning("push: APNs config incomplete")
		return

	# JWT valid for 60 minutes
	issued_at = int(time.time())
	token = pyjwt.encode(
		{"iss": team_id, "iat": issued_at},
		private_key,
		algorithm="ES256",
		headers={"alg": "ES256", "kid": key_id},
	)

	apns_payload = {
		"aps": {
			"alert": {"title": title, "body": body},
			"sound": "default",
			"content-available": 1,
		},
		**{f"pulse_{k}": str(v) for k, v in data.items()},
	}

	host = "api.push.apple.com"
	with httpx.Client(http2=True, base_url=f"https://{host}") as client:
		for token_str in tokens:
			try:
				resp = client.post(
					f"/3/device/{token_str}",
					json=apns_payload,
					headers={
						"authorization": f"bearer {token}",
						"apns-topic": bundle_id,
						"apns-push-type": "alert",
					},
					timeout=5,
				)
				if resp.status_code != 200:
					logger.warning("push: APNs %s → %s", token_str[:12], resp.status_code)
			except Exception as exc:
				logger.warning("push: APNs error for %s: %s", token_str[:12], exc)


# ── Helpers ───────────────────────────────────────────────────────────────────

def _get_config() -> dict | None:
	try:
		doc = frappe.get_single("Pulse Push Config")
		return {
			"enabled": bool(doc.enabled),
			"fcm_server_key": doc.get_password("fcm_server_key") or "",
			"fcm_project_id": doc.fcm_project_id or "",
			"apns_key_id": doc.apns_key_id or "",
			"apns_team_id": doc.apns_team_id or "",
			"apns_bundle_id": doc.apns_bundle_id or "",
			"apns_private_key": doc.apns_private_key or "",
			"trigger_doctypes": _parse_lines(doc.trigger_doctypes),
			"trigger_event_types": doc.trigger_event_types or "ALL",
		}
	except Exception:
		return None


def _should_notify(event: dict, cfg: dict) -> bool:
	allowed_types = cfg.get("trigger_event_types", "ALL")
	if allowed_types != "ALL" and event.get("event_type") != allowed_types:
		return False
	trigger_doctypes = cfg.get("trigger_doctypes", set())
	if trigger_doctypes and event.get("doc_type") not in trigger_doctypes:
		return False
	return True


def _build_message(event: dict) -> tuple[str, str]:
	event_type = event.get("event_type", "UPDATED").title()
	doc_type = event.get("doc_type", "Document")
	doc_name = event.get("doc_name", "")
	return f"{doc_type} {event_type}", f"{doc_name} was {event_type.lower()} on another node"


def _parse_lines(text: str | None) -> set[str]:
	if not text:
		return set()
	return {l.strip() for l in text.splitlines() if l.strip()}

# FILE: /Users/bluephoenix/frappe-bench/bench-r2/apps/frappe_pulse/frappe_pulse/mobile/response.py
"""
Bandwidth-optimized mobile response layer.

When ?_mobile=1 is sent by the Flutter client:
  - Non-essential fields are stripped from event payloads
  - Response is serialized as MessagePack (binary, ~30% smaller than JSON)
  - Falls back to JSON if msgpack not installed

Field stripping: keeps only fields the mobile client needs to render
the document. Strips audit fields, internal Frappe metadata, and any
field listed in the DocType's mobile_strip_fields config (future).
"""

import json
import logging

import frappe

logger = logging.getLogger("frappe_pulse.mobile")

# Fields stripped from every payload when ?_mobile=1
_STRIP_ALWAYS = frozenset([
	"creation", "modified", "modified_by", "owner",
	"__islocal", "__unsaved", "__run_link_triggers",
	"docstatus", "idx", "naming_series",
	"_user_tags", "_comments", "_assign", "_liked_by",
	"meta_image", "form_grid_templates",
])

# Fields kept even if heavy — mobile needs these for display
_KEEP_ALWAYS = frozenset([
	"name", "doctype", "status", "workflow_state",
	"company", "customer", "supplier",
	"grand_total", "total", "outstanding_amount",
	"posting_date", "transaction_date", "due_date",
])


def strip_payload(payload: dict) -> dict:
	"""Remove non-essential fields from a document payload."""
	if not isinstance(payload, dict):
		return payload
	return {
		k: v for k, v in payload.items()
		if k in _KEEP_ALWAYS or k not in _STRIP_ALWAYS
	}


def encode_mobile_response(data: dict) -> tuple[bytes, str]:
	"""
	Encode response as MessagePack if available, else JSON.
	Returns (body_bytes, content_type).
	"""
	try:
		import msgpack
		body = msgpack.packb(data, use_bin_type=True, default=_msgpack_default)
		return body, "application/msgpack"
	except ImportError:
		body = json.dumps(data, default=str).encode("utf-8")
		return body, "application/json"


def _msgpack_default(obj):
	"""Fallback encoder for non-serializable types."""
	if hasattr(obj, "isoformat"):
		return obj.isoformat()
	return str(obj)


def is_mobile_request() -> bool:
	"""True if the current request has ?_mobile=1."""
	try:
		return bool(frappe.local.form_dict.get("_mobile"))
	except Exception:
		return False

"""
Pulse Metrics collector — runs every 60 seconds via scheduler.
Writes one Pulse Node Metric row per node per tick.
Fires alerts if thresholds are breached.
"""

import logging
from datetime import datetime, timezone

import frappe

logger = logging.getLogger("frappe_pulse.observability")

# Alert thresholds
LAG_ALERT_SECONDS = 600       # 10 minutes
BACKLOG_ALERT_COUNT = 500
ERROR_STREAK_ALERT = 3


def collect_metrics() -> None:
	"""Collect a metric snapshot for every registered node + this local node."""
	now = datetime.now(timezone.utc).isoformat()

	# ── Local node metrics ────────────────────────────────────────────────
	local_node_id = frappe.db.get_single_value("Pulse Settings", "node_id") or "local"
	local_node_name = frappe.db.get_single_value("Pulse Settings", "node_name") or frappe.local.site

	pending = frappe.db.count("Pulse Event Log", {"sync_status": "PENDING"})
	failed = frappe.db.count("Pulse Event Log", {"sync_status": "FAILED"})
	synced_hour = _synced_last_hour()
	lag = _oldest_pending_lag_seconds()
	offline_q = _offline_queue_count()

	_write_metric(
		node_id=local_node_id,
		node_name=local_node_name,
		collected_at=now,
		pending=pending,
		failed=failed,
		synced_hour=synced_hour,
		lag=lag,
		offline_q=offline_q,
		error_streak=0,
	)

	# ── Remote node metrics ───────────────────────────────────────────────
	nodes = frappe.get_all(
		"Pulse Node",
		filters={"enabled": 1},
		fields=["node_id", "node_name", "sync_status", "last_error", "last_synced_at"],
	)

	for node in nodes:
		streak = _error_streak(node.node_id)
		node_lag = _node_lag_seconds(node.last_synced_at)

		_write_metric(
			node_id=node.node_id,
			node_name=node.node_name,
			collected_at=now,
			pending=0,
			failed=0,
			synced_hour=0,
			lag=node_lag,
			offline_q=0,
			error_streak=streak,
		)

		_check_alerts(node.node_id, node.node_name, lag=node_lag, backlog=0, error_streak=streak)

	_check_alerts(local_node_id, local_node_name, lag=lag, backlog=pending, error_streak=0)
	frappe.db.commit()


def _write_metric(**kw) -> None:
	m = frappe.new_doc("Pulse Node Metric")
	m.node_id = kw["node_id"]
	m.node_name = kw["node_name"]
	m.collected_at = kw["collected_at"]
	m.pending_events = kw["pending"]
	m.failed_events = kw["failed"]
	m.synced_last_hour = kw["synced_hour"]
	m.lag_seconds = kw["lag"]
	m.offline_queue = kw["offline_q"]
	m.error_streak = kw["error_streak"]
	m.alert_fired = int(
		kw["lag"] > LAG_ALERT_SECONDS
		or kw["backlog"] > BACKLOG_ALERT_COUNT
		if "backlog" in kw else kw["lag"] > LAG_ALERT_SECONDS
		or kw["error_streak"] >= ERROR_STREAK_ALERT
	)
	m.insert(ignore_permissions=True)


def _check_alerts(node_id: str, node_name: str, lag: float, backlog: int, error_streak: int) -> None:
	alerts = []
	if lag > LAG_ALERT_SECONDS:
		alerts.append(f"sync lag {lag:.0f}s > {LAG_ALERT_SECONDS}s")
	if backlog > BACKLOG_ALERT_COUNT:
		alerts.append(f"backlog {backlog} > {BACKLOG_ALERT_COUNT}")
	if error_streak >= ERROR_STREAK_ALERT:
		alerts.append(f"error streak {error_streak}")
	for msg in alerts:
		logger.warning("ALERT node=%s (%s): %s", node_id, node_name, msg)
		try:
			frappe.publish_realtime(
				"pulse_alert",
				{"node_id": node_id, "node_name": node_name, "message": msg},
				user="Administrator",
			)
		except Exception:
			pass


def _synced_last_hour() -> int:
	result = frappe.db.sql(
		"""
		SELECT COUNT(*) FROM `tabPulse Event Log`
		WHERE sync_status = 'SYNCED'
		  AND timestamp >= DATE_SUB(NOW(), INTERVAL 1 HOUR)
		"""
	)
	return result[0][0] if result else 0


def _oldest_pending_lag_seconds() -> float:
	result = frappe.db.sql(
		"""
		SELECT TIMESTAMPDIFF(SECOND, MIN(timestamp), NOW())
		FROM `tabPulse Event Log`
		WHERE sync_status = 'PENDING'
		"""
	)
	val = result[0][0] if result else None
	return float(val) if val else 0.0


def _node_lag_seconds(last_synced_at) -> float:
	if not last_synced_at:
		return 9999.0
	result = frappe.db.sql(

"""
Offline mode detector.

Checks whether this node can reach any enabled remote Pulse Node.
Result is cached for 60 seconds to avoid hammering on every event capture.
"""

import socket
import time
from urllib.parse import urlparse

import frappe

_cache: dict = {"result": None, "checked_at": 0.0}
_CACHE_TTL = 60  # seconds
_CONNECT_TIMEOUT = 3  # seconds


def _parse_host_port(endpoint_url: str) -> tuple[str, int] | None:
    try:
        parsed = urlparse(endpoint_url)
        host = parsed.hostname or ""
        port = parsed.port or (443 if parsed.scheme == "https" else 80)
        return host, port
    except Exception:
        return None


def _is_reachable(host: str, port: int) -> bool:
    try:
        with socket.create_connection((host, port), timeout=_CONNECT_TIMEOUT):
            return True
    except OSError:
        return False


def is_offline() -> bool:
    """
    Return True if this node cannot reach any enabled remote peer.
    Result cached for _CACHE_TTL seconds.
    """
    now = time.monotonic()
    if _cache["result"] is not None and (now - _cache["checked_at"]) < _CACHE_TTL:
        return _cache["result"]

    result = _check_connectivity()
    _cache["result"] = result
    _cache["checked_at"] = now
    return result


def _check_connectivity() -> bool:
    try:
        nodes = frappe.get_all(
            "Pulse Node",
            filters={"enabled": 1, "sync_status": ("!=", "PAUSED")},
            fields=["endpoint_url"],
            limit=5,
        )
    except Exception:
        return False  # Can't even query DB — assume online

    if not nodes:
        return False  # No peers configured — don't go offline

    for node in nodes:
        hp = _parse_host_port(node.endpoint_url or "")
        if hp and _is_reachable(*hp):
            return False  # At least one peer reachable

    return True  # All peers unreachable


def invalidate_cache() -> None:
    """Force re-check on next call (call after reconnect or node config change)."""
    _cache["result"] = None
    _cache["checked_at"] = 0.0

"""
On-reconnect flush: drains the SQLite offline store into the normal sync pipeline.

Called by the scheduler every 5 minutes (same cadence as sync worker).
If we were offline and are now reachable, this fires immediately.
"""

import json
import logging

import frappe

from frappe_pulse.offline.detector import invalidate_cache, is_offline
from frappe_pulse.offline.store import OfflineStore
from frappe_pulse.sync.outbound import push_to_node

logger = logging.getLogger("frappe_pulse.offline")

BATCH_SIZE = 100


def flush_offline_queue() -> None:
    """
    Drain the offline SQLite store to all enabled remote nodes.
    No-op if still offline. Called by the scheduler.
    """
    invalidate_cache()  # always re-check at flush time
    if is_offline():
        return

    store = OfflineStore.get()
    pending = store.pending(limit=BATCH_SIZE)
    if not pending:
        store.purge_flushed()
        return

    logger.info("offline.flush: %d events to flush", len(pending))

    nodes = frappe.get_all(
        "Pulse Node",
        filters={"enabled": 1, "sync_status": ("!=", "PAUSED")},
        fields=["name", "endpoint_url", "api_key", "sync_direction"],
    )

    flushed_ids: list[str] = []

    for node in nodes:
        if node.sync_direction not in ("PUSH", "BIDIRECTIONAL"):
            continue
        node_doc = frappe.get_doc("Pulse Node", node.name)
        try:
            # Temporarily write offline events into Pulse Event Log as PENDING
            # so push_to_node can pick them up normally
            _promote_to_event_log(pending)
            push_to_node(node_doc, batch_size=BATCH_SIZE)
            flushed_ids.extend(e["event_id"] for e in pending)
        except Exception as exc:
            logger.warning("offline.flush: push to %s failed: %s", node.name, exc)

    if flushed_ids:
        store.mark_flushed(list(set(flushed_ids)))
        logger.info("offline.flush: marked %d events flushed", len(flushed_ids))

    store.purge_flushed()


def _promote_to_event_log(events: list[dict]) -> None:
    """Insert offline events into Pulse Event Log as PENDING if not already present."""
    for event in events:
        if frappe.db.exists("Pulse Event Log", {"event_id": event["event_id"]}):
            continue
        log = frappe.new_doc("Pulse Event Log")
        log.event_id = event["event_id"]
        log.event_type = event["event_type"]
        log.doc_type = event["doc_type"]
        log.doc_name = event["doc_name"]
        log.node_id = event["node_id"]
        log.schema_version = event.get("schema_version", "")
        log.payload = json.dumps(event.get("payload", {}), default=str)
        log.checksum = event.get("checksum", "")
        log.timestamp = event.get("timestamp", "")
        log.sync_status = "PENDING"
        log.insert(ignore_permissions=True)
    frappe.db.commit()

SQLite-backed durable event store for offline edge nodes.

(write-before-ack) instead of the normal Frappe DB queue.  On reconnect,

# Path to the compiled VFS extension (.so / .dylib).
# Built with:  cd frappe_pulse/offline/vfs && make
_VFS_SO = Path(__file__).parent / ".." / ".." / ".." / "pulse_vfs.so"
_VFS_SO = _VFS_SO.resolve()
_VFS_AVAILABLE: bool | None = None  # None = not yet checked


def _try_load_vfs(conn: sqlite3.Connection) -> bool:
    """
    Attempt to load the pulse_vfs C extension into conn.
    Returns True if loaded, False if .so not built (pure Python fallback used).
    Pure Python WAL + PRAGMA synchronous=NORMAL remains active either way.
    """
    global _VFS_AVAILABLE
    if _VFS_AVAILABLE is False:
        return False
    if not _VFS_SO.exists():
        if _VFS_AVAILABLE is None:
            logger.debug("pulse_vfs.so not found — using pure Python SQLite (build offline/vfs/ to enable)")
        _VFS_AVAILABLE = False
        return False
    try:
        conn.enable_load_extension(True)
        conn.load_extension(str(_VFS_SO.with_suffix("")))  # strip .so — SQLite adds it
        conn.enable_load_extension(False)
        if _VFS_AVAILABLE is None:
            logger.info("pulse_vfs C extension loaded — forced fsync active")
        _VFS_AVAILABLE = True
        return True
    except Exception as exc:
        conn.enable_load_extension(False)
        logger.warning("pulse_vfs load failed (%s) — pure Python SQLite in use", exc)
        _VFS_AVAILABLE = False
        return False

    site_path = Path(frappe.get_site_path())

class OfflineStore:
    """Thread-safe SQLite event store."""

    _instances: dict[str, "OfflineStore"] = {}
    _lock = threading.Lock()

    @classmethod
    def get(cls) -> "OfflineStore":
        site = frappe.local.site
        with cls._lock:
            if site not in cls._instances:
                cls._instances[site] = cls(_db_path())
            return cls._instances[site]

    def __init__(self, path: Path):
        path.parent.mkdir(parents=True, exist_ok=True)
        self._path = str(path)
        self._local = threading.local()
        self._init_schema()

    def _conn(self) -> sqlite3.Connection:
        if not hasattr(self._local, "conn"):
            conn = sqlite3.connect(self._path, check_same_thread=False)
            conn.row_factory = sqlite3.Row
            conn.execute("PRAGMA journal_mode=WAL")
            # If the C VFS extension is loaded it forces SQLITE_SYNC_FULL on
            # every xSync call, making FULL redundant but harmless.  Without it
            # NORMAL is acceptable — WAL already guarantees no torn pages.
            vfs_loaded = _try_load_vfs(conn)
            conn.execute(f"PRAGMA synchronous={'FULL' if vfs_loaded else 'NORMAL'}")
            self._local.conn = conn
        return self._local.conn

    def _init_schema(self) -> None:
        conn = self._conn()
        conn.executescript(_CREATE_TABLE)
        conn.commit()

    def enqueue(self, event: dict) -> None:
        """Write an event to the durable store (write-before-ack)."""
        conn = self._conn()
        conn.execute(
            """
            INSERT OR IGNORE INTO offline_events
                (event_id, event_type, doc_type, doc_name, node_id,
                 schema_version, payload, checksum, timestamp)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
            """,
            (
                event["event_id"],
                event["event_type"],
                event["doc_type"],
                event["doc_name"],
                event.get("node_id", "local"),
                event.get("schema_version", ""),
                json.dumps(event.get("payload", {}), default=str),
                event.get("checksum", ""),
                str(event.get("timestamp", "")),
            ),
        )
        conn.commit()

    def pending(self, limit: int = 500) -> list[dict]:
        """Return unflushed events in timestamp order."""
        conn = self._conn()
        rows = conn.execute(
            """
            SELECT event_id, event_type, doc_type, doc_name, node_id,
                   schema_version, payload, checksum, timestamp
            FROM offline_events
            WHERE flushed = 0
            ORDER BY timestamp ASC
            LIMIT ?
            """,
            (limit,),
        ).fetchall()
        result = []
        for row in rows:
            e = dict(row)
            try:
                e["payload"] = json.loads(e["payload"])
            except Exception:
                e["payload"] = {}
            result.append(e)
        return result

    def mark_flushed(self, event_ids: list[str]) -> None:
        """Mark a batch of events as successfully flushed."""
        if not event_ids:
            return
        conn = self._conn()
        placeholders = ",".join("?" * len(event_ids))
        conn.execute(
            f"UPDATE offline_events SET flushed = 1 WHERE event_id IN ({placeholders})",
            event_ids,
        )
        conn.commit()

    def count_pending(self) -> int:
        conn = self._conn()
        return conn.execute(
            "SELECT COUNT(*) FROM offline_events WHERE flushed = 0"
        ).fetchone()[0]

    def purge_flushed(self, older_than_days: int = 7) -> int:
        """Delete already-flushed events older than N days. Returns count deleted."""
        conn = self._conn()
        cur = conn.execute(
            """
            DELETE FROM offline_events
            WHERE flushed = 1
              AND datetime(timestamp) < datetime('now', ? || ' days')
            """,
            (f"-{older_than_days}",),
        )
        conn.commit()
        return cur.rowcount

def run_replay_job(replay_id: str) -> None:
	"""Background job — executes one replay job to completion."""
	job_name = frappe.db.get_value("Pulse Replay Job", {"replay_id": replay_id}, "name")
	if not job_name:
		logger.error("replay: job %s not found", replay_id)
		return

	job = frappe.get_doc("Pulse Replay Job", job_name)
	job.db_set("status", "RUNNING")
	job.db_set("started_at", datetime.now(timezone.utc).isoformat())
	frappe.db.commit()

	filters = _build_filters(job)
	total = frappe.db.count("Pulse Event Log", filters)
	job.db_set("total_events", total)
	frappe.db.commit()

	applied = skipped = failed = 0
	offset = 0
	local_schema = get_local_schema_version()

	try:
		while True:
			events = frappe.get_all(
				"Pulse Event Log",
				filters=filters,
				fields=[
					"event_id", "event_type", "doc_type", "doc_name",
					"node_id", "schema_version", "payload", "checksum", "timestamp",
				],
				order_by="timestamp asc",
				limit=BATCH_SIZE,
				start=offset,
			)
			if not events:
				break

			for event in events:
				if isinstance(event.get("payload"), str):
					try:
						event["payload"] = json.loads(event["payload"])
					except Exception:
						event["payload"] = {}
				event["timestamp"] = str(event["timestamp"])

				if job.dry_run:
					# Dry run — only count, don't apply
					skipped += 1
					continue

				outcome = apply_event(event, local_schema)
				if outcome == "applied":
					applied += 1
				elif outcome == "skipped":
					skipped += 1
				else:
					failed += 1

			offset += BATCH_SIZE
			# Update progress after each batch
			job.db_set("applied_events", applied)
			job.db_set("skipped_events", skipped)
			job.db_set("failed_events", failed)
			frappe.db.commit()

		job.db_set("status", "COMPLETE")
	except Exception as exc:
		logger.error("replay: job %s failed: %s", replay_id, exc)
		job.db_set("status", "FAILED")
		job.db_set("error_log", str(exc)[:1000])

	job.db_set("completed_at", datetime.now(timezone.utc).isoformat())
	frappe.db.commit()
	logger.info(
		"replay: job %s done — applied=%d skipped=%d failed=%d dry_run=%s",
		replay_id, applied, skipped, failed, bool(job.dry_run),
	)


def _build_filters(job) -> dict:
	filters: dict = {}
	if job.from_timestamp:
		filters["timestamp"] = (">=", job.from_timestamp)
	if job.to_timestamp:
		existing = filters.get("timestamp")
		if existing:
			# frappe doesn't support AND on same field easily — use raw SQL workaround
			filters["timestamp"] = ("between", [job.from_timestamp, job.to_timestamp])
		else:
			filters["timestamp"] = ("<=", job.to_timestamp)
	if job.filter_doc_type:
		filters["doc_type"] = job.filter_doc_type
	if job.filter_node_id:
		filters["node_id"] = job.filter_node_id
	return filters

def run_retention() -> None:

	# Find all distinct doc_types in SYNCED events

def _process_doctype(doc_type: str, retain_days: int, archive: bool, archive_dir: str) -> None:
	offset = 0
	deleted_total = 0

	while True:
		events = frappe.db.sql(
			"""
			SELECT name, event_id, event_type, doc_name, payload, timestamp
			FROM `tabPulse Event Log`
			WHERE sync_status = 'SYNCED'
			  AND doc_type = %s
			  AND timestamp < DATE_SUB(NOW(), INTERVAL %s DAY)
			ORDER BY timestamp ASC
			LIMIT %s
			""",
			(doc_type, retain_days, BATCH_SIZE),
			as_dict=True,
		)
		if not events:
			break

		if archive:
			_archive_batch(doc_type, events, archive_dir)

		frappe.db.delete("Pulse Event Log", {"name": ("in", names)})
		deleted_total += len(names)

		if len(events) < BATCH_SIZE:
			break

	if deleted_total:
		logger.info("retention: %s — deleted %d events (retain=%d days)", doc_type, deleted_total, retain_days)

	# Rust compaction on remaining events for this doctype
	_compact_doctype(doc_type)


def _archive_batch(doc_type: str, events: list, archive_dir: str) -> None:
	"""Append events to a .jsonl.gz archive file."""
	dir_path = Path(archive_dir)
	dir_path.mkdir(parents=True, exist_ok=True)

	date_str = datetime.now(timezone.utc).strftime("%Y-%m")
	safe_name = doc_type.replace(" ", "_").lower()
	file_path = dir_path / f"{safe_name}_{date_str}.jsonl.gz"

	try:
		with gzip.open(file_path, "at", encoding="utf-8") as f:
			for event in events:
				row = {
					"event_id": event.event_id,
					"event_type": event.event_type,
					"doc_type": doc_type,
					"doc_name": event.doc_name,
					"timestamp": str(event.timestamp),
					"payload": event.payload if isinstance(event.payload, dict)
					           else _try_parse(event.payload),
				}
				f.write(json.dumps(row, default=str) + "\n")
	except Exception as exc:
		logger.error("retention: archive write failed for %s: %s", doc_type, exc)


def _compact_doctype(doc_type: str) -> None:
	"""Use Rust pulse_core to compact SYNCED events for this DocType."""
	try:
		import pulse_core

		events = frappe.db.sql(
			"""
			SELECT event_id, event_type, doc_type, doc_name,
			       node_id, schema_version, payload, checksum, timestamp, sync_status
			FROM `tabPulse Event Log`
			WHERE doc_type = %s AND sync_status = 'SYNCED'
			LIMIT 5000
			""",
			(doc_type,),
			as_dict=True,
		)
		if not events:
			return

		for e in events:
			if isinstance(e.get("payload"), dict):
				e["payload"] = json.dumps(e["payload"], default=str)
			e["timestamp"] = str(e["timestamp"])

		events_json = json.dumps([dict(e) for e in events], default=str)
		savings = pulse_core.compaction_savings(events_json)

		if savings > 0:
			compacted_json = pulse_core.compact_events(events_json)
			compacted = json.loads(compacted_json)
			keep_ids = {e["event_id"] for e in compacted}
			all_ids = {e["event_id"] for e in events}
			drop_ids = list(all_ids - keep_ids)

			if drop_ids:
				frappe.db.delete("Pulse Event Log", {"event_id": ("in", drop_ids)})
				logger.info("retention: compacted %s — removed %d redundant events", doc_type, len(drop_ids))

	except ImportError:
		pass  # Rust extension not built — skip compaction
	except Exception as exc:
		logger.warning("retention: compaction error for %s: %s", doc_type, exc)

"""
Selective sync scope filter.

Called server-side on the pull endpoint to filter which events
a requesting node is allowed to receive.
"""

import json
import logging

import frappe

logger = logging.getLogger("frappe_pulse.selective")


def _get_scope(node_id: str) -> dict | None:
	"""Load the active Pulse Sync Scope for node_id, merging template if set."""
	scope_name = frappe.db.get_value(
		"Pulse Sync Scope", {"node": node_id, "enabled": 1}, "name"
	)
	if not scope_name:
		return None

	scope_doc = frappe.get_doc("Pulse Sync Scope", scope_name)

	allowed_doctypes: set[str] = set()
	blocked_doctypes: set[str] = set()
	field_filters: dict = {}

	# Apply template first (lower priority)
	if scope_doc.scope_template:
		try:
			tmpl = frappe.get_doc("Pulse Scope Template", scope_doc.scope_template)
			allowed_doctypes |= _parse_lines(tmpl.allowed_doctypes)
			blocked_doctypes |= _parse_lines(tmpl.blocked_doctypes)
			field_filters.update(_parse_json(tmpl.field_filters))
		except Exception:
			pass

	# Scope overrides template
	allowed_doctypes |= _parse_lines(scope_doc.allowed_doctypes)
	blocked_doctypes |= _parse_lines(scope_doc.blocked_doctypes)
	field_filters.update(_parse_json(scope_doc.field_filters))

	return {
		"allowed_doctypes": allowed_doctypes,
		"blocked_doctypes": blocked_doctypes,
		"field_filters": field_filters,
	}


def filter_events(events: list[dict], node_id: str) -> list[dict]:
	"""
	Return only the events that node_id is scoped to receive.
	If no scope is configured for the node, all events pass through.
	"""
	scope = _get_scope(node_id)
	if scope is None:
		return events

	allowed = scope["allowed_doctypes"]
	blocked = scope["blocked_doctypes"]
	field_filters = scope["field_filters"]

	filtered = []
	for event in events:
		doc_type = event.get("doc_type", "")

		if doc_type in blocked:
			continue
		if allowed and doc_type not in allowed:
			continue

		# Field-level filter: e.g. {"Sales Invoice": {"company": "Branch A"}}
		if doc_type in field_filters:
			payload = event.get("payload", {})
			rules = field_filters[doc_type]
			if not _matches_field_rules(payload, rules):
				continue

		filtered.append(event)

	before = len(events)
	after = len(filtered)
	if before != after:
		logger.debug("scope filter: node=%s kept %d/%d events", node_id, after, before)

	return filtered


def _matches_field_rules(payload: dict, rules: dict) -> bool:
	"""Return True if payload matches all field=value rules."""
	for field, expected in rules.items():
		actual = payload.get(field)
		if isinstance(expected, list):
			if actual not in expected:
				return False
		else:
			if actual != expected:
				return False
	return True


def _parse_lines(text: str | None) -> set[str]:
	if not text:
		return set()
	return {line.strip() for line in text.splitlines() if line.strip()}


def _parse_json(text: str | None) -> dict:
	if not text:
		return {}
	try:
		return json.loads(text)
	except Exception:
		return {}

def pull_from_node(node: "frappe.Document", batch_size: int = 100) -> dict:
	"""
	Pull events from the remote node that arrived after our last sync timestamp.
	Returns a result dict with counts.
	"""
	after = str(node.last_synced_at or "2000-01-01 00:00:00")

	try:
		events = _fetch_events(node, after, batch_size)
	except Exception as exc:
		node.mark_error(str(exc))
		return {"pulled": 0, "applied": 0, "skipped": 0, "schema_mismatch": 0, "error": str(exc)}

	if not events:
		return {"pulled": 0, "applied": 0, "skipped": 0, "schema_mismatch": 0}

	local_schema = get_local_schema_version()
	results = {"pulled": len(events), "applied": 0, "skipped": 0, "schema_mismatch": 0, "error_count": 0}

	for event in events:
		outcome = apply_event(event, local_schema)
		if outcome == "applied":
			results["applied"] += 1
		elif outcome == "skipped":
			results["skipped"] += 1
		elif outcome == "schema_mismatch":
			results["schema_mismatch"] += 1
		else:
			results["error_count"] += 1

	# Update schema mismatch count on node record
	if results["schema_mismatch"] > 0:
		current = node.schema_mismatch_count or 0
		frappe.db.set_value(
			"Pulse Node", node.name, "schema_mismatch_count", current + results["schema_mismatch"]
		)

	return results

def push_to_node(node: "frappe.Document", batch_size: int = 100) -> dict:
	"""
	Push local PENDING events to the given Pulse Node.
	Returns a result dict with counts.
	"""
	pending = frappe.get_all(
		"Pulse Event Log",
		filters={"sync_status": "PENDING"},
		fields=[
			"name", "event_id", "event_type", "doc_type", "doc_name",
			"node_id", "schema_version", "payload", "checksum", "timestamp",
		],
		order_by="timestamp asc",
		limit=batch_size,
	)

	if not pending:
		return {"pushed": 0, "applied": 0, "failed": 0}

	events = []
	for row in pending:
		events.append({
			"event_id": row.event_id,
			"event_type": row.event_type,
			"doc_type": row.doc_type,
			"doc_name": row.doc_name,
			"node_id": row.node_id,
			"schema_version": row.schema_version,
			"payload": row.payload if isinstance(row.payload, dict) else json.loads(row.payload or "{}"),
			"checksum": row.checksum,
			"timestamp": str(row.timestamp),
		})

	try:
		resp = _post_events(node, events)
	except Exception as exc:
		node.mark_error(str(exc))
		return {"pushed": len(events), "applied": 0, "failed": len(events), "error": str(exc)}

	applied = resp.get("applied", 0)
	failed = resp.get("failed", 0)

	# Mark pushed events as SYNCED
	event_ids = [e["event_id"] for e in events]
	if event_ids:
		frappe.db.sql(
			"""
			UPDATE `tabPulse Event Log`
			SET sync_status = 'SYNCED'
			WHERE event_id IN ({placeholders})
			""".format(placeholders=", ".join(["%s"] * len(event_ids))),
			event_ids,
		)
		frappe.db.commit()

	return {"pushed": len(events), "applied": applied, "failed": failed}

"""
Shared helpers for the sync protocol — building event batches,
applying received events, and schema version comparison.
"""

import hashlib
import json

import frappe


def get_local_schema_version() -> str:
	try:
		return (
			frappe.db.get_value("Installed Applications", {"app_name": "frappe"}, "app_version")
			or ""
		)
	except Exception:
		return ""


def is_schema_compatible(local_version: str, remote_version: str) -> bool:
	"""
	Compare major.minor only — patch differences are safe to apply.
	Empty version strings are treated as compatible (legacy nodes).
	"""
	if not local_version or not remote_version:
		return True
	try:
		local_parts = [int(x) for x in local_version.split(".")[:2]]
		remote_parts = [int(x) for x in remote_version.split(".")[:2]]
		return local_parts >= remote_parts
	except (ValueError, AttributeError):
		return True


def verify_checksum(payload: dict, checksum: str) -> bool:
	canonical = json.dumps(payload, sort_keys=True, default=str)
	try:
		import pulse_core
		return pulse_core.verify_checksum(canonical, checksum)
	except ImportError:
		return hashlib.sha256(canonical.encode("utf-8")).hexdigest() == checksum


def apply_event(event: dict, local_schema_version: str) -> str:
	"""
	Apply a single received event to the local Frappe instance.
	Returns: 'applied' | 'skipped' | 'schema_mismatch' | 'error'
	"""
	event_id = event.get("event_id", "")
	doc_type = event.get("doc_type", "")
	event_type = event.get("event_type", "")
	payload = event.get("payload", {})
	remote_schema = event.get("schema_version", "")
	checksum = event.get("checksum", "")

	# Skip if already applied (idempotent)
	if frappe.db.exists("Pulse Event Log", {"event_id": event_id, "sync_status": "SYNCED"}):
		return "skipped"

	# Schema compatibility check
	if not is_schema_compatible(local_schema_version, remote_schema):
		_record_event(event, "SCHEMA_MISMATCH")
		return "schema_mismatch"

	# Checksum verification
	if checksum and not verify_checksum(payload, checksum):
		frappe.logger("frappe_pulse").warning(
			f"Pulse: checksum mismatch for event {event_id} ({doc_type}/{payload.get('name')})"
		)
		_record_event(event, "FAILED")
		return "error"

	try:
		# Conflict detection — only for UPDATE events on existing docs
		if event_type in ("UPDATE", "SUBMIT", "CANCEL") and frappe.db.exists(doc_type, payload.get("name")):
			from frappe_pulse.conflict.strategies import resolve as resolve_conflict

			local_doc = frappe.get_doc(doc_type, payload.get("name")).as_dict()
			local_event = frappe.db.get_value(
				"Pulse Event Log",
				{"doc_type": doc_type, "doc_name": payload.get("name"), "sync_status": "SYNCED"},
				"event_id",
				order_by="timestamp desc",
			) or ""

			conflict_outcome = resolve_conflict(event, local_doc, local_event)

			if conflict_outcome == "no_conflict":
				_apply_doc_event(event_type, doc_type, payload)
			elif conflict_outcome == "pending_manual":
				_record_event(event, "FAILED")
				return "conflict"
			# applied / partial_conflict — resolution already handled inside strategy

		else:
			_apply_doc_event(event_type, doc_type, payload)

		_record_event(event, "SYNCED")
		return "applied"
	except Exception as exc:
		frappe.logger("frappe_pulse").error(
			f"Pulse: failed to apply event {event_id}: {exc}"
		)
		_record_event(event, "FAILED")
		return "error"


def _apply_doc_event(event_type: str, doc_type: str, payload: dict) -> None:
	"""Apply a document change from a received event."""
	name = payload.get("name")
	if not name:
		return

	if event_type == "DELETE":
		if frappe.db.exists(doc_type, name):
			frappe.delete_doc(doc_type, name, ignore_permissions=True, force=True)
		return

	if event_type in ("CREATE", "UPDATE", "SUBMIT", "CANCEL"):
		if frappe.db.exists(doc_type, name):
			doc = frappe.get_doc(doc_type, name)
			_update_doc_fields(doc, payload)
			doc.flags.ignore_permissions = True
			doc.flags.ignore_validate = True
			doc.flags.ignore_mandatory = True
			doc.flags.from_pulse_sync = True
			doc.save()
		else:
			doc = frappe.new_doc(doc_type)
			_update_doc_fields(doc, payload)
			doc.flags.ignore_permissions = True
			doc.flags.ignore_validate = True
			doc.flags.ignore_mandatory = True
			doc.flags.from_pulse_sync = True
			doc.insert()

	frappe.db.commit()


def _update_doc_fields(doc, payload: dict) -> None:
	"""Safely update doc fields from payload, skipping meta fields."""
	skip = frozenset([
		"creation", "modified", "modified_by", "owner",
		"doctype", "__islocal", "__unsaved",
	])
	for key, value in payload.items():
		if key not in skip and hasattr(doc, key):
			setattr(doc, key, value)


def _record_event(event: dict, sync_status: str) -> None:
	"""Write the received event into local Pulse Event Log."""
	try:
		if frappe.db.exists("Pulse Event Log", {"event_id": event.get("event_id")}):
			frappe.db.set_value(
				"Pulse Event Log",
				{"event_id": event.get("event_id")},
				"sync_status",
				sync_status,
			)
		else:
			log = frappe.new_doc("Pulse Event Log")
			log.event_id = event.get("event_id")
			log.event_type = event.get("event_type")
			log.doc_type = event.get("doc_type")
			log.doc_name = event.get("payload", {}).get("name", "")
			log.node_id = event.get("node_id", "remote")
			log.schema_version = event.get("schema_version", "")
			log.payload = event.get("payload", {})
			log.checksum = event.get("checksum", "")
			log.timestamp = event.get("timestamp")
			log.sync_status = sync_status
			log.insert(ignore_permissions=True)
		frappe.db.commit()
	except Exception:
		pass

"""

Called by the after_migrate hook so retries happen immediately post-upgrade.
"""

import frappe

from frappe_pulse.sync.protocol import apply_event, get_local_schema_version


def retry_schema_mismatches() -> dict:
	"""
	Re-attempt all SCHEMA_MISMATCH events using the current (post-upgrade) schema version.
	Safe to call multiple times — already-applied events are skipped.
	"""
	mismatches = frappe.get_all(
		"Pulse Event Log",
		filters={"sync_status": "SCHEMA_MISMATCH"},
		fields=[
			"name", "event_id", "event_type", "doc_type", "doc_name",
			"node_id", "schema_version", "payload", "checksum", "timestamp",
		],
		order_by="timestamp asc",
		limit=500,
	)

	if not mismatches:
		return {"retried": 0, "applied": 0, "still_mismatch": 0}

	local_schema = get_local_schema_version()
	results = {"retried": len(mismatches), "applied": 0, "still_mismatch": 0, "failed": 0}

	for row in mismatches:
		event = {
			"event_id": row.event_id,
			"event_type": row.event_type,
			"doc_type": row.doc_type,
			"payload": row.payload if isinstance(row.payload, dict) else {},
			"node_id": row.node_id,
			"schema_version": row.schema_version,
			"checksum": row.checksum,
			"timestamp": str(row.timestamp),
		}
		outcome = apply_event(event, local_schema)
		if outcome == "applied":
			results["applied"] += 1
		elif outcome == "schema_mismatch":
			results["still_mismatch"] += 1
		else:
			results["failed"] += 1

	# Reset schema_mismatch_count on all nodes if we cleared them all
	if results["still_mismatch"] == 0:
		frappe.db.sql("UPDATE `tabPulse Node` SET schema_mismatch_count = 0")
		frappe.db.commit()

	frappe.logger("frappe_pulse").info(
		f"Pulse schema retry: {results}"
	)
	return results

"""
Sync worker — called by the scheduler every N minutes.
Iterates all enabled Pulse Nodes and runs push/pull per direction.
"""

import frappe

from frappe_pulse.sync.inbound import pull_from_node
from frappe_pulse.sync.outbound import push_to_node


def run_sync():
	"""Entry point for the scheduler. Processes all active Pulse Nodes."""
	if not frappe.db.get_single_value("Pulse Settings", "enabled"):
		return

	batch_size = int(frappe.db.get_single_value("Pulse Settings", "worker_batch_size") or 100)

	nodes = frappe.get_all(
		"Pulse Node",
		filters={"enabled": 1, "sync_status": ["!=", "PAUSED"]},
		fields=["name", "sync_direction", "last_synced_at"],
	)

	for node_meta in nodes:
		try:
			_sync_node(node_meta.name, node_meta.sync_direction, batch_size)
		except Exception as exc:
			frappe.logger("frappe_pulse").error(
				f"Pulse sync worker: unhandled error for node {node_meta.name}: {exc}"
			)


def _sync_node(node_name: str, direction: str, batch_size: int) -> None:
	node = frappe.get_doc("Pulse Node", node_name)

	result = {"push": {}, "pull": {}}

	if direction in ("PUSH", "BIDIRECTIONAL"):
		result["push"] = push_to_node(node, batch_size)

	if direction in ("PULL", "BIDIRECTIONAL"):
		result["pull"] = pull_from_node(node, batch_size)

	# Only mark synced if no errors in either direction
	push_err = result["push"].get("error")
	pull_err = result["pull"].get("error")

	if push_err or pull_err:
		node.mark_error(push_err or pull_err)
	else:
		node.mark_synced()
		_update_remote_schema_version(node)

	frappe.logger("frappe_pulse").info(
		f"Pulse sync: node={node_name} direction={direction} "
		f"push={result['push']} pull={result['pull']}"
	)


def _update_remote_schema_version(node) -> None:
	"""Fetch and store the remote node's Frappe version after a successful sync."""
	try:
		import requests
		url = f"{node.endpoint_url}/api/method/frappe_pulse.api.sync.status"
		headers = {"Authorization": f"token {node.get_password('api_key')}"}
		resp = requests.get(url, headers=headers, timeout=10)
		if resp.ok:
			data = resp.json().get("message", {})
			remote_version = data.get("frappe_version", "")
			if remote_version:
				frappe.db.set_value("Pulse Node", node.name, "frappe_version", remote_version)
	except Exception:
		pass

    client.send_batch(to_node_id="abc", addr="192.168.1.10:7731", events_json=b"[...]")

    client.disconnect()
"""

import json
import socket
import struct
import threading
import logging

logger = logging.getLogger("frappe_pulse.transport")

# --- Frame constants (must match Go internal/frame/frame.go) ---
_MAGIC = b"PULS"
_VERSION = 1
_HEADER_SIZE = 10

_TYPE_HANDSHAKE    = 0x01
_TYPE_HANDSHAKE_ACK = 0x02
_TYPE_EVENT_BATCH  = 0x10
_TYPE_ACK          = 0x11
_TYPE_PING         = 0x20
_TYPE_PONG         = 0x21
_TYPE_ERROR        = 0xFF


def _write_frame(sock: socket.socket, ftype: int, payload: bytes) -> None:
    length = len(payload)
    header = _MAGIC + struct.pack("<BBi", ftype, _VERSION, length)
    # Pad header to 10 bytes: magic(4) + type(1) + version(1) + length(4)
    sock.sendall(header + payload)


def _read_frame(sock: socket.socket) -> tuple[int, bytes]:
    hdr = _recvall(sock, _HEADER_SIZE)
    if hdr[:4] != _MAGIC:
        raise ValueError(f"bridge_client: bad magic {hdr[:4]!r}")
    ftype = hdr[4]
    length = struct.unpack_from("<I", hdr, 6)[0]
    payload = _recvall(sock, length) if length else b""
    return ftype, payload


def _recvall(sock: socket.socket, n: int) -> bytes:
    buf = bytearray()
    while len(buf) < n:
        chunk = sock.recv(n - len(buf))
        if not chunk:
            raise ConnectionError("bridge_client: connection closed")
        buf.extend(chunk)
    return bytes(buf)


class BridgeClient:
    """Thread-safe client for the pulse-transport Unix socket bridge."""

    def __init__(self, socket_path: str = "/tmp/pulse-transport.sock"):
        self.socket_path = socket_path
        self._sock: socket.socket | None = None
        self._lock = threading.Lock()
        self._reader: threading.Thread | None = None
        self._running = False
        self.on_batch = None  # callback(from_node_id: str, events_json: bytes)

    def connect(self) -> None:
        self._sock = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
        self._sock.connect(self.socket_path)
        self._running = True
        self._reader = threading.Thread(target=self._read_loop, daemon=True)
        self._reader.start()
        logger.info("bridge_client: connected to %s", self.socket_path)

    def disconnect(self) -> None:
        self._running = False
        if self._sock:
            try:
                self._sock.close()
            except Exception:
                pass
        self._sock = None

    def send_batch(self, to_node_id: str, addr: str, events_json: bytes) -> None:
        """Send an event batch to a remote node via the Go transport."""
        dest = json.dumps({"to_node_id": to_node_id, "addr": addr}).encode()
        with self._lock:
            if not self._sock:
                raise ConnectionError("bridge_client: not connected")
            _write_frame(self._sock, _TYPE_HANDSHAKE, dest)
            _write_frame(self._sock, _TYPE_EVENT_BATCH, events_json)

    def _read_loop(self) -> None:
        """Read inbound event batches delivered by the Go transport."""
        while self._running and self._sock:
            try:
                ftype, payload = _read_frame(self._sock)
                if ftype == _TYPE_HANDSHAKE:
                    # This is the envelope: {"from_node_id": "..."}
                    envelope = json.loads(payload)
                    from_node_id = envelope.get("from_node_id", "")
                    # Next frame is the actual event batch
                    ftype2, batch_payload = _read_frame(self._sock)
                    if ftype2 == _TYPE_EVENT_BATCH and self.on_batch:
                        try:
                            self.on_batch(from_node_id, batch_payload)
                        except Exception as exc:
                            logger.error("bridge_client: on_batch error: %s", exc)
                elif ftype == _TYPE_PING:
                    with self._lock:
                        _write_frame(self._sock, _TYPE_PONG, b"")
            except Exception as exc:
                if self._running:
                    logger.warning("bridge_client: read error: %s", exc)
                break


# Module-level singleton — initialised by transport/manager.py
_client: BridgeClient | None = None


def get_client() -> BridgeClient | None:
    return _client


def init_client(socket_path: str = "/tmp/pulse-transport.sock") -> BridgeClient:
    global _client
    _client = BridgeClient(socket_path)
    return _client

import frappe


def process_event(
	event_id: str,
	event_type: str,
	doc_type: str,
	doc_name: str,
	node_id: str,
	schema_version: str,
	payload: dict,
	checksum: str,
	timestamp: str,
) -> None:
	"""Write a captured event to Pulse Event Log."""
	try:
		log = frappe.new_doc("Pulse Event Log")
		log.event_id = event_id
		log.event_type = event_type
		log.doc_type = doc_type
		log.doc_name = doc_name
		log.node_id = node_id
		log.schema_version = schema_version
		log.payload = payload
		log.checksum = checksum
		log.timestamp = timestamp
		log.sync_status = "PENDING"
		log.insert(ignore_permissions=True)
		frappe.db.commit()
	except Exception:
		frappe.log_error(frappe.get_traceback(), "Pulse Event Worker Error")

import frappe


def run_health_check():
	"""
	Scheduled task — runs every 10 minutes.
	Re-queues PENDING events that were never picked up (worker crash / restart),
	and marks events as FAILED after exceeding retry limit.
	"""
	try:
		retry_limit = int(frappe.db.get_single_value("Pulse Settings", "worker_retry_limit") or 3)
		batch_size = int(frappe.db.get_single_value("Pulse Settings", "worker_batch_size") or 100)
	except Exception:
		retry_limit = 3
		batch_size = 100

	# Mark as FAILED if retry_count exceeded (we track this via a custom field in future;
	# for now just surface events stuck as PENDING for over 30 minutes)
	frappe.db.sql(
		"""
		UPDATE `tabPulse Event Log`
		SET sync_status = 'FAILED'
		WHERE sync_status = 'PENDING'
		  AND timestamp < DATE_SUB(NOW(), INTERVAL 30 MINUTE)
		LIMIT %s
		""",
		(batch_size,),
	)
	frappe.db.commit()

	stuck_count = frappe.db.count(
		"Pulse Event Log", {"sync_status": "FAILED"}
	)

	if stuck_count > 0:
		frappe.logger("frappe_pulse").warning(
			f"frappe-pulse health check: {stuck_count} events in FAILED state"
		)
