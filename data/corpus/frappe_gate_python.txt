import frappe
from frappe import _


@frappe.whitelist(allow_guest=False)
def receive_batch():
	"""
	POST /api/method/frappe_gate.api.audit.receive_batch
	Called by the Go gate process to flush audit entries into Frappe DB.
	"""
	import json
	data = frappe.local.form_dict
	entries = data.get("entries") or []
	if isinstance(entries, str):
		try:
			entries = json.loads(entries)
		except Exception:
			frappe.throw(_("Invalid entries payload"))

	for e in entries:
		try:
			log = frappe.new_doc("Gate Audit Log")
			log.timestamp = e.get("ts", "")
			log.device_id = e.get("device_id", "")
			log.role = e.get("role", "")
			log.method = e.get("method", "")
			log.path = e.get("path", "")
			log.status_code = int(e.get("status", 0))
			log.ip_address = e.get("ip", "")
			log.outcome = e.get("outcome", "")
			log.duration_ms = int(e.get("duration_ms", 0))
			log.anomaly = e.get("anomaly", "")
			log.insert(ignore_permissions=True)
		except Exception:
			pass
	frappe.db.commit()
	return {"inserted": len(entries)}


@frappe.whitelist()
def get_recent(limit: int = 100, outcome: str = ""):
	filters = {}
	if outcome:
		filters["outcome"] = outcome
	return frappe.get_all(
		"Gate Audit Log",
		filters=filters,
		fields=["timestamp", "device_id", "method", "path", "status_code",
		        "ip_address", "outcome", "duration_ms", "anomaly"],
		order_by="timestamp desc",
		limit=int(limit),
	)


@frappe.whitelist()
def get_stats():
	total = frappe.db.count("Gate Audit Log")
	denied = frappe.db.count("Gate Audit Log", {"outcome": "denied"})
	errors = frappe.db.count("Gate Audit Log", {"outcome": "error"})
	anomalies = frappe.db.count("Gate Audit Log", {"anomaly": ("!=", "")})
	return {
		"total": total,
		"denied": denied,
		"errors": errors,
		"anomalies": anomalies,
	}

import frappe


@frappe.whitelist()
def check_policy(role: str = "", path: str = "", method: str = "GET"):
	"""Test whether a given role/path/method would be allowed."""
	from frappe_gate.api.policy import _eval_policies
	allowed = _eval_policies(role, path, method)
	return {"role": role, "path": path, "method": method, "allowed": allowed}


def _eval_policies(role: str, path: str, method: str) -> bool:
	import fnmatch
	policies = frappe.get_all(
		"Gate Policy",
		filters={"enabled": 1},
		fields=["role", "path_glob", "methods", "action"],
		order_by="action desc",
	)
	for p in policies:
		r = p.role
		if r != "*" and r.lower() != role.lower():
			continue
		if not fnmatch.fnmatch(path, p.path_glob):
			continue
		if p.methods != "*" and method.upper() not in [m.strip().upper() for m in p.methods.split(",")]:
			continue
		return p.action == "allow"
	return False


@frappe.whitelist(allow_guest=False)
def get_policies():
	"""Called by the Go gate process every reload_seconds to fetch current rules."""
	policies = frappe.get_all(
		"Gate Policy",
		filters={"enabled": 1},
		fields=["role", "path_glob", "methods", "action", "rate_limit"],
		order_by="action desc",  # deny rules first for safety
	)
	return policies

import frappe
from frappe_gate.doctype.gate_security_event.gate_security_event import log_event


@frappe.whitelist(allow_guest=False)
def receive_security_event(
    event_type: str,
    severity: str,
    device_id: str = "",
    ip_address: str = "",
    detail: str = "",
    raw_context: str = "{}",
):
    """Called by the Go gate process to persist a security event into Frappe."""
    import json
    try:
        ctx = json.loads(raw_context)
    except Exception:
        ctx = {}
    log_event(event_type, severity, device_id, ip_address, detail, ctx)
    return {"status": "ok"}

@frappe.whitelist(allow_guest=False)
def resolve_event(name: str):
    doc = frappe.get_doc("Gate Security Event", name)
    doc.resolved = 1
    doc.resolved_at = frappe.utils.now_datetime()
    doc.save(ignore_permissions=True)
    frappe.db.commit()
    return {"status": "resolved"}


@frappe.whitelist(allow_guest=False)
def unblock_ip(ip_address: str):
    """Admin action: tell the gate process to unblock an IP (via Gate Settings admin URL)."""
    import requests
    settings = frappe.get_single("Gate Settings")
    if not settings.gate_admin_url:
        return {"status": "error", "detail": "Gate admin URL not configured"}
    try:
        resp = requests.post(
            f"{settings.gate_admin_url}/gate/hardening/unblock",
            json={"ip": ip_address},
            headers={"X-Gate-Admin-Key": settings.gate_admin_key or ""},
            timeout=5,
        )
        return {"status": "ok", "code": resp.status_code}
    except Exception as e:
        return {"status": "error", "detail": str(e)}

import frappe
from frappe.model.document import Document


class GateDevice(Document):
	def on_update(self):
		# When admin revokes from desk, push revocation to in-memory gate cache via API
		if self.status == "REVOKED" and not self.revoke_reason:
			self.db_set("revoke_reason", f"Revoked by {frappe.session.user}")
		_notify_gate(self.device_id, self.status)

	def approve(self):
		self.db_set("status", "APPROVED")
		_notify_gate(self.device_id, "APPROVED")

	def revoke(self, reason: str = ""):
		self.db_set("status", "REVOKED")
		self.db_set("revoke_reason", reason or f"Revoked by {frappe.session.user}")
		_notify_gate(self.device_id, "REVOKED")


def _notify_gate(device_id: str, status: str) -> None:
	"""Push device status change to the running gate process via its admin API."""
	try:
		gate_url = frappe.db.get_single_value("Gate Settings", "gate_admin_url") or ""
		gate_key = frappe.db.get_single_value("Gate Settings", "gate_admin_key") or ""
		if not gate_url:
			return
		import requests
		endpoint = f"{gate_url}/gate/device/{'revoke' if status == 'REVOKED' else 'unrevoke'}"
		requests.post(
			endpoint,
			json={"device_id": device_id},
			headers={"X-Gate-Admin-Key": gate_key},
			timeout=3,
		)
	except Exception:
		pass  # Gate may not be running — revocation persists in DB on next restart

import frappe
from frappe.model.document import Document


class GateSecurityEvent(Document):
    def after_insert(self):
        if self.severity in ("HIGH", "CRITICAL"):
            _send_notification(self)


def _send_notification(event):
    try:
        admins = frappe.get_all("Has Role", filters={"role": "System Manager"}, fields=["parent"])
        for row in admins:
            frappe.publish_realtime(
                "gate_security_alert",
                {
                    "event_type": event.event_type,
                    "severity": event.severity,
                    "ip_address": event.ip_address,
                    "device_id": event.device_id,
                    "detail": event.detail,
                    "name": event.name,
                },
                user=row.parent,
            )
    except Exception:
        pass


def log_event(
    event_type: str,
    severity: str,
    device_id: str = "",
    ip_address: str = "",
    detail: str = "",
    raw_context: dict | None = None,
):
    import json
    doc = frappe.new_doc("Gate Security Event")
    doc.event_type = event_type
    doc.severity = severity
    doc.device_id = device_id
    doc.ip_address = ip_address
    doc.detail = detail
    doc.timestamp = frappe.utils.now_datetime()
    doc.raw_context = json.dumps(raw_context or {})
    doc.insert(ignore_permissions=True)
    frappe.db.commit()

import frappe
from frappe.model.document import Document


class GateSettings(Document):
	def on_update(self):
		if self.auto_policy_for_pulse:
			_ensure_pulse_policies(self.pulse_sync_role or "System Manager")

	def activate_lockdown(self, reason: str = "") -> None:
		self.db_set("lockdown_active", 1)
		self.db_set("lockdown_reason", reason)
		self.db_set("lockdown_activated_by", frappe.session.user)
		self.db_set("lockdown_activated_at", frappe.utils.now())
		_push_lockdown_to_gate(True)

	def lift_lockdown(self) -> None:
		self.db_set("lockdown_active", 0)
		self.db_set("lockdown_reason", "")
		_push_lockdown_to_gate(False)


def _ensure_pulse_policies(role: str) -> None:
	"""Auto-create Gate Policy records for frappe-pulse sync endpoints."""
	pulse_paths = [
		"/api/method/frappe_pulse.api.sync.pull",
		"/api/method/frappe_pulse.api.sync.receive",
		"/api/method/frappe_pulse.api.sync.status",
	]
	for path in pulse_paths:
		if not frappe.db.exists("Gate Policy", {"path_glob": path, "role": role}):
			doc = frappe.new_doc("Gate Policy")
			doc.role = role
			doc.path_glob = path
			doc.methods = "*"
			doc.action = "allow"
			doc.description = "Auto-created for frappe-pulse sync"
			doc.insert(ignore_permissions=True)
	frappe.db.commit()


def _push_lockdown_to_gate(active: bool) -> None:
	try:
		gate_url = frappe.db.get_single_value("Gate Settings", "gate_admin_url") or ""
		gate_key = frappe.db.get_single_value("Gate Settings", "gate_admin_key") or ""
		if not gate_url:
			return
		import requests
		endpoint = f"{gate_url}/gate/lockdown/{'activate' if active else 'lift'}"
		requests.post(endpoint, headers={"X-Gate-Admin-Key": gate_key}, timeout=3)
	except Exception:
		pass

						<td>${frappe.utils.escape_html(e.ip_address||"")}</td>

						<td style="color:#e65100">${frappe.utils.escape_html(e.anomaly||"")}</td>
					</tr>`).join("");
				$("#audit-rows").html(rows || '<tr><td colspan="9" class="text-center text-muted">No entries</td></tr>');
			},
		});
	}

	// CSV export
	$("#audit-export").on("click", function () {
		const rows = [];
		$("#audit-rows tr").each(function () {
			rows.push(Array.from($(this).find("td")).map(td => `"${$(td).text().trim()}"`).join(","));
		});
		const csv = "Timestamp,Device,Method,Path,Status,Outcome,IP,Duration,Anomaly\n" + rows.join("\n");
		const blob = new Blob([csv], { type: "text/csv" });
		const url = URL.createObjectURL(blob);
		const a = document.createElement("a"); a.href = url;
		a.download = `gate_audit_${frappe.datetime.now_date()}.csv`;
		a.click();
		URL.revokeObjectURL(url);
	});

	$("#audit-outcome-filter, #audit-refresh").on("change click", load);

	load();
	const timer = setInterval(load, 15000);
	$(wrapper).on("hide", () => clearInterval(timer));
};

					<th>User</th><th>IP Mode</th><th>Last Seen</th><th>Actions</th>
				</tr>
			</thead>
			<tbody id="gate-device-rows"></tbody>
		</table>

	function statusBadge(s) {
		const c = {APPROVED:"green",PENDING:"orange",REVOKED:"red",BLOCKED:"darkred"};
		return `<span class="indicator ${c[s]||"gray"}">${s}</span>`;
	}

	function statCard(label, val, color) {
		return `<div class="col-sm-3"><div class="card text-center p-3" style="border-left:4px solid ${color}">
			<div style="font-size:28px;font-weight:700;color:${color}">${val}</div>
			<div style="color:#888;font-size:12px">${label}</div></div></div>`;

	function load() {
		const status = $("#gate-status-filter").val();
		frappe.call({
			method: "frappe_gate.api.device.get_devices",
			args: { status },
			callback(r) {
				const devices = r.message || [];
				const pending = devices.filter(d => d.status === "PENDING").length;
				const approved = devices.filter(d => d.status === "APPROVED").length;
				const revoked = devices.filter(d => d.status === "REVOKED").length;
				$stats.html(
					statCard("Total", devices.length, "#607d8b") +
					statCard("Pending", pending, "#ff9800") +
					statCard("Approved", approved, "#4caf50") +
					statCard("Revoked", revoked, "#f44336")
				);
				const rows = devices.map(d => `
					<tr>
						<td><code>${frappe.utils.escape_html(d.device_id||"")}</code></td>
						<td>${frappe.utils.escape_html(d.device_name||"")}</td>
						<td>${statusBadge(d.status)}</td>
						<td>${frappe.utils.escape_html(d.platform||"")}</td>
						<td>${frappe.utils.escape_html(d.assigned_user||"")}</td>
						<td>${frappe.utils.escape_html(d.ip_binding_mode||"")}</td>
						<td>${d.last_seen_at ? frappe.datetime.str_to_user(d.last_seen_at) : "—"}</td>
						<td>
							${d.status === "PENDING" ? `<button class="btn btn-xs btn-success approve-btn" data-id="${d.device_id}">Approve</button>` : ""}
							${d.status === "APPROVED" ? `<button class="btn btn-xs btn-danger revoke-btn" data-id="${d.device_id}">Revoke</button>` : ""}
						</td>
					</tr>`).join("");
				$("#gate-device-rows").html(rows || '<tr><td colspan="8" class="text-center text-muted">No devices</td></tr>');
			},
		});
	}

	// Lockdown status check
	function checkLockdown() {
		frappe.db.get_single_value("Gate Settings", "lockdown_active").then(val => {
			$lockdown.toggle(!!val);
		});
	}

	$body.on("click", ".approve-btn", function () {
		const id = $(this).data("id");
		frappe.call({ method: "frappe_gate.api.device.approve_device", args: { device_id: id }, callback: load });
	});

	$body.on("click", ".revoke-btn", function () {
		const id = $(this).data("id");
		frappe.prompt("Reason for revocation", ({ value }) => {
			frappe.call({ method: "frappe_gate.api.device.revoke_device", args: { device_id: id, reason: value }, callback: load });
		}, "Revoke Device", "Revoke");
	});

	$("#activate-lockdown").on("click", function () {
		frappe.prompt("Reason for lockdown", ({ value }) => {
			frappe.call({
				method: "frappe_gate.api.device.activate_lockdown",
				args: { reason: value },
				callback() { frappe.msgprint("🔒 Lockdown activated."); checkLockdown(); },
			});
		}, "Emergency Lockdown", "Activate");
	});

	$("#lift-lockdown").on("click", function () {
		frappe.confirm("Lift emergency lockdown?", () => {
			frappe.call({ method: "frappe_gate.api.device.lift_lockdown", callback() { checkLockdown(); load(); } });
		});
	});

	$("#gate-status-filter, #gate-refresh").on("change click", load);

	checkLockdown();
	load();
	const timer = setInterval(() => { load(); checkLockdown(); }, 30000);
	$(wrapper).on("hide", () => clearInterval(timer));
};

frappe.pages["gate-policy-builder"].on_page_load = function (wrapper) {
	const page = frappe.ui.make_app_page({
		parent: wrapper,
		title: "Gate Policy Builder",
		single_column: true,
	});
	const $body = $(wrapper).find(".layout-main-section");

	// ── Policy table ─────────────────────────────────────────────────────
	const $table = $(`
		<div>
			<div style="margin-bottom:12px">
				<button class="btn btn-primary btn-sm" id="add-policy">+ Add Policy</button>
				<button class="btn btn-default btn-sm" id="test-policy" style="margin-left:8px">Test a Request</button>
			</div>
			<table class="table table-bordered" style="font-size:13px">
				<thead>
					<tr><th>Role</th><th>Path Glob</th><th>Methods</th><th>Action</th>
					    <th>Rate Limit</th><th>Enabled</th><th>Actions</th></tr>
				</thead>
				<tbody id="policy-rows"></tbody>
			</table>
		</div>
	`).appendTo($body);

	function loadPolicies() {
		frappe.db.get_list("Gate Policy", {
			fields: ["name", "role", "path_glob", "methods", "action", "rate_limit", "enabled"],
			order_by: "action desc, role asc",
			limit: 200,
		}).then(rows => {
			const html = rows.map(p => `
				<tr>
					<td>${frappe.utils.escape_html(p.role)}</td>
					<td><code>${frappe.utils.escape_html(p.path_glob)}</code></td>
					<td>${frappe.utils.escape_html(p.methods)}</td>
					<td><span class="badge ${p.action==="allow"?"badge-success":"badge-danger"}">${p.action.toUpperCase()}</span></td>
					<td>${p.rate_limit || "—"}</td>
					<td>${p.enabled ? "✓" : "—"}</td>
					<td>
						<button class="btn btn-xs btn-default edit-policy" data-name="${p.name}">Edit</button>
						<button class="btn btn-xs btn-danger del-policy" data-name="${p.name}" style="margin-left:4px">Delete</button>
					</td>
				</tr>`).join("");
			$("#policy-rows").html(html || '<tr><td colspan="7" class="text-center text-muted">No policies</td></tr>');
		});
	}

	function openPolicyDialog(name) {
		const d = new frappe.ui.Dialog({
			title: name ? "Edit Policy" : "Add Policy",
			fields: [
				{ fieldname: "role", fieldtype: "Data", label: "Role", reqd: 1, description: "Frappe role or * for any" },
				{ fieldname: "path_glob", fieldtype: "Data", label: "Path Glob", reqd: 1 },
				{ fieldname: "methods", fieldtype: "Data", label: "Methods", default: "*", description: "GET,POST or *" },
				{ fieldname: "action", fieldtype: "Select", label: "Action", options: "allow\ndeny", default: "allow" },
				{ fieldname: "rate_limit", fieldtype: "Int", label: "Rate Limit (req/min, 0=unlimited)" },
				{ fieldname: "enabled", fieldtype: "Check", label: "Enabled", default: 1 },
				{ fieldname: "description", fieldtype: "Small Text", label: "Description" },
			],
			primary_action_label: name ? "Save" : "Create",
			primary_action(vals) {
				const method = name
					? "frappe.client.set_value"
					: "frappe.client.insert";
				const args = name
					? { doctype: "Gate Policy", name, fieldname: vals }
					: { doc: { doctype: "Gate Policy", ...vals } };
				frappe.call({ method, args, callback() { d.hide(); loadPolicies(); } });
			},
		});
		if (name) {
			frappe.db.get_doc("Gate Policy", name).then(doc => { d.set_values(doc); d.show(); });
		} else {
			d.show();
		}
	}

	$body.on("click", "#add-policy", () => openPolicyDialog(null));

	$body.on("click", ".edit-policy", function () {
		openPolicyDialog($(this).data("name"));
	});

	$body.on("click", ".del-policy", function () {
		const name = $(this).data("name");
		frappe.confirm("Delete this policy?", () => {
			frappe.db.delete_doc("Gate Policy", name).then(loadPolicies);
		});
	});

	$body.on("click", "#test-policy", function () {
		const d = new frappe.ui.Dialog({
			title: "Test Policy Check",
			fields: [
				{ fieldname: "role", fieldtype: "Data", label: "Role" },
				{ fieldname: "path", fieldtype: "Data", label: "Path" },
				{ fieldname: "method", fieldtype: "Select", label: "Method", options: "GET\nPOST\nPUT\nDELETE\nPATCH", default: "GET" },
			],
			primary_action_label: "Check",
			primary_action(vals) {
				frappe.call({
					method: "frappe_gate.api.policy.check_policy",
					args: vals,
					callback(r) {
						const res = r.message || {};
						const color = res.allowed ? "green" : "red";
						frappe.msgprint(`<span style="color:${color};font-size:16px;font-weight:bold">
							${res.allowed ? "✓ ALLOWED" : "✗ DENIED"}
						</span>`);
					},
				});
			},
		});
		d.show();
	});

	loadPolicies();
};
