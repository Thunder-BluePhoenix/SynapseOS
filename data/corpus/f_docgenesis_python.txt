@frappe.whitelist()
def generate_with_ai(prompt: str, doctype: str, count: int = 10,
                     create_records: bool = True, provider: Optional[str] = None) -> Dict[str, Any]:
	"""
	Generate data using AI from natural language prompt

	This enhanced generation function uses AI to create realistic, contextually
	appropriate data based on your description.

	Args:
		prompt: Natural language description of what to generate
		doctype: Target doctype
		count: Number of records to generate (default: 10)
		create_records: Actually create the records vs preview (default: True)
		provider: AI provider ('openai', 'anthropic', 'local') or None for auto-detect

	Returns:
		Generated data and stats

	Example:
		frappe.call({
			method: 'f_docgenesis.api.ai_generation.generate_with_ai',
			args: {
				prompt: 'Create employees for a fintech startup with diverse roles',
				doctype: 'Employee',
				count: 20,
				create_records: true
			}
		})
	"""
	try:
		# Validate inputs
		count = int(count)
		create_records = bool(create_records)

		if count <= 0:
			frappe.throw("Count must be greater than 0")

		if count > 100:
			frappe.throw("Maximum count is 100 per request")

		# Check permissions
		if not frappe.has_permission(doctype, 'read'):
			frappe.throw(f"Insufficient permissions to read {doctype}")

		if create_records and not frappe.has_permission(doctype, 'create'):
			frappe.throw(f"Insufficient permissions to create {doctype}")

		if not frappe.db.exists('DocType', doctype):
			frappe.throw(f"DocType '{doctype}' does not exist")

		# Generate with AI
		ai_gen = AIDataGenerator(provider=provider)
		documents = ai_gen.generate_from_prompt(prompt, doctype, count)

		created_records = []
		errors = []

		if create_records:
			# Create the records
			for idx, doc_data in enumerate(documents):
				try:
					doc = frappe.get_doc(doc_data)

					# Mark as AI-generated
					if hasattr(doc, 'custom_fdocgenesis_flag'):
						doc.custom_fdocgenesis_flag = 1
					if hasattr(doc, 'custom_fdocgenesis_ai_generated'):
						doc.custom_fdocgenesis_ai_generated = 1

					doc.insert(ignore_permissions=True)
					doc.save(ignore_permissions=True)

					created_records.append(doc.name)

				except Exception as e:
					error_msg = f"Record {idx + 1}: {str(e)}"
					frappe.log_error(f"Error creating {doctype}: {str(e)}")
					errors.append(error_msg)
					continue

			frappe.db.commit()

		return {
			'success': True,
			'provider': ai_gen.provider,
			'model': ai_gen.model,
			'prompt': prompt,
			'doctype': doctype,
			'requested_count': count,
			'generated_count': len(documents),
			'created_count': len(created_records) if create_records else 0,
			'errors': errors if errors else None,
			'preview': documents[:5] if not create_records else None,  # Show first 5 in preview
			'created_records': created_records if create_records else None
		}

	except Exception as e:
		frappe.log_error(f"AI generation error: {str(e)}", "F-DocGenesis AI")
		return {
			'success': False,
			'error': str(e)
		}


@frappe.whitelist()
def generate_scenario(scenario_key: str, parameters: str,
                     create_records: bool = True, provider: Optional[str] = None) -> Dict[str, Any]:
	"""
	Generate a pre-built business scenario

	Pre-built scenarios provide complete business contexts with multiple
	related doctypes populated with realistic, interconnected data.

	Args:
		scenario_key: Name of the scenario (e.g., 'ecommerce_black_friday')
		parameters: JSON string of scenario parameters
		create_records: Actually create the records vs preview
		provider: AI provider to use

	Returns:
		Generated scenario data

	Example:
		frappe.call({
			method: 'f_docgenesis.api.ai_generation.generate_scenario',
			args: {
				scenario_key: 'startup_team',
				parameters: JSON.stringify({
					employee_count: 25
				}),
				create_records: true
			}
		})
	"""
	try:
		# Parse parameters
		if isinstance(parameters, str):
			params = json.loads(parameters)
		else:
			params = parameters

		create_records = bool(create_records)

		# Generate scenario
		scenario_gen = ScenarioGenerator()
		result = scenario_gen.generate_scenario(scenario_key, params, ai_provider=provider)

		# Create records if requested
		if create_records and result.get('success'):
			created_by_doctype = {}

			for doctype, doctype_result in result['results'].items():
				if 'documents' in doctype_result:
					created = []

					for doc_data in doctype_result['documents']:
						try:
							doc = frappe.get_doc(doc_data)

							# Mark as AI-generated scenario
							if hasattr(doc, 'custom_fdocgenesis_flag'):
								doc.custom_fdocgenesis_flag = 1
							if hasattr(doc, 'custom_fdocgenesis_ai_generated'):
								doc.custom_fdocgenesis_ai_generated = 1
							if hasattr(doc, 'custom_fdocgenesis_scenario'):
								doc.custom_fdocgenesis_scenario = scenario_key

							doc.insert(ignore_permissions=True)
							doc.save(ignore_permissions=True)

							created.append(doc.name)

						except Exception as e:
							frappe.log_error(f"Error creating {doctype}: {str(e)}")
							continue

					created_by_doctype[doctype] = created

			frappe.db.commit()

			result['created_records'] = created_by_doctype

		return result

	except Exception as e:
		frappe.log_error(f"Scenario generation error: {str(e)}", "F-DocGenesis AI")
		return {
			'success': False,
			'error': str(e)
		}

@frappe.whitelist()
def check_ai_config() -> Dict[str, Any]:
	"""
	Check AI configuration and available providers

	Checks which AI providers are configured and available for use.

	Returns:
		Configuration status for each provider
	"""
	try:
		config_status = {
			'openai': {
				'configured': bool(frappe.conf.get('openai_api_key')),
				'model': frappe.conf.get('openai_model', 'gpt-4-turbo-preview')
			},
			'anthropic': {
				'configured': bool(frappe.conf.get('anthropic_api_key')),
				'model': frappe.conf.get('anthropic_model', 'claude-3-5-sonnet-20241022')
			},
			'local': {
				'configured': bool(frappe.conf.get('local_llm_enabled')),
				'model': frappe.conf.get('local_llm_model', 'llama3:70b'),
				'url': frappe.conf.get('ollama_url', 'http://localhost:11434')
			}
		}

		# Determine default provider
		default_provider = 'openai'  # Default
		if config_status['openai']['configured']:
			default_provider = 'openai'
		elif config_status['anthropic']['configured']:
			default_provider = 'anthropic'
		elif config_status['local']['configured']:
			default_provider = 'local'

		return {
			'success': True,
			'providers': config_status,
			'default_provider': default_provider,
			'has_ai_configured': any(p['configured'] for p in config_status.values())
		}

	except Exception as e:
		frappe.log_error(f"Check AI config error: {str(e)}", "F-DocGenesis AI")
		return {
			'success': False,
			'error': str(e)
		}

"""
Copy-Paste API for F-DocGenesis
Enhanced copy-paste with deep link traversal
"""

import frappe
from typing import Dict, List, Optional, Any
import json

from f_docgenesis.main.deep_copy_engine import (
	deep_copy_doc,
	deep_copy_multiple_docs,
	paste_docs
)


@frappe.whitelist()
def deep_copy_document(doctype: str, name: str, max_depth: int = 5) -> Dict[str, Any]:
	"""
	Deep copy a single document with all linked dependencies

	This enhanced copy function:
	- Copies the main document
	- Recursively traverses and copies all linked documents
	- Includes child table linked documents
	- Returns a complete clipboard package

	Args:
		doctype: Source doctype
		name: Source document name
		max_depth: Maximum depth for link traversal (default: 5)

	Returns:
		Clipboard package with all copied documents

	Example:
		result = deep_copy_document('Employee', 'EMP-001', max_depth=3)
		# Copies Employee, City, Country, Designation, etc.
	"""
	try:
		# Check permissions
		if not frappe.has_permission(doctype, 'read'):
			frappe.throw(f"Insufficient permissions to read {doctype}")

		if not frappe.db.exists(doctype, name):
			frappe.throw(f"Document {doctype} {name} does not exist")

		max_depth = int(max_depth)

		result = deep_copy_doc(doctype, name, max_depth=max_depth)

		return result

	except Exception as e:
		# Truncate error to avoid title overflow
		error_msg = str(e)[:200]
		frappe.log(f"Deep copy API error: {error_msg}")
		return {
			'success': False,
			'error': error_msg
		}


@frappe.whitelist()
def deep_copy_multiple(documents: str, max_depth: int = 5) -> Dict[str, Any]:
	"""
	Deep copy multiple documents with their dependencies

	Perfect for ListView multi-select copy operation.

	Args:
		documents: JSON string of document list [{'doctype': ..., 'name': ...}, ...]
		max_depth: Maximum depth for link traversal

	Returns:
		Clipboard package with all copied documents

	Example:
		docs = [
			{'doctype': 'Employee', 'name': 'EMP-001'},
			{'doctype': 'Employee', 'name': 'EMP-002'}
		]
		result = deep_copy_multiple(json.dumps(docs), max_depth=3)
	"""
	try:
		# Parse documents
		if isinstance(documents, str):
			docs_list = json.loads(documents)
		else:
			docs_list = documents

		if not isinstance(docs_list, list):
			frappe.throw("Documents must be a list")

		# Check permissions for all documents
		for doc_ref in docs_list:
			doctype = doc_ref.get('doctype')
			name = doc_ref.get('name')

			if not doctype or not name:
				frappe.throw("Each document must have 'doctype' and 'name'")

			if not frappe.has_permission(doctype, 'read'):
				frappe.throw(f"Insufficient permissions to read {doctype}")

			if not frappe.db.exists(doctype, name):
				frappe.throw(f"Document {doctype} {name} does not exist")

		max_depth = int(max_depth)

		result = deep_copy_multiple_docs(docs_list, max_depth=max_depth)

		return result

	except Exception as e:
		# Truncate error to avoid title overflow
		error_msg = str(e)[:200]
		frappe.log(f"Multi copy API error: {error_msg}")
		return {
			'success': False,
			'error': error_msg
		}


@frappe.whitelist()
def paste_documents(clipboard_data: str, skip_duplicates: bool = True) -> Dict[str, Any]:
	"""
	Paste documents from clipboard with smart duplicate handling

	This paste function:
	- Creates all documents from clipboard
	- Checks for duplicates using unique fields
	- If duplicate found and skip_duplicates=True, uses existing document
	- Updates all link fields to point to correct documents
	- Maintains all relationships

	Args:
		clipboard_data: JSON string of clipboard package from deep_copy
		skip_duplicates: Skip creating duplicates, use existing instead

	Returns:
		Paste result with created documents and stats

	Example:
		# Copy
		clipboard = deep_copy_document('Employee', 'EMP-001')

		# Paste (will skip duplicates and link to existing)
		result = paste_documents(json.dumps(clipboard), skip_duplicates=True)
		# Returns: {
		#   'success': True,
		#   'created_count': 5,
		#   'duplicate_skipped': 3,
		#   'created_docs': ['Employee:EMP-001-NEW', ...]
		# }
	"""
	try:
		skip_duplicates = bool(skip_duplicates)

		result = paste_docs(clipboard_data, skip_duplicates=skip_duplicates)

		return result

	except Exception as e:
		# Use frappe.log to avoid error cascade from long error messages
		error_msg = str(e)[:200]  # Truncate to avoid title overflow in Error Log
		frappe.log(f"Paste API error: {error_msg}")
		return {
			'success': False,
			'error': error_msg
		}


@frappe.whitelist()
def get_clipboard_info(clipboard_data: str) -> Dict[str, Any]:
	"""
	Get information about clipboard contents without pasting

	Preview what will be created/skipped when pasting.

	Args:
		clipboard_data: JSON string of clipboard package

	Returns:
		Information about clipboard contents
	"""
	try:
		if isinstance(clipboard_data, str):
			data = json.loads(clipboard_data)
		else:
			data = clipboard_data

		if not data.get('success') or data.get('format') != 'fdocgenesis-deep-copy-v1':
			frappe.throw("Invalid clipboard format")

		metadata = data.get('metadata', {})
		documents = data.get('documents', {})

		# Analyze what will happen on paste
		preview = {
			'total_documents': len(documents),
			'by_doctype': metadata.get('by_doctype', {}),
			'relationships': len(metadata.get('link_relationships', [])),
			'primary_documents': [],
			'linked_documents': []
		}

		for doc_key, doc_info in documents.items():
			doc_summary = {
				'doctype': doc_info['doctype'],
				'name': doc_info['name'],
				'depth': doc_info['depth']
			}

			if doc_info.get('is_primary'):
				preview['primary_documents'].append(doc_summary)
			else:
				preview['linked_documents'].append(doc_summary)

		return {
			'success': True,
			'preview': preview
		}

	except Exception as e:
		# Truncate error to avoid title overflow
		error_msg = str(e)[:200]
		frappe.log(f"Clipboard info error: {error_msg}")
		return {
			'success': False,
			'error': error_msg
		}

"""
Deletion API for F-DocGenesis
REST API endpoints for demo data deletion
"""

import frappe
from typing import Dict, Optional, Any
import json

from f_docgenesis.main.deletion_engine import (
	delete_generated_data,
	delete_by_doctype,
	delete_doctype_and_linked,
	cleanup_old_data
)


@frappe.whitelist()
def delete_data(session_id: Optional[str] = None, dry_run: bool = False,
                filters: Optional[str] = None) -> Dict[str, Any]:
	"""
	Delete generated demo data

	Args:
		session_id: Optional session ID
		dry_run: Preview mode
		filters: JSON string of filters

	Returns:
		Deletion result dictionary
	"""
	try:
		# Parse filters
		if isinstance(filters, str):
			filters = json.loads(filters) if filters else None

		dry_run = bool(dry_run)

		# Check permissions
		if not frappe.has_permission('System Manager'):
			frappe.throw("Only System Managers can delete demo data")

		result = delete_generated_data(
			session_id=session_id,
			dry_run=dry_run,
			filters=filters
		)

		return result

	except Exception as e:
		frappe.log_error(f"Delete API error: {str(e)}", "F-DocGenesis Deletion")
		return {
			'success': False,
			'error': str(e)
		}


@frappe.whitelist()
def delete_doctype_data(doctype: str, dry_run: bool = False,
                        include_linked: bool = False) -> Dict[str, Any]:
	"""
	Delete all generated data for a specific doctype

	Args:
		doctype: Doctype name
		dry_run: Preview mode
		include_linked: Also delete linked documents created during generation

	Returns:
		Deletion result dictionary
	"""
	try:
		dry_run = bool(dry_run)
		include_linked = bool(include_linked)

		if not frappe.db.exists('DocType', doctype):
			frappe.throw(f"DocType '{doctype}' does not exist")

		# Check permissions
		if not frappe.has_permission(doctype, 'delete'):
			frappe.throw(f"Insufficient permissions to delete {doctype}")

		result = delete_by_doctype(doctype=doctype, dry_run=dry_run,
		                           include_linked=include_linked)

		return result

	except Exception as e:
		frappe.log_error(f"Delete doctype API error: {str(e)}", "F-DocGenesis Deletion")
		return {
			'success': False,
			'error': str(e)
		}

@frappe.whitelist()
def count_generated_records(doctype: Optional[str] = None) -> Dict[str, Any]:
	"""
	Count generated records

	Args:
		doctype: Optional specific doctype

	Returns:
		Count information
	"""
	try:
		if doctype:
			if not frappe.db.exists('DocType', doctype):
				frappe.throw(f"DocType '{doctype}' does not exist")

			if not frappe.db.has_column(doctype, 'custom_fdocgenesis_flag'):
				return {
					'success': True,
					'doctype': doctype,
					'count': 0,
					'message': 'Doctype does not have tracking field'
				}

			count = frappe.db.count(doctype, {'custom_fdocgenesis_flag': 1})

			return {
				'success': True,
				'doctype': doctype,
				'count': count
			}

		else:
			# Count across all doctypes
			all_doctypes = frappe.get_all('DocType',
				filters={'custom': 0, 'istable': 0},
				fields=['name']
			)

			total_count = 0
			by_doctype = {}

			for dt in all_doctypes:
				doctype_name = dt['name']

				try:
					if not frappe.db.has_column(doctype_name, 'custom_fdocgenesis_flag'):
						continue

					count = frappe.db.count(doctype_name, {'custom_fdocgenesis_flag': 1})

					if count > 0:
						by_doctype[doctype_name] = count
						total_count += count

				except Exception:
					continue

			return {
				'success': True,
				'total_count': total_count,
				'by_doctype': by_doctype
			}

	except Exception as e:

@frappe.whitelist()
def get_generation_config(doctype: str) -> Dict[str, Any]:
	"""
	Get suggested generation configuration for a doctype

	Args:
		doctype: Target doctype name

	Returns:
		Suggested configuration
	"""
	try:
		if not frappe.db.exists('DocType', doctype):
			frappe.throw(f"DocType '{doctype}' does not exist")

		from f_docgenesis.main.field_parser import DoctypeFieldParser
		parser = DoctypeFieldParser(doctype)

		# Build suggested config
		config = {}

		# Link fields
		for link_field in parser.get_link_fields():
			fieldname = link_field['fieldname']
			target_doctype = link_field['options']

			# Check if target doctype has existing records
			existing_count = frappe.db.count(target_doctype)

			config[fieldname] = {
				'mode': 'existing' if existing_count > 0 else 'new',
				'existing_count': min(existing_count, 5),
				'new_count': 1,
				'filters': {}
			}

		# Child tables
		for child_table in parser.get_child_tables():
			fieldname = child_table['fieldname']
			child_doctype = child_table['options']

			child_config = {
				'rows': 3
			}

			# Child link fields
			for child_link in child_table.get('child_link_fields', []):
				child_fieldname = child_link['fieldname']
				child_target = child_link['options']

				existing_count = frappe.db.count(child_target)
				child_config[child_fieldname] = {
					'mode': 'existing' if existing_count > 0 else 'new',
					'existing_count': min(existing_count, 3),
					'new_count': 1,
					'filters': {}
				}

			config[fieldname] = child_config

		return {
			'success': True,
			'data': {
				'doctype': doctype,
				'config': config,
				'link_fields': parser.get_link_fields(),
				'child_tables': parser.get_child_tables()
			}
		}

	except Exception as e:
		frappe.log_error(f"Generation config API error: {str(e)}", "F-DocGenesis Config")
		return {
			'success': False,
			'error': str(e)
		}

"""
JSON Paste API for F-DocGenesis
Create records from pasted JSON data
"""

import frappe
from typing import Dict, List, Any
import json


@frappe.whitelist()
def create_from_json(json_data: str, doctype: str = None) -> Dict[str, Any]:
	"""
	Create records from JSON data (single or multiple)

	Supports:
	1. Single record: {"name": "X", "field1": "value1", ...}
	2. Multiple records as JSON lines: one record per line
	3. Array of records: [{"name": "X", ...}, {"name": "Y", ...}]

	Args:
		json_data: JSON string containing record(s)
		doctype: Optional doctype (if not in JSON)

	Returns:
		Result with created records count
	"""
	try:
		created_records = []
		failed_records = []
		skipped_duplicates = []

		# Parse JSON data
		json_data = json_data.strip()

		# Try to parse as array first
		try:
			data = json.loads(json_data)
			if isinstance(data, list):
				records = data
			elif isinstance(data, dict):
				records = [data]
			else:
				frappe.throw("Invalid JSON format")
		except json.JSONDecodeError:
			# Try parsing as JSON lines (one record per line)
			records = []
			for line in json_data.split('\n'):
				line = line.strip()
				if line:
					try:
						record = json.loads(line)
						if isinstance(record, dict):
							records.append(record)
					except:
						continue

		if not records:
			frappe.throw("No valid records found in JSON data")

		# Process each record
		for idx, record_data in enumerate(records):
			try:
				# Determine doctype
				rec_doctype = record_data.get('doctype') or doctype

				if not rec_doctype:
					failed_records.append({
						'index': idx,
						'error': 'DocType not specified'
					})
					continue

				# Check permissions
				if not frappe.has_permission(rec_doctype, 'create'):
					failed_records.append({
						'index': idx,
						'doctype': rec_doctype,
						'error': f'Insufficient permissions to create {rec_doctype}'
					})
					continue

				# Check if record already exists
				rec_name = record_data.get('name')
				if rec_name and frappe.db.exists(rec_doctype, rec_name):
					skipped_duplicates.append({
						'doctype': rec_doctype,
						'name': rec_name
					})
					continue

				# Clean up the data
				clean_data = _clean_record_data(record_data, rec_doctype)

				# Create new document
				doc = frappe.get_doc(clean_data)
				doc.insert(ignore_permissions=False)

				created_records.append({
					'doctype': rec_doctype,
					'name': doc.name
				})

			except Exception as e:
				failed_records.append({
					'index': idx,
					'doctype': record_data.get('doctype'),
					'name': record_data.get('name'),
					'error': str(e)
				})
				continue

		# Commit all changes
		frappe.db.commit()

		return {
			'success': True,
			'created_count': len(created_records),
			'failed_count': len(failed_records),
			'skipped_count': len(skipped_duplicates),
			'created_records': created_records,
			'failed_records': failed_records,
			'skipped_duplicates': skipped_duplicates,
			'total_processed': len(records)
		}

	except Exception as e:
		frappe.log_error(f"JSON paste error: {str(e)}", "F-DocGenesis JSON Paste")
		return {
			'success': False,
			'error': str(e)
		}


def _clean_record_data(data: Dict, doctype: str) -> Dict:
	"""
	Clean record data before insertion

	Removes:
	- System fields that shouldn't be copied
	- Invalid fields for the doctype
	- Empty values
	"""
	# Fields to remove
	remove_fields = [
		'__last_sync_on', '__unsaved', '__onload', '__run_link_triggers',
		'_liked_by', '_comments', '_assign', '_user_tags',
		'creation', 'modified', 'modified_by', 'owner',
		'docstatus'  # Will be 0 by default
	]

	# Get valid fields for this doctype
	meta = frappe.get_meta(doctype)
	valid_fields = set([df.fieldname for df in meta.fields])
	valid_fields.add('doctype')
	valid_fields.add('name')

	# Clean the data
	clean_data = {}
	for key, value in data.items():
		# Skip removed fields
		if key in remove_fields:
			continue

		# Skip invalid fields
		if key not in valid_fields:
			continue

		# Skip empty values (except 0 and False)
		if value in [None, '', []]:
			continue

		clean_data[key] = value

	# Ensure doctype is set
	clean_data['doctype'] = doctype

	return clean_data


@frappe.whitelist()
def validate_json(json_data: str) -> Dict[str, Any]:
	"""
	Validate JSON data before creating records

	Returns validation results without creating records
	"""
	try:
		json_data = json_data.strip()

		# Try to parse
		try:
			data = json.loads(json_data)
			if isinstance(data, list):
				records = data
			elif isinstance(data, dict):
				records = [data]
			else:
				return {
					'success': False,
					'error': 'Invalid JSON format. Must be object or array.'
				}
		except json.JSONDecodeError as e:
			# Try JSON lines format
			records = []
			for line in json_data.split('\n'):
				line = line.strip()
				if line:
					try:
						record = json.loads(line)
						if isinstance(record, dict):
							records.append(record)
					except:
						continue

		if not records:
			return {
				'success': False,
				'error': 'No valid records found in JSON'
			}

		# Validate each record
		validation_results = []
		for idx, record in enumerate(records):
			result = {
				'index': idx,
				'doctype': record.get('doctype'),
				'name': record.get('name'),
				'valid': True,
				'warnings': [],
				'errors': []
			}

			# Check doctype
			if not record.get('doctype'):
				result['valid'] = False
				result['errors'].append('DocType not specified')
			else:
				# Check if doctype exists
				if not frappe.db.exists('DocType', record['doctype']):
					result['valid'] = False
					result['errors'].append(f"DocType '{record['doctype']}' does not exist")

				# Check permissions
				if not frappe.has_permission(record['doctype'], 'create'):
					result['valid'] = False
					result['errors'].append(f"No permission to create {record['doctype']}")

				# Check if record exists
				if record.get('name') and frappe.db.exists(record['doctype'], record['name']):
					result['warnings'].append(f"Record {record['name']} already exists (will be skipped)")

			validation_results.append(result)

		valid_count = sum(1 for r in validation_results if r['valid'])

		return {
			'success': True,
			'total_records': len(records),
			'valid_records': valid_count,
			'invalid_records': len(records) - valid_count,
			'validation_results': validation_results
		}

	except Exception as e:
		return {
			'success': False,
			'error': str(e)
		}

	results = {}

	for doctype, count in doctypes.items():
		try:
			records = quick_generate(doctype, count, use_existing)
			results[doctype] = records
			frappe.db.commit()
		except Exception as e:
			frappe.log_error(f"Error generating {doctype}: {str(e)}")
			results[doctype] = []

	return results


def generate_with_template(template_name: str, count: Optional[int] = None) -> Dict[str, Any]:

	Generate data using a saved template

	Example:
		>>> from f_docgenesis.api import generate_with_template

		template_name: Name of saved template

		Generation result dictionary
	"""
	from f_docgenesis.main.template_manager import get_template, get_template_manager

	# Get template
	template = get_template(template_name)

	if not template:
		frappe.throw(f'Template "{template_name}" not found')

	# Increment usage count
	manager = get_template_manager()
	manager.increment_usage(template_name)

	# Generate using template config
	result = generate(
		doctype=template['doctype'],
		count=count or 10,  # Default count if not specified
		config=template['config'],
		max_depth=10,
		dry_run=False
	)

	# Add template info to result
	if result.get('success'):
		result['template_used'] = template_name

	return result

@fdocgenesis.command('generate')
@click.option('--doctype', '-d', required=True, help='Target doctype')
@click.option('--count', '-c', default=10, help='Number of records to generate')
@click.option('--config', '-cfg', help='JSON config file path or JSON string')
@click.option('--max-depth', '-m', default=10, help='Maximum dependency depth')
@click.option('--dry-run', is_flag=True, help='Preview without creating records')
@click.option('--use-existing', is_flag=True, help='Use existing records for links')
@pass_context
def generate_data(context, doctype, count, config, max_depth, dry_run, use_existing):
	"""
	Generate demo data for a doctype

	Examples:
		bench fdocgenesis generate -d "Customer" -c 10
		bench fdocgenesis generate -d "Sales Invoice" -c 5 --dry-run
		bench fdocgenesis generate -d "ToDo" -c 20 --config config.json
	"""
	site = get_site(context)

	frappe.init(site=site)
	frappe.connect()

	try:
		from f_docgenesis.api.python_api import generate, quick_generate

		# Parse config
		config_dict = None
		if config:
			try:
				# Try to load as file
				with open(config, 'r') as f:
					config_dict = json.load(f)
			except FileNotFoundError:
				# Try to parse as JSON string
				try:
					config_dict = json.loads(config)
				except json.JSONDecodeError:
					click.echo(f"Error: Invalid config JSON")
					return

		# Generate data
		click.echo(f"Generating {count} {doctype} records...")

		if use_existing and not config_dict:
			records = quick_generate(doctype, count=count, use_existing=True)
			click.echo(f"✅ Created {len(records)} records")
			for i, record in enumerate(records[:10], 1):
				click.echo(f"  {i}. {record}")
			if len(records) > 10:
				click.echo(f"  ... and {len(records) - 10} more")

		else:
			result = generate(
				doctype=doctype,
				count=count,
				config=config_dict or {},
				max_depth=max_depth,
				dry_run=dry_run
			)

			if result.get('success'):
				click.echo(f"\n✅ Generation completed!")
				click.echo(f"  Session ID: {result['session_id']}")
				click.echo(f"  Records created: {result['records_created']}")
				click.echo(f"  Total documents: {result['total_documents']}")
				click.echo(f"  Generation time: {result['generation_time']}s")

				if result.get('stats'):
					click.echo(f"\n  Statistics by doctype:")
					for dt, stats in result['stats'].items():
						click.echo(f"    {dt}: {stats.get('new', 0)} new")

			else:
				click.echo(f"❌ Generation failed: {result.get('error')}")

	except Exception as e:
		click.echo(f"❌ Error: {str(e)}")

	finally:
		frappe.destroy()

"""
AI Engine Module for F-DocGenesis
AI-powered data generation with multiple provider support
"""

import frappe
import json
from typing import Dict, List, Any, Optional
import re

from f_docgenesis.main.field_parser import DoctypeFieldParser


class AIDataGenerator:
	"""AI-powered data generation engine"""

	def __init__(self, provider: Optional[str] = None):
		"""
		Initialize AI engine

		Args:
			provider: 'openai', 'anthropic', or 'local'. If None, auto-detect from config
		"""
		self.provider = provider or self._detect_provider()
		self.api_key = self._get_api_key()
		self.model = self._get_model()

	def _detect_provider(self) -> str:
		"""Auto-detect available AI provider from site config"""
		if frappe.conf.get('openai_api_key'):
			return 'openai'
		elif frappe.conf.get('anthropic_api_key'):
			return 'anthropic'
		elif frappe.conf.get('local_llm_enabled'):
			return 'local'
		else:
			# Default to openai, will fail if no key provided
			return 'openai'

	def _get_api_key(self) -> Optional[str]:
		"""Get API key for the selected provider"""
		if self.provider == 'openai':
			return frappe.conf.get('openai_api_key')
		elif self.provider == 'anthropic':
			return frappe.conf.get('anthropic_api_key')
		elif self.provider == 'local':
			return None  # Local doesn't need API key
		return None

	def _get_model(self) -> str:
		"""Get model name based on provider"""
		models = {
			'openai': frappe.conf.get('openai_model', 'gpt-4-turbo-preview'),
			'anthropic': frappe.conf.get('anthropic_model', 'claude-3-5-sonnet-20241022'),
			'local': frappe.conf.get('local_llm_model', 'llama3:70b')
		}
		return models.get(self.provider, 'gpt-4-turbo-preview')

	def generate_from_prompt(self, prompt: str, doctype: str,
	                          count: int = 10) -> List[Dict]:
		"""
		Generate data from natural language prompt

		Args:
			prompt: Natural language description
			doctype: Target Frappe doctype
			count: Number of records to generate

		Returns:
			List of document dictionaries
		"""
		try:
			# Get doctype schema
			parser = DoctypeFieldParser(doctype)
			schema = self._build_schema_description(parser)

			# Build AI prompt
			ai_prompt = self._build_ai_prompt(prompt, schema, doctype, count)

			frappe.log(f"Generating {count} {doctype} records with AI ({self.provider})")

			# Call AI API
			response = self._call_ai_api(ai_prompt)

			# Parse and validate response
			documents = self._parse_ai_response(response, doctype)

			frappe.log(f"AI generated {len(documents)} documents successfully")

			return documents

		except Exception as e:
			frappe.log_error(f"AI generation error: {str(e)}", "F-DocGenesis AI")
			raise

	def _build_schema_description(self, parser: DoctypeFieldParser) -> str:
		"""Build human-readable schema description for AI"""
		fields = parser.get_all_fields()

		schema_parts = [f"DocType: {parser.doctype}", "", "Fields to populate:"]

		for field in fields:
			# Skip certain field types
			if field['fieldtype'] in ['Section Break', 'Column Break', 'HTML', 'Table']:
				continue

			field_desc = f"- {field['fieldname']} ({field['fieldtype']})"

			if field.get('reqd'):
				field_desc += " [REQUIRED]"

			if field.get('options'):
				if field['fieldtype'] == 'Select':
					field_desc += f" Options: {field['options']}"
				elif field['fieldtype'] == 'Link':
					field_desc += f" (Links to {field['options']} - use realistic names)"

			if field.get('description'):
				field_desc += f" - {field['description']}"

			schema_parts.append(field_desc)

		return "\n".join(schema_parts)

	def _build_ai_prompt(self, user_prompt: str, schema: str,
	                     doctype: str, count: int) -> str:
		"""Build complete AI prompt"""
		return f"""You are a data generation expert for Frappe/ERPNext systems.

USER REQUEST: {user_prompt}

TARGET SCHEMA:
{schema}

TASK: Generate {count} realistic records for the "{doctype}" doctype.

CRITICAL REQUIREMENTS:
1. Return ONLY valid JSON array - no markdown, no explanations, no code blocks
2. Each object must match the schema exactly
3. All REQUIRED fields must be filled
4. Make data contextually appropriate and realistic
5. Ensure consistency (e.g., if name is "John", email should be john@...)
6. Use proper formats for dates (YYYY-MM-DD), emails, phones
7. For Link fields, use realistic simple names that could exist (e.g., "Mumbai", "Software Engineer")
8. Generate diverse but realistic data
9. Do not include any field that is not in the schema above

OUTPUT FORMAT:
[
    {{
        "fieldname1": "value1",
        "fieldname2": "value2"
    }},
    ...
]

Generate exactly {count} records now:"""

	def _call_ai_api(self, prompt: str) -> str:
		"""Call AI API based on provider"""
		if self.provider == 'openai':
			return self._call_openai(prompt)
		elif self.provider == 'anthropic':
			return self._call_anthropic(prompt)
		elif self.provider == 'local':
			return self._call_local_llm(prompt)
		else:
			raise ValueError(f"Unsupported AI provider: {self.provider}")

	def _call_openai(self, prompt: str) -> str:
		"""Call OpenAI API"""
		try:
			import openai
		except ImportError:
			frappe.throw("OpenAI library not installed. Run: pip install openai")

		if not self.api_key:
			frappe.throw("OpenAI API key not configured. Add 'openai_api_key' to site_config.json")

		try:
			client = openai.OpenAI(api_key=self.api_key)

			response = client.chat.completions.create(
				model=self.model,
				messages=[
					{"role": "system", "content": "You are a data generation expert. Always return valid JSON arrays only."},
					{"role": "user", "content": prompt}
				],
				temperature=0.7,
				max_tokens=4000
			)

			return response.choices[0].message.content

		except Exception as e:
			frappe.log_error(f"OpenAI API error: {str(e)}", "F-DocGenesis AI")
			raise

	def _call_anthropic(self, prompt: str) -> str:
		"""Call Anthropic Claude API"""
		try:
			import anthropic
		except ImportError:
			frappe.throw("Anthropic library not installed. Run: pip install anthropic")

		if not self.api_key:
			frappe.throw("Anthropic API key not configured. Add 'anthropic_api_key' to site_config.json")

		try:
			client = anthropic.Anthropic(api_key=self.api_key)

			message = client.messages.create(
				model=self.model,
				max_tokens=4000,
				temperature=0.7,
				messages=[
					{"role": "user", "content": prompt}
				]
			)

			return message.content[0].text

		except Exception as e:
			frappe.log_error(f"Anthropic API error: {str(e)}", "F-DocGenesis AI")
			raise

	def _call_local_llm(self, prompt: str) -> str:
		"""Call local LLM (Ollama)"""
		try:
			import requests
		except ImportError:
			frappe.throw("Requests library not installed. Run: pip install requests")

		ollama_url = frappe.conf.get('ollama_url', 'http://localhost:11434')

		try:
			response = requests.post(
				f'{ollama_url}/api/generate',
				json={
					'model': self.model,
					'prompt': prompt,
					'stream': False,
					'options': {
						'temperature': 0.7
					}
				},
				timeout=120
			)

			if response.status_code == 200:
				return response.json()['response']
			else:
				frappe.throw(f"Ollama API error: {response.status_code} - {response.text}")

		except requests.exceptions.ConnectionError:
			frappe.throw("Cannot connect to Ollama. Make sure Ollama is running on localhost:11434")
		except Exception as e:
			frappe.log_error(f"Local LLM error: {str(e)}", "F-DocGenesis AI")
			raise

	def _parse_ai_response(self, response: str, doctype: str) -> List[Dict]:
		"""Parse AI response into document dictionaries"""
		try:
			# Remove markdown code blocks if present
			response = response.strip()
			if response.startswith('```'):
				# Remove opening ```json or ```
				response = re.sub(r'^```(?:json)?\s*\n', '', response)
				# Remove closing ```
				response = re.sub(r'\n```\s*$', '', response)

			# Find JSON array in response
			json_start = response.find('[')
			json_end = response.rfind(']') + 1

			if json_start == -1 or json_end == 0:
				frappe.log_error(f"No valid JSON array found in AI response: {response[:500]}")
				frappe.throw("AI did not return valid JSON. Please try again.")

			json_str = response[json_start:json_end]
			documents = json.loads(json_str)

			if not isinstance(documents, list):
				frappe.throw("AI response is not a JSON array")

			# Add doctype to each document
			validated_docs = []
			for doc_data in documents:
				if not isinstance(doc_data, dict):
					continue

				doc_data['doctype'] = doctype

				# Clean up None values
				doc_data = {k: v for k, v in doc_data.items() if v is not None}

				validated_docs.append(doc_data)

			return validated_docs

		except json.JSONDecodeError as e:
			frappe.log_error(f"JSON parse error: {str(e)}\nResponse: {response[:1000]}")
			frappe.throw(f"Invalid JSON from AI: {str(e)}")
		except Exception as e:
			frappe.log_error(f"Response parsing error: {str(e)}")
			raise

	def generate_scenario(self, scenario_key: str, parameters: Dict[str, Any],
	                     ai_provider: Optional[str] = None) -> Dict[str, Any]:
		"""
		Generate a complete business scenario

		Args:
			scenario_key: Key of the scenario to generate
			parameters: Scenario parameters
			ai_provider: AI provider to use

		Returns:
			Generated scenario data
		"""
		if scenario_key not in self.SCENARIOS:
			frappe.throw(f"Unknown scenario: {scenario_key}")

		scenario = self.SCENARIOS[scenario_key]

		# Build prompt from template
		prompt = scenario['prompt_template'].format(**parameters)

		ai_gen = AIDataGenerator(provider=ai_provider)

		results = {}
		total_created = 0

		for doctype in scenario['doctypes']:
			try:
				# Determine count from parameters
				count_key = f"{doctype.lower().replace(' ', '_')}_count"
				count = parameters.get(count_key, 10)

				frappe.log(f"Generating {count} {doctype} for scenario {scenario_key}")

				# Generate with AI
				docs = ai_gen.generate_from_prompt(prompt, doctype, count)

				results[doctype] = {
					'generated_count': len(docs),
					'documents': docs
				}

				total_created += len(docs)

			except Exception as e:
				frappe.log_error(f"Error generating {doctype} for scenario: {str(e)}")
				results[doctype] = {
					'error': str(e),
					'generated_count': 0
				}

		return {
			'success': True,
			'scenario': scenario_key,
			'scenario_name': scenario['name'],
			'total_created': total_created,
			'results': results
		}

"""
Analytics Module for F-DocGenesis
Tracks usage patterns, performance metrics, and provides insights
"""

import frappe
from typing import Dict, List, Optional, Any
from datetime import datetime, timedelta
from collections import defaultdict


class AnalyticsCollector:
	"""Collects and analyzes F-DocGenesis usage analytics"""

	def __init__(self):
		self.cache_key_prefix = "fdocgenesis_analytics_"

	def track_generation(self, session_id: str, doctype: str, count: int,
	                     generation_time: float, total_documents: int):
		"""
		Track a generation event

		Args:
			session_id: Generation session ID
			doctype: Target doctype
			count: Records created
			generation_time: Time taken
			total_documents: Total documents including linked
		"""
		event = {
			'session_id': session_id,
			'doctype': doctype,
			'count': count,
			'generation_time': generation_time,
			'total_documents': total_documents,
			'timestamp': frappe.utils.now_datetime().isoformat(),
			'user': frappe.session.user
		}

		# Store in cache
		self._append_to_cache('generation_events', event)

		# Update doctype stats
		self._increment_doctype_stat(doctype, count, generation_time)

	def get_usage_stats(self, days: int = 30) -> Dict[str, Any]:
		"""
		Get usage statistics

		Args:
			days: Number of days to analyze

		Returns:
			Usage statistics dictionary
		"""
		events = self._get_cached_events('generation_events', days)

		if not events:
			return {
				'total_sessions': 0,
				'total_records': 0,
				'total_documents': 0,
				'avg_generation_time': 0,
				'by_doctype': {},
				'by_user': {}
			}

		# Calculate stats
		total_sessions = len(events)
		total_records = sum(e['count'] for e in events)
		total_documents = sum(e['total_documents'] for e in events)
		avg_time = sum(e['generation_time'] for e in events) / len(events)

		# Group by doctype
		by_doctype = defaultdict(lambda: {
			'sessions': 0,
			'records': 0,
			'avg_time': 0,
			'total_time': 0
		})

		for event in events:
			dt = event['doctype']
			by_doctype[dt]['sessions'] += 1
			by_doctype[dt]['records'] += event['count']
			by_doctype[dt]['total_time'] += event['generation_time']

		# Calculate averages
		for dt in by_doctype:
			if by_doctype[dt]['sessions'] > 0:
				by_doctype[dt]['avg_time'] = (
					by_doctype[dt]['total_time'] / by_doctype[dt]['sessions']
				)

		# Group by user
		by_user = defaultdict(lambda: {
			'sessions': 0,
			'records': 0
		})

		for event in events:
			user = event.get('user', 'Unknown')
			by_user[user]['sessions'] += 1
			by_user[user]['records'] += event['count']

		return {
			'total_sessions': total_sessions,
			'total_records': total_records,
			'total_documents': total_documents,
			'avg_generation_time': round(avg_time, 2),
			'by_doctype': dict(by_doctype),
			'by_user': dict(by_user),
			'period_days': days
		}

	def get_performance_metrics(self, doctype: Optional[str] = None) -> Dict[str, Any]:
		"""
		Get performance metrics

		Args:
			doctype: Optional specific doctype

		Returns:
			Performance metrics dictionary
		"""
		events = self._get_cached_events('generation_events')

		if doctype:
			events = [e for e in events if e['doctype'] == doctype]

		if not events:
			return {
				'count': 0,
				'avg_time': 0,
				'min_time': 0,
				'max_time': 0,
				'avg_docs_per_second': 0
			}

		times = [e['generation_time'] for e in events]
		total_docs = sum(e['total_documents'] for e in events)
		total_time = sum(times)

		return {
			'count': len(events),
			'avg_time': round(sum(times) / len(times), 2),
			'min_time': round(min(times), 2),
			'max_time': round(max(times), 2),
			'avg_docs_per_second': round(total_docs / total_time, 2) if total_time > 0 else 0,
			'total_documents': total_docs,
			'doctype': doctype
		}

	def get_popular_doctypes(self, limit: int = 10) -> List[Dict[str, Any]]:
		"""
		Get most popular doctypes

		Args:
			limit: Number of results

		Returns:
			List of popular doctypes with counts
		"""
		events = self._get_cached_events('generation_events')

		doctype_counts = defaultdict(int)
		for event in events:
			doctype_counts[event['doctype']] += event['count']

		sorted_doctypes = sorted(
			doctype_counts.items(),
			key=lambda x: x[1],
			reverse=True
		)

		return [
			{'doctype': dt, 'count': count}
			for dt, count in sorted_doctypes[:limit]
		]

	def get_trends(self, days: int = 30) -> Dict[str, Any]:
		"""
		Get usage trends over time

		Args:
			days: Number of days

		Returns:
			Trends data
		"""
		events = self._get_cached_events('generation_events', days)

		# Group by date
		by_date = defaultdict(lambda: {
			'sessions': 0,
			'records': 0,
			'avg_time': 0,
			'times': []
		})

		for event in events:
			date = event['timestamp'][:10]  # Get date part
			by_date[date]['sessions'] += 1
			by_date[date]['records'] += event['count']
			by_date[date]['times'].append(event['generation_time'])

		# Calculate averages
		for date in by_date:
			times = by_date[date]['times']
			by_date[date]['avg_time'] = sum(times) / len(times) if times else 0
			del by_date[date]['times']  # Remove raw times

		return {
			'by_date': dict(by_date),
			'period_days': days
		}

	def _append_to_cache(self, cache_type: str, data: Dict):
		"""Append data to cache"""
		cache_key = f"{self.cache_key_prefix}{cache_type}"

		# Get existing data

"""
Background Jobs Module for F-DocGenesis
Handles asynchronous generation using Frappe's queue system
"""

import frappe
from typing import Dict, Optional, Any
import json


def enqueue_generation(doctype: str, count: int = 10, config: Optional[Dict] = None,
                       max_depth: int = 10, dry_run: bool = False,
                       queue: str = 'default', timeout: int = 3600) -> Dict[str, Any]:
	"""
	Enqueue a generation job to run in the background

	Args:
		doctype: Target doctype
		count: Number of records
		config: Generation configuration
		max_depth: Maximum dependency depth
		dry_run: Preview mode
		queue: Queue name (short/default/long)
		timeout: Job timeout in seconds

	Returns:
		Job information dictionary
	"""
	try:
		job_id = frappe.generate_hash(length=16)

		# Enqueue the job
		frappe.enqueue(
			method='f_docgenesis.main.background_jobs.execute_generation',
			queue=queue,
			timeout=timeout,
			is_async=True,
			job_id=job_id,
			doctype=doctype,
			count=count,
			config=config or {},
			max_depth=max_depth,
			dry_run=dry_run
		)

		return {
			'success': True,
			'job_id': job_id,
			'queue': queue,
			'status': 'enqueued',
			'message': f'Generation job enqueued for {doctype}'
		}

	except Exception as e:
		frappe.log_error(f"Enqueue error: {str(e)}")
		return {
			'success': False,
			'error': str(e)
		}

	"""

def enqueue_batch_generation(doctypes: Dict[str, int], use_existing: bool = True,
                              queue: str = 'default') -> Dict[str, Any]:
	"""
	Enqueue batch generation for multiple doctypes

	Args:
		doctypes: Dictionary mapping doctype names to counts
		use_existing: Use existing records for links
		queue: Queue name

	Returns:
		Batch job information
	"""
	try:
		job_ids = []

		for doctype, count in doctypes.items():
			# Auto-configure if use_existing
			config = {}
			if use_existing:
				from f_docgenesis.main.field_parser import DoctypeFieldParser
				parser = DoctypeFieldParser(doctype)

				for link_field in parser.get_link_fields():
					fieldname = link_field['fieldname']
					target_doctype = link_field['options']

					existing_count = frappe.db.count(target_doctype)
					if existing_count > 0:
						config[fieldname] = {
							'mode': 'existing',
							'existing_count': min(existing_count, 5)
						}

			# Enqueue each doctype
			result = enqueue_generation(
				doctype=doctype,
				count=count,
				config=config,
				queue=queue
			)

			if result.get('success'):
				job_ids.append(result['job_id'])

		return {
			'success': True,
			'job_ids': job_ids,
			'total_jobs': len(job_ids),
			'queue': queue
		}

	except Exception as e:
		frappe.log_error(f"Batch enqueue error: {str(e)}")
		return {
			'success': False,
			'error': str(e)
		}

"""
Deep Copy Engine Module for F-DocGenesis
Enhanced copy-paste functionality with recursive link traversal
"""

import frappe
from typing import Dict, List, Optional, Any, Set
import json
import hashlib

from f_docgenesis.main.field_parser import DoctypeFieldParser


class DeepCopySession:
	"""Manages deep copying of documents with linked dependencies"""

	def __init__(self, max_depth: int = 5):
		"""
		Initialize deep copy session

		Args:
			max_depth: Maximum depth for traversing links
		"""
		self.max_depth = max_depth
		self.copied_docs: Dict[str, Any] = {}  # Key: doctype:name, Value: doc data
		self.link_map: Dict[str, str] = {}  # Old name -> New name mapping
		self.processed: Set[str] = set()  # Track processed documents
		self.stats = {
			'total_copied': 0,
			'by_doctype': {},
			'duplicate_skipped': 0,
			'link_relationships': []
		}

	def deep_copy_document(self, doctype: str, name: str) -> Dict[str, Any]:
		"""
		Deep copy a document with all linked dependencies

		Args:
			doctype: Source doctype
			name: Source document name

		Returns:
			Dictionary containing all copied documents and metadata
		"""
		try:
			# Copy the primary document and its links
			self._copy_document_recursive(doctype, name, depth=0, is_primary=True)

			# Build the clipboard package
			package = self._build_clipboard_package()

			return package

		except Exception as e:
			frappe.log_error(f"Deep copy failed: {str(e)}", "F-DocGenesis Deep Copy")
			return {
				'success': False,
				'error': str(e)
			}

	def deep_copy_multiple(self, documents: List[Dict[str, str]]) -> Dict[str, Any]:
		"""
		Deep copy multiple documents

		Args:
			documents: List of dicts with 'doctype' and 'name' keys

		Returns:
			Clipboard package with all documents
		"""
		try:
			for doc_ref in documents:
				doctype = doc_ref.get('doctype')
				name = doc_ref.get('name')
				if doctype and name:
					self._copy_document_recursive(doctype, name, depth=0, is_primary=True)

			package = self._build_clipboard_package()
			return package

		except Exception as e:
			frappe.log_error(f"Multi deep copy failed: {str(e)}", "F-DocGenesis Deep Copy")
			return {
				'success': False,
				'error': str(e)
			}

	def _copy_document_recursive(self, doctype: str, name: str, depth: int = 0,
	                              is_primary: bool = False) -> Optional[str]:
		"""
		Recursively copy document and its links

		Args:
			doctype: Doctype to copy
			name: Document name
			depth: Current recursion depth
			is_primary: Is this a primary document

		Returns:
			Document key (doctype:name)
		"""
		if depth > self.max_depth:
			return None

		doc_key = f"{doctype}:{name}"

		# Skip if already processed
		if doc_key in self.processed:
			return doc_key

		self.processed.add(doc_key)

		try:
			# Get the document
			doc = frappe.get_doc(doctype, name)
			doc_dict = doc.as_dict()

			# Parse fields
			parser = DoctypeFieldParser(doctype)

			# Process link fields recursively
			link_fields = parser.get_link_fields()
			for field in link_fields:
				fieldname = field['fieldname']
				target_doctype = field['options']
				link_value = doc_dict.get(fieldname)

				if link_value and frappe.db.exists(target_doctype, link_value):
					# Recursively copy linked document
					linked_key = self._copy_document_recursive(
						target_doctype,
						link_value,
						depth=depth + 1,
						is_primary=False
					)

					# Track the relationship
					if linked_key:
						self.stats['link_relationships'].append({
							'parent': doc_key,
							'parent_field': fieldname,
							'child': linked_key
						})

			# Process child tables
			child_tables = parser.get_child_tables()
			for child_table in child_tables:
				fieldname = child_table['fieldname']
				child_doctype = child_table['options']
				child_rows = doc_dict.get(fieldname, [])

				# Process each child row's links
				for child_row in child_rows:
					child_parser = DoctypeFieldParser(child_doctype)
					child_links = child_parser.get_link_fields()

					for child_link in child_links:
						child_fieldname = child_link['fieldname']
						child_target = child_link['options']
						child_link_value = child_row.get(child_fieldname)

						if child_link_value and frappe.db.exists(child_target, child_link_value):
							# Recursively copy child's linked document
							self._copy_document_recursive(
								child_target,
								child_link_value,
								depth=depth + 1,
								is_primary=False
							)

			# Store the document
			self.copied_docs[doc_key] = {
				'doctype': doctype,
				'name': name,
				'data': doc_dict,
				'is_primary': is_primary,
				'depth': depth
			}

			# Update stats
			self._increment_stat(doctype)

			return doc_key

		except Exception as e:
			frappe.log_error(f"Error copying {doctype} {name}: {str(e)}")
			return None

	def paste_documents(self, clipboard_data: Dict[str, Any],
	                    skip_duplicates: bool = True) -> Dict[str, Any]:
		"""
		Paste documents from clipboard with smart duplicate handling
		Optimized for performance using depth-based sorting (one-pass creation)
		"""
		try:
			docs_to_create = clipboard_data.get('documents', {})
			# Sort by depth descending (dependencies first)
			sorted_docs = sorted(
				docs_to_create.items(),
				key=lambda x: x[1].get('depth', 0),
				reverse=True
			)

			created_docs = {}
			duplicate_map = {}
			failed_records = []
			self.link_map = {} # Reset link map

			# Single pass: Create documents with resolved links
			for index, (doc_key, doc_info) in enumerate(sorted_docs):
				doctype = doc_info['doctype']
				doc_data = doc_info['data'].copy()
				original_name = doc_info['name']

				try:
					# 1. Check for duplicates
					duplicate_name = None
					if skip_duplicates:
						duplicate_name = self._find_duplicate(doctype, doc_data)

					if duplicate_name:
						duplicate_map[doc_key] = f"{doctype}:{duplicate_name}"
						self.link_map[original_name] = duplicate_name
						self.stats['duplicate_skipped'] += 1
						frappe.log(f"Skipped duplicate {doctype}: {duplicate_name}")
						continue

					# 2. Prepare data for insertion
					cleaned_data = self._clean_document_data(doc_data)

					# 3. Resolve links immediately (since we process dependencies first)
					self._resolve_links_in_data(doctype, cleaned_data, created_docs, duplicate_map)

					# 4. Create document
					try:
						new_doc = frappe.get_doc(cleaned_data)
						new_doc.insert(ignore_permissions=True)

						created_key = f"{doctype}:{new_doc.name}"
						created_docs[doc_key] = created_key
						self.link_map[original_name] = new_doc.name

						frappe.log(f"Created {created_key}")

					except frappe.DuplicateEntryError:
						# If insert failed due to duplicate (e.g. naming collision), treat as duplicate
						if skip_duplicates:
							# Assume the name we tried to insert is the one that exists
							# cleaned_data might not have name if it was autoname
							# But if it was manual name, it's in cleaned_data
							dup_name = cleaned_data.get('name') or original_name

							# Verify it actually exists
							if frappe.db.exists(doctype, dup_name):
								duplicate_map[doc_key] = f"{doctype}:{dup_name}"
								self.link_map[original_name] = dup_name
								self.stats['duplicate_skipped'] += 1
								frappe.log(f"Recovered from duplicate error {doctype}: {dup_name}")
							else:
								raise # Real error
						else:
							raise

				except Exception as e:
					error_msg = str(e)
					frappe.log(f"Error processing {doctype} {original_name}: {error_msg}")
					failed_records.append({
						'index': index + 1,
						'doctype': doctype,
						'name': original_name,
						'error': error_msg
					})

			frappe.db.commit()

			return {
				'success': True,
				'created_count': len(created_docs),
				'duplicate_skipped': self.stats['duplicate_skipped'],
				'failed_count': len(failed_records),
				'failed_records': failed_records,
				'created_docs': list(created_docs.values()),
				'link_map': self.link_map,
				'total_processed': len(sorted_docs)
			}

		except Exception as e:
			error_msg = str(e)[:200]
			frappe.log(f"Paste failed: {error_msg}")
			return {
				'success': False,
				'error': error_msg
			}

	def _resolve_links_in_data(self, doctype: str, doc_data: Dict,
	                           created_docs: Dict[str, str],
	                           duplicate_map: Dict[str, str]):
		"""
		Resolve link fields in document data before insertion
		"""
		parser = DoctypeFieldParser(doctype)

		# 1. Standard Link Fields
		link_fields = parser.get_link_fields()
		for field in link_fields:
			fieldname = field['fieldname']
			target_doctype = field['options']
			old_value = doc_data.get(fieldname)

			if old_value:
				old_key = f"{target_doctype}:{old_value}"
				new_name = None

				if old_key in created_docs:
					_, new_name = created_docs[old_key].split(':', 1)
				elif old_key in duplicate_map:
					_, new_name = duplicate_map[old_key].split(':', 1)

				if new_name:
					doc_data[fieldname] = new_name

		# 2. Child Tables
		child_tables = parser.get_child_tables()
		for child_table in child_tables:
			fieldname = child_table['fieldname']
			child_doctype = child_table['options']
			child_rows = doc_data.get(fieldname, [])

			if not child_rows:
				continue

			child_parser = DoctypeFieldParser(child_doctype)
			child_links = child_parser.get_link_fields()

			for row in child_rows:
				for child_link in child_links:
					child_fieldname = child_link['fieldname']
					child_target = child_link['options']
					old_value = row.get(child_fieldname)

					if old_value:
						old_key = f"{child_target}:{old_value}"
						new_name = None

						if old_key in created_docs:
							_, new_name = created_docs[old_key].split(':', 1)
						elif old_key in duplicate_map:
							_, new_name = duplicate_map[old_key].split(':', 1)

						if new_name:
							row[child_fieldname] = new_name

	def _increment_stat(self, doctype: str):
		"""Increment statistics"""
		if doctype not in self.stats['by_doctype']:
			self.stats['by_doctype'][doctype] = 0
		self.stats['by_doctype'][doctype] += 1
		self.stats['total_copied'] += 1

	def _build_clipboard_package(self) -> Dict[str, Any]:
		"""
		Build clipboard package for export

		Returns:
			Complete clipboard package
		"""
		return {
			'success': True,
			'format': 'fdocgenesis-deep-copy-v1',
			'documents': self.copied_docs,
			'metadata': {
				'total_documents': len(self.copied_docs),
				'by_doctype': self.stats['by_doctype'],
				'link_relationships': self.stats['link_relationships'],
				'max_depth': self.max_depth
			}
		}

	def _clean_document_data(self, doc_data: Dict[str, Any]) -> Dict[str, Any]:
		"""
		Clean document data for insertion
		Removes system fields and handles naming
		"""
		cleaned = doc_data.copy()

		# Fields to remove
		system_fields = [
			'modified', 'creation', 'modified_by', 'owner',
			'docstatus', 'parent', 'parentfield', 'parenttype',
			'idx', '__islocal', '__unsaved', '_user_tags',
			'_comments', '_assign', '_liked_by', 'workflow_state'
		]

		for field in system_fields:
			if field in cleaned:
				del cleaned[field]

		# Handle name
		# If we want to keep the name (e.g. for manual naming), we keep it
		# But if it's a naming series, we might want to remove it to generate a new one
		# For deep copy, we usually want to generate new names unless it's a master data type
		# Here we'll keep name if it looks like a manual name (not matching series pattern)
		# But simpler strategy: let Frappe handle it. If we pass 'name', it tries to insert with that name.
		# If it exists, we catch DuplicateEntryError.

		return cleaned

	def _find_duplicate(self, doctype: str, doc_data: Dict[str, Any]) -> Optional[str]:
		"""
		Check if a duplicate document already exists
		Returns the name of the duplicate if found, else None
		"""
		# 1. Check by Name if provided
		name = doc_data.get('name')
		if name and frappe.db.exists(doctype, name):
			return name

		# 2. Check by unique fields (if any)
		# This is complex to do generically without schema inspection
		# For now, we rely primarily on Name matching for deep copy

class DeletionSession:
	"""Manages deletion of generated demo data"""

	def __init__(self, session_id: Optional[str] = None, dry_run: bool = False):
		"""
		Initialize deletion session

		Args:
			session_id: Generation session ID to delete (if None, delete all)
			dry_run: Preview mode without actual deletion
		"""
		self.session_id = session_id
		self.dry_run = dry_run
		self.deletion_id = self._generate_deletion_id()

		self.deleted_records: List[Dict] = []
		self.rollback_data: List[Dict] = []
		self.stats = {
			'total_deleted': 0,
			'by_doctype': {},
			'start_time': None,
			'end_time': None,
			'can_rollback': True
		}

	def _generate_deletion_id(self) -> str:
		"""Generate unique deletion session ID"""
		timestamp = frappe.utils.now_datetime().strftime('%Y%m%d%H%M%S')
		return f"fdg-del-{timestamp}-{frappe.generate_hash(length=8)}"

	def delete_generated_data(self, filters: Optional[Dict] = None) -> Dict[str, Any]:
		"""
		Delete generated demo data

		Args:
			filters: Optional additional filters

		Returns:
			Deletion result dictionary
		"""
		try:
			self.stats['start_time'] = datetime.now()

			# Find all generated records
			generated_records = self._find_generated_records(filters)

			if not generated_records:
				return {
					'success': True,
					'message': 'No generated records found',
					'deleted_count': 0
				}

			# Group by doctype
			records_by_doctype = self._group_by_doctype(generated_records)

			# Delete in reverse dependency order (children first)
			deletion_order = self._calculate_deletion_order(records_by_doctype)

			for doctype in deletion_order:
				if doctype in records_by_doctype:
					self._delete_doctype_records(
						doctype,
						records_by_doctype[doctype]
					)

			self.stats['end_time'] = datetime.now()
			self.stats['total_deleted'] = len(self.deleted_records)

			return self._build_result()

		except Exception as e:
			frappe.log_error(f"Deletion failed: {str(e)}", "F-DocGenesis Deletion")
			return {
				'success': False,
				'error': str(e),
				'deleted_count': len(self.deleted_records)
			}

	def _find_generated_records(self, filters: Optional[Dict] = None) -> List[Dict]:
		"""
		Find all records generated by F-DocGenesis

		Args:
			filters: Optional filters

		Returns:
			List of record dictionaries
		"""

		# For now, we'll use a simple approach with custom field

		records = []

		# Get all doctypes with the fdocgenesis flag field

			filters={'custom': 0, 'istable': 0},

		)

			doctype = dt['name']

				# Check if doctype has the tracking field
				if not frappe.db.has_column(doctype, 'custom_fdocgenesis_flag'):
					continue

				# Build query filters
				query_filters = {'custom_fdocgenesis_flag': 1}

				if self.session_id:
					# If we had session tracking, add it here
					pass

					query_filters.update(filters)

				# Fetch records

	def _group_by_doctype(self, records: List[Dict]) -> Dict[str, List[str]]:
		"""Group records by doctype"""
		grouped = {}
		for record in records:
			doctype = record['doctype']
			if doctype not in grouped:
				grouped[doctype] = []
			grouped[doctype].append(record['name'])
		return grouped

	def _calculate_deletion_order(self, records_by_doctype: Dict[str, List[str]]) -> List[str]:
		"""
		Calculate deletion order (reverse dependency order)

		Args:
			records_by_doctype: Records grouped by doctype

		Returns:
			List of doctypes in deletion order
		"""
		doctypes = list(records_by_doctype.keys())

		if not doctypes:
			return []

		# Separate child tables and main doctypes
		child_tables = []
		main_doctypes = []

		for doctype in doctypes:
			try:
				meta = frappe.get_meta(doctype)
				if meta.istable:
					child_tables.append(doctype)
				else:
					main_doctypes.append(doctype)
			except:
				main_doctypes.append(doctype)

		# Use dependency resolver to get proper order for main doctypes
		deletion_order = []

		# First, add child tables (they should always be deleted first)
		deletion_order.extend(child_tables)

		# Then, calculate proper order for main doctypes using dependency graph
		# We need to delete in reverse dependency order (delete dependents before dependencies)
		try:
			# Build dependency graph for all main doctypes
			dependency_graph = {}
			for doctype in main_doctypes:
				try:
					resolver = DependencyResolver(max_depth=1)
					deps = resolver.get_dependencies(doctype)
					# Only include dependencies that are in our deletion list
					dependency_graph[doctype] = [d for d in deps.values() if d in main_doctypes]
				except:
					dependency_graph[doctype] = []

			# Topological sort in reverse (delete leaves first)
			visited = set()
			temp_mark = set()
			result = []

			def visit(doctype: str):
				if doctype in temp_mark:
					# Circular dependency, just add it
					return
				if doctype in visited:
					return

				temp_mark.add(doctype)

				# Visit dependencies first
				for dep in dependency_graph.get(doctype, []):
					if dep in main_doctypes:
						visit(dep)

				temp_mark.remove(doctype)
				visited.add(doctype)
				result.insert(0, doctype)  # Insert at beginning for reverse order

			for doctype in main_doctypes:
				if doctype not in visited:
					visit(doctype)

			deletion_order.extend(result)

		except Exception as e:
			frappe.log_error(f"Error calculating deletion order: {str(e)}")
			# Fallback to simple alphabetical order
			deletion_order.extend(sorted(main_doctypes))

		return deletion_order

	def _delete_doctype_records(self, doctype: str, record_names: List[str]):
		"""
		Delete records for a specific doctype

		Args:
			doctype: Doctype name
			record_names: List of record names to delete
		"""
		for name in record_names:
			try:
				# Store for rollback
				if self.stats['can_rollback'] and not self.dry_run:
					try:
						doc_data = frappe.get_doc(doctype, name).as_dict()
						self.rollback_data.append({
							'doctype': doctype,
							'name': name,
							'data': doc_data,
							'deleted_at': frappe.utils.now_datetime()
						})
					except Exception as e:
						frappe.log(f"Could not store rollback data for {doctype} {name}: {str(e)}")

				# Delete the record
				if not self.dry_run:
					try:
						frappe.delete_doc(doctype, name, ignore_permissions=True, force=True)
					except Exception as delete_error:
						error_msg = str(delete_error).lower()
						# Check if it's a foreign key constraint error
						if 'foreign key' in error_msg or 'constraint' in error_msg or 'cannot delete' in error_msg:
							frappe.log(f"Foreign key constraint preventing deletion of {doctype} {name}: {str(delete_error)}")
							frappe.log(f"Will retry this record later...")
							# Don't track as deleted, will be retried
							continue
						else:
							# Other error, log and continue
							frappe.log_error(f"Error deleting {doctype} {name}: {str(delete_error)}")
							continue

				# Track deletion
				self.deleted_records.append({
					'doctype': doctype,
					'name': name
				})

				self._increment_stat(doctype)

			except Exception as e:
				frappe.log_error(f"Unexpected error deleting {doctype} {name}: {str(e)}")
				continue

		# Commit after each doctype
		if not self.dry_run:
			frappe.db.commit()

	def _increment_stat(self, doctype: str):
		"""Increment deletion statistics"""
		if doctype not in self.stats['by_doctype']:
			self.stats['by_doctype'][doctype] = 0
		self.stats['by_doctype'][doctype] += 1

	def _build_result(self) -> Dict[str, Any]:
		"""Build deletion result dictionary"""
		duration = (self.stats['end_time'] - self.stats['start_time']).total_seconds()

		return {
			'success': True,
			'deletion_id': self.deletion_id,
			'session_id': self.session_id,
			'dry_run': self.dry_run,
			'deleted_count': len(self.deleted_records),
			'deletion_time': round(duration, 2),
			'stats': self.stats['by_doctype'],
			'can_rollback': self.stats['can_rollback'] and not self.dry_run,
			'rollback_window_hours': 24
		}

	def rollback(self) -> Dict[str, Any]:
		"""
		Rollback deletion by restoring deleted records

		Returns:
			Rollback result dictionary
		"""
		if not self.rollback_data:
			return {
				'success': False,
				'error': 'No rollback data available'
			}

		restored_count = 0

		try:
			# Restore in reverse deletion order
			for record in reversed(self.rollback_data):
				try:
					doc = frappe.get_doc(record['data'])
					doc.insert(ignore_permissions=True)
					restored_count += 1
				except Exception as e:
					frappe.log_error(f"Rollback error for {record['doctype']} {record['name']}: {str(e)}")
					continue

			frappe.db.commit()

			return {
				'success': True,
				'restored_count': restored_count,
				'total_records': len(self.rollback_data)
			}

		except Exception as e:
			frappe.log_error(f"Rollback failed: {str(e)}")
			return {
				'success': False,
				'error': str(e),
				'restored_count': restored_count
			}

def delete_doctype_and_linked(doctype: str, dry_run: bool = False) -> Dict[str, Any]:
	"""
	Delete all generated records for a specific doctype AND all linked documents
	that were created during generation.

	This is the recommended method when you want to clean up a doctype's demo data
	along with all its dependencies. It uses session tracking to only delete
	records that were generated together.

	Args:
		doctype: Doctype name
		dry_run: Preview mode

	Returns:
		Deletion result dictionary
	"""
	try:
		from f_docgenesis.main.dependency_resolver import DependencyResolver
		from f_docgenesis.main.field_parser import DoctypeFieldParser

		# Check if doctype exists
		if not frappe.db.exists('DocType', doctype):
			return {
				'success': False,
				'error': f'DocType {doctype} does not exist'
			}

		# Get all generated records for the primary doctype
		if not frappe.db.has_column(doctype, 'custom_fdocgenesis_flag'):
			return {
				'success': False,
				'message': f'Doctype {doctype} does not have tracking field. No data was deleted because F-DocGenesis only deletes data it created.',
				'deleted_count': 0
			}

		# Get all generated records for this doctype with their session IDs
		has_session_field = frappe.db.has_column(doctype, 'custom_fdocgenesis_session')

		if has_session_field:
			primary_records = frappe.get_all(
				doctype,
				filters={'custom_fdocgenesis_flag': 1},
				fields=['name', 'custom_fdocgenesis_session as session_id']
			)
		else:
			primary_records = frappe.get_all(
				doctype,
				filters={'custom_fdocgenesis_flag': 1},
				fields=['name']
			)

		if not primary_records:
			return {
				'success': True,
				'message': f'No generated {doctype} records found',
				'deleted_count': 0
			}

		# If we have session tracking, collect session IDs and delete all records from those sessions
		if has_session_field:
			# Extract unique session IDs from primary records
			session_ids = set()
			for record in primary_records:
				if record.get('session_id'):
					session_ids.add(record['session_id'])

			if not session_ids:
				# No session IDs found, fall back to basic deletion
				has_session_field = False
			else:
				frappe.log(f"Found {len(session_ids)} generation sessions for {doctype}")

				# Find ALL records (across all doctypes) that belong to these sessions
				session = DeletionSession(dry_run=dry_run)
				all_generated = session._find_generated_records()

				# Filter to only include records from our target sessions
				records_to_delete = {}
				for record in all_generated:
					record_doctype = record['doctype']

					# Check if this doctype has session tracking
					if not frappe.db.has_column(record_doctype, 'custom_fdocgenesis_session'):
						continue

					# Get the record's session ID
					try:
						rec_session = frappe.db.get_value(record_doctype, record['name'], 'custom_fdocgenesis_session')
						if rec_session in session_ids:
							if record_doctype not in records_to_delete:
								records_to_delete[record_doctype] = []
							records_to_delete[record_doctype].append(record['name'])
					except Exception as e:
						frappe.log(f"Could not get session for {record_doctype} {record['name']}: {str(e)}")
						continue

		# Fallback: No session tracking - use dependency traversal
		if not has_session_field:
			# Collect all doctypes to check for generated records
			doctypes_to_check = set([doctype])

			# Get dependencies (linked doctypes)
			try:
				resolver = DependencyResolver(max_depth=5)
				deps = resolver.get_dependencies(doctype)
				for dep_doctype in deps.values():
					if dep_doctype:
						doctypes_to_check.add(dep_doctype)
			except Exception as e:
				frappe.log(f"Could not resolve dependencies for {doctype}: {str(e)}")

			# Get child tables
			try:
				parser = DoctypeFieldParser(doctype)
				child_tables = parser.get_child_tables()
				for child in child_tables:
					doctypes_to_check.add(child['options'])
			except Exception as e:
				frappe.log(f"Could not get child tables for {doctype}: {str(e)}")

			# Find all generated records across relevant doctypes
			session = DeletionSession(dry_run=dry_run)
			all_generated = session._find_generated_records()

			# Filter to only include doctypes we care about
			records_to_delete = {}
			for record in all_generated:
				if record['doctype'] in doctypes_to_check:
					if record['doctype'] not in records_to_delete:
						records_to_delete[record['doctype']] = []
					records_to_delete[record['doctype']].append(record['name'])

		if not records_to_delete:
			return {
				'success': True,
				'message': f'No generated records found for {doctype} or its dependencies',
				'deleted_count': 0
			}

		# Calculate deletion order
		deletion_order = session._calculate_deletion_order(records_to_delete)

		# Delete in order
		for dt in deletion_order:
			if dt in records_to_delete:
				session._delete_doctype_records(dt, records_to_delete[dt])

		return {
			'success': True,
			'primary_doctype': doctype,
			'deleted_count': len(session.deleted_records),
			'doctypes_affected': list(session.stats['by_doctype'].keys()),
			'stats': session.stats['by_doctype'],
			'dry_run': dry_run
		}

	except Exception as e:
		frappe.log_error(f"Delete doctype and linked failed: {str(e)}")
		return {
			'success': False,
			'error': str(e)
		}

"""
Dependency Resolver Module for F-DocGenesis
Handles recursive link resolution and dependency graph building
"""

import frappe
from typing import Dict, List, Set, Optional, Tuple, Any
from collections import defaultdict


class DependencyNode:
	"""Represents a node in the dependency graph"""

	def __init__(self, doctype: str, field_name: Optional[str] = None,
	             depth: int = 0, parent: Optional['DependencyNode'] = None):
		self.doctype = doctype
		self.field_name = field_name
		self.depth = depth
		self.parent = parent
		self.children: List['DependencyNode'] = []
		self.mode = 'new'  # new/existing/hybrid
		self.count = 1

	def add_child(self, child: 'DependencyNode'):
		"""Add a child node"""
		self.children.append(child)

	def __repr__(self):
		return f"Node({self.doctype}, field={self.field_name}, depth={self.depth})"


class DependencyGraph:
	"""Represents a complete dependency graph"""

	def __init__(self, root_doctype: str, max_depth: int = 10):
		self.root_doctype = root_doctype
		self.max_depth = max_depth
		self.root = DependencyNode(root_doctype, depth=0)
		self.all_nodes: List[DependencyNode] = [self.root]
		self.doctype_counts: Dict[str, int] = defaultdict(int)
		self.doctype_counts[root_doctype] = 1

	def add_dependency(self, parent_node: DependencyNode, child_doctype: str,
	                   field_name: str) -> Optional[DependencyNode]:
		"""Add a dependency to the graph"""
		if parent_node.depth >= self.max_depth:
			return None

		child_node = DependencyNode(
			doctype=child_doctype,
			field_name=field_name,
			depth=parent_node.depth + 1,
			parent=parent_node
		)

		parent_node.add_child(child_node)
		self.all_nodes.append(child_node)
		self.doctype_counts[child_doctype] += 1

		return child_node

	def get_nodes_by_depth(self) -> Dict[int, List[DependencyNode]]:
		"""Get nodes grouped by depth level"""
		nodes_by_depth = defaultdict(list)
		for node in self.all_nodes:
			nodes_by_depth[node.depth].append(node)
		return dict(nodes_by_depth)

	def get_generation_order(self) -> List[DependencyNode]:
		"""Get nodes in order for generation (deepest first)"""
		return sorted(self.all_nodes, key=lambda n: n.depth, reverse=True)

	def get_summary(self) -> Dict[str, Any]:
		"""Get summary of the dependency graph"""
		return {
			'root_doctype': self.root_doctype,
			'max_depth_reached': max(n.depth for n in self.all_nodes),
			'total_nodes': len(self.all_nodes),
			'doctype_distribution': dict(self.doctype_counts),
			'nodes_by_depth': {
				depth: [n.doctype for n in nodes]
				for depth, nodes in self.get_nodes_by_depth().items()
			}
		}


class DependencyResolver:
	"""Resolves dependencies and builds dependency graphs"""

	def __init__(self, max_depth: int = 10):
		"""
		Initialize dependency resolver

		Args:
			max_depth: Maximum recursion depth (default: 10)
		"""
		self.max_depth = max_depth
		self.visited_in_path: Set[str] = set()  # For cycle detection

	def resolve_dependencies(self, doctype: str, config: Optional[Dict] = None) -> DependencyGraph:
		"""
		Resolve all dependencies for a doctype

		Args:
			doctype: Root doctype
			config: Configuration for link fields

		Returns:
			DependencyGraph instance
		"""
		graph = DependencyGraph(doctype, self.max_depth)
		config = config or {}

		# Clear visited set for new resolution
		self.visited_in_path = set()

		# Build the graph recursively
		self._resolve_node(graph.root, config)

		return graph

	def _resolve_node(self, node: DependencyNode, config: Dict):
		"""
		Recursively resolve dependencies for a node

		Args:
			node: Current dependency node
			config: Configuration dictionary
		"""
		# Check depth limit
		if node.depth >= self.max_depth:
			return

		# Check for cycles
		if node.doctype in self.visited_in_path:
			# Circular dependency detected, skip
			frappe.log(f"Circular dependency detected: {node.doctype} already in path")
			return

		# Add to visited path
		self.visited_in_path.add(node.doctype)

		try:
			# Get field parser
			from f_docgenesis.main.field_parser import DoctypeFieldParser
			parser = DoctypeFieldParser(node.doctype)

			# Process link fields
			for link_field in parser.get_link_fields():
				field_name = link_field['fieldname']
				target_doctype = link_field['options']

				# Get configuration for this field
				field_config = config.get(field_name, {})
				mode = field_config.get('mode', 'new')

				# Skip if mode is 'existing' and not generating new
				if mode == 'existing':
					continue

				# Skip system doctypes
				if self._is_system_doctype(target_doctype):
					continue

				# Add child dependency
				child_node = node.parent.add_dependency(
					parent_node=node,
					child_doctype=target_doctype,
					field_name=field_name
				) if hasattr(node, 'parent') and node.parent else None

				if not child_node:
					# For root node, add to graph
					if not hasattr(node, 'parent') or node.parent is None:
						from f_docgenesis.main.dependency_resolver import DependencyNode as DN
						child_node = DN(
							doctype=target_doctype,
							field_name=field_name,
							depth=node.depth + 1,
							parent=node
						)
						node.add_child(child_node)

				# Set mode and count
				if child_node:
					child_node.mode = mode
					child_node.count = field_config.get('new_count', 1)

					# Recursively resolve child dependencies
					self._resolve_node(child_node, config.get(field_name, {}))

			# Process child tables
			for child_table in parser.get_child_tables():
				child_doctype = child_table['options']
				field_name = child_table['fieldname']

				# Get child table config
				child_config = config.get(field_name, {})

				# Resolve dependencies for child table doctype
				child_parser = DoctypeFieldParser(child_doctype)
				for child_link in child_parser.get_link_fields():
					link_field_name = f"{field_name}.{child_link['fieldname']}"
					target_doctype = child_link['options']

					# Skip system doctypes
					if self._is_system_doctype(target_doctype):
						continue

					# Get config for child link field
					link_config = child_config.get(child_link['fieldname'], {})
					mode = link_config.get('mode', 'new')

					if mode != 'existing':
						# Add dependency for child table link
						pass  # Will be handled during generation

		finally:
			# Remove from visited path when done
			self.visited_in_path.discard(node.doctype)

	def _is_system_doctype(self, doctype: str) -> bool:
		"""
		Check if doctype is a system doctype that should be skipped

		Args:
			doctype: Doctype name

		Returns:
			True if system doctype
		"""
		system_doctypes = [
			'User', 'Role', 'DocType', 'DocField', 'Module Def',
			'File', 'Communication', 'Comment', 'Activity Log',
			'Version', 'Error Log', 'Scheduled Job Type',
			'Prepared Report', 'Report', 'Dashboard'
		]

		return doctype in system_doctypes

	def get_link_targets(self, doctype: str, field_name: str,
	                     filters: Optional[Dict] = None,
	                     limit: int = 10) -> List[str]:
		"""
		Get existing records for a link field

		Args:
			doctype: Target doctype
			field_name: Field name (for reference)
			filters: Optional filters
			limit: Maximum records to fetch

		Returns:
			List of document names
		"""
		try:
			filters = filters or {}
			records = frappe.get_all(
				doctype,
				filters=filters,
				fields=['name'],
				limit=limit,
				order_by='modified DESC'
			)
			return [r.name for r in records]
		except Exception as e:
			frappe.log_error(f"Error fetching link targets for {doctype}: {str(e)}")
			return []

	def check_circular_dependency(self, doctype: str,
	                               path: Optional[List[str]] = None) -> Tuple[bool, Optional[List[str]]]:
		"""
		Check if there's a circular dependency

		Args:
			doctype: Doctype to check
			path: Current dependency path

		Returns:
			Tuple of (has_circular, circular_path)
		"""
		path = path or []

		if doctype in path:
			return True, path + [doctype]

		from f_docgenesis.main.field_parser import DoctypeFieldParser
		parser = DoctypeFieldParser(doctype)

		for link_field in parser.get_link_fields():
			target_doctype = link_field['options']

			if self._is_system_doctype(target_doctype):
				continue

			has_circular, circular_path = self.check_circular_dependency(
				target_doctype,
				path + [doctype]
			)

			if has_circular:
				return True, circular_path

		return False, None

	def calculate_generation_complexity(self, doctype: str,
	                                     config: Optional[Dict] = None) -> Dict[str, Any]:
		"""
		Calculate complexity of generation

		Args:
			doctype: Root doctype
			config: Generation configuration

		Returns:
			Complexity metrics
		"""
		graph = self.resolve_dependencies(doctype, config)

		total_docs = sum(graph.doctype_counts.values())
		max_depth = max(n.depth for n in graph.all_nodes)

		return {
			'total_doctypes': len(graph.doctype_counts),
			'estimated_total_documents': total_docs,
			'max_dependency_depth': max_depth,
			'complexity_score': total_docs * max_depth,  # Simple metric
			'doctype_distribution': dict(graph.doctype_counts)
		}


def resolve_dependencies(doctype: str, config: Optional[Dict] = None,
                         max_depth: int = 10) -> DependencyGraph:
	"""
	Quick function to resolve dependencies

	Args:
		doctype: Root doctype
		config: Configuration dictionary
		max_depth: Maximum recursion depth

	Returns:
		DependencyGraph instance
	"""
	resolver = DependencyResolver(max_depth=max_depth)
	return resolver.resolve_dependencies(doctype, config)


def check_circular_dependency(doctype: str) -> Tuple[bool, Optional[List[str]]]:
	"""
	Quick function to check circular dependencies

	Args:
		doctype: Doctype to check

	Returns:
		Tuple of (has_circular, circular_path)
	"""
	resolver = DependencyResolver()
	return resolver.check_circular_dependency(doctype)

import frappe
import json
import os
from frappe.model.document import Document
from frappe.utils import get_site_path, now_datetime
from frappe.utils.file_manager import save_file

class DataSnapshot(Document):
	def before_save(self):
		if not self.timestamp:
			self.timestamp = now_datetime()

	@frappe.whitelist()
	def create_snapshot(self):
		"""
		Generates the snapshot file for the included doctypes.
		"""
		if not self.included_doctypes:
			frappe.throw("Please add at least one DocType to include in the snapshot.")

		data = {}
		total_count = 0

		for row in self.included_doctypes:
			doctype = row.doctype_name
			# Fetch all docs for this doctype
			# We use frappe.get_all to get names, then get_doc for full content including children
			# This might be slow for huge datasets, but safer for integrity
			names = frappe.get_all(doctype, pluck="name")
			row.count = len(names)
			total_count += len(names)

			docs = []
			for name in names:
				doc = frappe.get_doc(doctype, name)
				docs.append(doc.as_dict())

			data[doctype] = docs

		self.record_count = total_count

		# Save data to a JSON file
		content = json.dumps(data, indent=2, default=str)
		filename = f"{self.name}.json"

		# Save file using Frappe's file manager
		saved_file = save_file(
			fname=filename,
			content=content,
			dt=self.doctype,
			dn=self.name,
			is_private=1
		)

		self.snapshot_file = saved_file.file_url
		self.save()

		return True

	@frappe.whitelist()
	def restore_snapshot(self):
		"""
		Restores data from the snapshot file.
		WARNING: This deletes existing data for the included DocTypes!
		"""
		if not self.snapshot_file:
			frappe.throw("No snapshot file found.")

		# Get file content
		file_doc = frappe.get_doc("File", {"file_url": self.snapshot_file})
		content = file_doc.get_content()
		data = json.loads(content)

		# 1. Delete existing data
		# We iterate in reverse order of dependencies if possible, but for now we just delete
		# Users should be careful about dependencies.
		for row in self.included_doctypes:
			doctype = row.doctype_name
			frappe.db.delete(doctype)
			# Also delete child tables if necessary, but frappe.db.delete usually handles the parent
			# For child tables, we rely on the fact that they are deleted with parent

		frappe.db.commit()

		# 2. Insert data from snapshot
		for doctype, docs in data.items():
			for doc_data in docs:
				# Remove standard fields that shouldn't be set manually if they cause issues
				# But for a snapshot restore, we usually WANT to keep them (creation, modified, etc.)
				# We use frappe.get_doc(doc_data).insert()

				# We need to handle the case where the doc might still exist (if delete failed?)
				# But we just deleted it.

				# Clean up doc_data
				if "doctype" not in doc_data:
					doc_data["doctype"] = doctype

				d = frappe.get_doc(doc_data)
				d.flags.ignore_permissions = True
				d.flags.ignore_links = True # Important to avoid validation errors during bulk restore
				d.flags.ignore_validate = True # Optional: skip validation for speed/fidelity
				d.insert()

		self.is_restored = 1
		self.save()

		return True

import frappe
import json
from frappe.model.document import Document
from f_docgenesis.main.generator import generate_demo_data

class GenerationScenario(Document):
	@frappe.whitelist()
	def execute_scenario(self):
		"""
		Executes the scenario steps in order.
		"""
		if not self.steps:
			frappe.throw("No steps defined in this scenario.")

		results = []

		# Sort steps by sequence just in case
		steps = sorted(self.steps, key=lambda x: x.sequence or 0)

		for step in steps:
			frappe.publish_progress(
				percent=0,
				title=f"Executing Step: {step.doctype_name}",
				description=f"Generating {step.count} records..."
			)

			config = {}
			if step.config:
				try:
					config = json.loads(step.config)
				except:
					frappe.log_error(f"Invalid JSON config in step for {step.doctype_name}")

			# Execute generation
			try:
				step_result = generate_demo_data(
					doctype=step.doctype_name,
					count=step.count,
					config=config,
					max_depth=5, # Default depth, maybe make configurable
					dry_run=False
				)

				results.append({
					"step": step.doctype_name,
					"count": step.count,
					"created": step_result.get("records_created", 0),
					"success": True
				})

			except Exception as e:
				frappe.log_error(f"Error in scenario step {step.doctype_name}: {str(e)}")
				results.append({
					"step": step.doctype_name,
					"error": str(e),
					"success": False
				})
				# Should we stop or continue? Let's stop on error for now.
				frappe.throw(f"Scenario failed at step {step.doctype_name}: {str(e)}")

		return {
			"success": True,
			"results": results
		}

class FakerEngine:
	"""Generate realistic fake data for Frappe fields"""

	def __init__(self, locale: str = 'en_US'):
		"""
		Initialize Faker engine

		Args:
			locale: Locale for data generation (default: en_US)
		"""
		self.locale = locale
		self._faker = None

	@property
	def faker(self):
		"""Lazy load Faker library"""
		if not self._faker:

				from faker import Faker

				frappe.log_error("Faker library not installed. Install with: pip install Faker")

		return self._faker

	                   field_name: Optional[str] = None, **kwargs) -> Any:

			field_name: Name of the field (for context-aware generation)
			**kwargs: Additional parameters

		Returns:
			Generated value appropriate for the fieldtype
		"""

"""
Field Parser Module for F-DocGenesis
Analyzes doctypes and extracts field information, dependencies, and constraints
"""

import frappe
from frappe.model import no_value_fields
from typing import Dict, List, Any, Optional


class DoctypeFieldParser:
	"""Parser to extract and analyze doctype fields"""

	def __init__(self, doctype: str):
		"""
		Initialize parser for a doctype

		Args:
			doctype: Name of the doctype to parse
		"""
		self.doctype = doctype
		self.meta = frappe.get_meta(doctype)
		self._field_cache = {}

	def get_all_fields(self) -> List[Dict[str, Any]]:
		"""
		Get all fields for the doctype

		Returns:
			List of field dictionaries with metadata
		"""
		if self._field_cache.get('all_fields'):
			return self._field_cache['all_fields']

		fields = []
		for field in self.meta.fields:
			if self._should_include_field(field):
				fields.append(self._parse_field(field))

		self._field_cache['all_fields'] = fields
		return fields

	def get_link_fields(self) -> List[Dict[str, Any]]:
		"""
		Get all Link fields that reference other doctypes

		Returns:
			List of link field dictionaries
		"""
		if self._field_cache.get('link_fields'):
			return self._field_cache['link_fields']

		link_fields = []
		for field in self.meta.fields:
			if field.fieldtype == 'Link' and not field.hidden:
				link_fields.append({
					'fieldname': field.fieldname,
					'label': field.label,
					'options': field.options,  # Target doctype
					'reqd': field.reqd,
					'default': field.default,
					'description': field.description
				})

		self._field_cache['link_fields'] = link_fields
		return link_fields

	def get_child_tables(self) -> List[Dict[str, Any]]:
		"""
		Get all child table fields

		Returns:
			List of child table field dictionaries
		"""
		if self._field_cache.get('child_tables'):
			return self._field_cache['child_tables']

		child_tables = []
		for field in self.meta.fields:
			if field.fieldtype == 'Table' and not field.hidden:
				child_parser = DoctypeFieldParser(field.options)
				child_tables.append({
					'fieldname': field.fieldname,
					'label': field.label,
					'options': field.options,  # Child doctype
					'reqd': field.reqd,
					'description': field.description,
					'child_fields': child_parser.get_all_fields(),
					'child_link_fields': child_parser.get_link_fields()
				})

		self._field_cache['child_tables'] = child_tables
		return child_tables

	def get_mandatory_fields(self) -> List[Dict[str, Any]]:
		"""
		Get all mandatory fields

		Returns:
			List of mandatory field dictionaries
		"""
		return [f for f in self.get_all_fields() if f.get('reqd')]

	def get_unique_fields(self) -> List[str]:
		"""
		Get fields with unique constraints

		Returns:
			List of unique field names
		"""
		unique_fields = []
		for field in self.meta.fields:
			if field.unique:
				unique_fields.append(field.fieldname)
		return unique_fields

	def get_dependencies(self) -> Dict[str, str]:
		"""
		Get all doctype dependencies (via Link fields)

		Returns:
			Dictionary mapping field names to target doctypes
		"""
		dependencies = {}

		# Direct link fields
		for field in self.get_link_fields():
			dependencies[field['fieldname']] = field['options']

		# Child table dependencies
		for child_table in self.get_child_tables():
			child_doctype = child_table['options']
			dependencies[child_table['fieldname']] = child_doctype

			# Link fields within child tables
			for child_link in child_table.get('child_link_fields', []):
				dependencies[f"{child_table['fieldname']}.{child_link['fieldname']}"] = child_link['options']

		return dependencies

	def get_select_options(self, fieldname: str) -> Optional[List[str]]:
		"""
		Get options for a Select field

		Args:
			fieldname: Name of the select field

		Returns:
			List of options or None if not a select field
		"""
		field = self.meta.get_field(fieldname)
		if field and field.fieldtype == 'Select' and field.options:
			options = field.options.split('\n')
			return [opt.strip() for opt in options if opt.strip()]
		return None

	def get_default_value(self, fieldname: str) -> Any:
		"""
		Get default value for a field

		Args:
			fieldname: Name of the field

		Returns:
			Default value or None
		"""
		field = self.meta.get_field(fieldname)
		if field and field.default:
			return field.default
		return None

	def is_read_only_field(self, fieldname: str) -> bool:
		"""
		Check if a field is read-only

		Args:
			fieldname: Name of the field

		Returns:
			True if read-only, False otherwise
		"""
		field = self.meta.get_field(fieldname)
		return bool(field and (field.read_only or field.read_only_depends_on))

	def get_field_type(self, fieldname: str) -> Optional[str]:
		"""
		Get the fieldtype of a specific field

		Args:
			fieldname: Name of the field

		Returns:
			Fieldtype string or None
		"""
		field = self.meta.get_field(fieldname)
		return field.fieldtype if field else None

	def _should_include_field(self, field) -> bool:
		"""
		Determine if a field should be included in generation

		Args:
			field: Frappe field object

		Returns:
			True if field should be included
		"""
		# Exclude system fields and hidden fields
		if field.fieldtype in no_value_fields:
			return False

		if field.hidden or field.print_hide:
			return False

		# Exclude internal/system fields
		if field.fieldname in ['name', 'owner', 'creation', 'modified',
		                        'modified_by', 'docstatus', 'idx', 'parent',
		                        'parentfield', 'parenttype']:
			return False

		return True

	def _parse_field(self, field) -> Dict[str, Any]:
		"""
		Parse a single field into a dictionary

		Args:
			field: Frappe field object

		Returns:
			Field information dictionary
		"""
		return {
			'fieldname': field.fieldname,
			'label': field.label,
			'fieldtype': field.fieldtype,
			'options': field.options,
			'reqd': field.reqd,
			'default': field.default,
			'description': field.description,
			'unique': field.unique,
			'read_only': field.read_only or field.read_only_depends_on,
			'depends_on': field.depends_on,
			'precision': getattr(field, 'precision', None),
			'length': getattr(field, 'length', None),
			'in_list_view': field.in_list_view,
			'in_standard_filter': field.in_standard_filter
		}

	def get_naming_series(self) -> Optional[str]:
		"""
		Get naming series for the doctype

		Returns:
			Naming series pattern or None
		"""
		if self.meta.autoname:
			return self.meta.autoname
		return None

	def has_workflow(self) -> bool:
		"""
		Check if doctype has workflow

		Returns:
			True if workflow exists
		"""
		return bool(frappe.db.exists('Workflow', {'document_type': self.doctype}))

	def get_field_summary(self) -> Dict[str, Any]:
		"""
		Get a complete summary of all field information

		Returns:
			Dictionary with complete field analysis
		"""
		return {
			'doctype': self.doctype,
			'module': self.meta.module,
			'istable': self.meta.istable,
			'issingle': self.meta.issingle,
			'is_submittable': self.meta.is_submittable,
			'track_changes': self.meta.track_changes,
			'autoname': self.get_naming_series(),
			'has_workflow': self.has_workflow(),
			'all_fields': self.get_all_fields(),
			'link_fields': self.get_link_fields(),
			'child_tables': self.get_child_tables(),
			'mandatory_fields': self.get_mandatory_fields(),
			'unique_fields': self.get_unique_fields(),
			'dependencies': self.get_dependencies(),
			'field_count': len(self.get_all_fields())
		}


def get_field_parser(doctype: str) -> DoctypeFieldParser:
	"""
	Get a field parser instance for a doctype

	Args:
		doctype: Name of the doctype

	Returns:
		DoctypeFieldParser instance
	"""
	return DoctypeFieldParser(doctype)


def analyze_doctype(doctype: str) -> Dict[str, Any]:

	Args:
		doctype: Name of the doctype

	Returns:
		Complete field summary

class GenerationSession:
	"""Manages a single generation session"""

	def __init__(self, doctype: str, count: int = 10, config: Optional[Dict] = None,
	             max_depth: int = 10, dry_run: bool = False, session_id: Optional[str] = None):
		"""
		Initialize generation session

		Args:
			doctype: Target doctype
			count: Number of records to generate
			config: Generation configuration
			max_depth: Maximum dependency depth
			dry_run: Preview mode without creating records
			session_id: Optional session ID
		"""
		self.doctype = doctype
		self.count = count
		self.config = config or {}
		self.max_depth = max_depth
		self.dry_run = dry_run
		self.session_id = session_id or self._generate_session_id()

		self.faker = get_faker_engine()
		self.parser = DoctypeFieldParser(doctype)
		self.resolver = DependencyResolver(max_depth=max_depth)

		# Track generation
		self.generated_records: List[Dict] = []
		self.link_cache: Dict[str, List[str]] = {}  # Cache existing link targets
		self.stats = {
			'total_created': 0,
			'by_doctype': {},
			'start_time': None,
			'end_time': None
		}

	def _generate_session_id(self) -> str:
		"""Generate unique session ID"""
		from frappe.utils import now_datetime
		timestamp = now_datetime().strftime('%Y%m%d%H%M%S')
		return f"fdg-{timestamp}-{frappe.generate_hash(length=8)}"

	def _ensure_custom_fields(self):
		"""Ensure tracking fields exist for the doctype"""
		fields = [
			{
				'fieldname': 'custom_fdocgenesis_flag',
				'label': 'F-DocGenesis Generated',
				'fieldtype': 'Check',
				'default': '0',
				'read_only': 1,
				'hidden': 1
			},
			{
				'fieldname': 'custom_fdocgenesis_session',
				'label': 'Generation Session ID',
				'fieldtype': 'Data',
				'read_only': 1,
				'hidden': 1
			}
		]

		for field in fields:
			if not frappe.db.has_column(self.doctype, field['fieldname']):
				# Check if Custom Field exists
				if not frappe.db.exists('Custom Field', {'dt': self.doctype, 'fieldname': field['fieldname']}):
					try:
						custom_field = frappe.get_doc({
							'doctype': 'Custom Field',
							'dt': self.doctype,
							'fieldname': field['fieldname'],
							'label': field['label'],
							'fieldtype': field['fieldtype'],
							'default': field.get('default'),
							'read_only': field['read_only'],
							'hidden': field['hidden'],
							'insert_after': 'naming_series' # Try to put it early
						})
						custom_field.insert(ignore_permissions=True)
						frappe.db.commit()
					except Exception as e:
						# Ignore if it fails (e.g. already exists but not in DB yet?)
						frappe.log_error(f"Failed to create custom field {field['fieldname']} for {self.doctype}: {str(e)}")

	def generate(self) -> Dict[str, Any]:
		"""
		Execute the generation

		Returns:
			Generation result dictionary
		"""
		try:
			self._ensure_custom_fields()

			self.stats['start_time'] = datetime.now()

			frappe.log(f"Starting generation session: {self.session_id}")
			frappe.log(f"Target: {self.doctype}, Count: {self.count}")

			# Pre-calculate dates if configured
			date_config = self.config.get('date_settings', {})
			date_field = date_config.get('field')
			generated_dates = []
			if date_field and date_config.get('start_date'):
				generated_dates = get_date_distribution(
					start_date=date_config.get('start_date'),
					end_date=date_config.get('end_date'),
					count=self.count,
					distribution_type=date_config.get('distribution', 'Uniform'),
					params=date_config.get('params')
				)

			# Generate records - continue until we have the requested count
			successful_count = 0
			max_attempts = self.count * 2  # Safety limit to prevent infinite loops
			attempt = 0

			while successful_count < self.count and attempt < max_attempts:
				try:
					override_data = {}
					if generated_dates and successful_count < len(generated_dates):
						override_data[date_field] = generated_dates[successful_count]

					doc = self._generate_single_record(
						self.doctype,
						depth=0,
						is_primary=True,
						override_data=override_data
					)

					if doc:
						self.generated_records.append({
							'doctype': self.doctype,
							'docname': doc.name if not self.dry_run else f"DRY-RUN-{successful_count+1}",
							'is_primary': True,
							'depth': 0
						})
						self._increment_stat(self.doctype)
						successful_count += 1
					else:
						frappe.log(f"Failed to generate record for {self.doctype} (attempt {attempt + 1}), retrying...")

					attempt += 1

				except Exception as e:
					frappe.log_error(f"Error generating record {successful_count + 1} (attempt {attempt + 1}): {str(e)}")
					attempt += 1
					if not self.config.get('continue_on_error', True):
						raise

			# Check if we reached the target count
			if successful_count < self.count:
				error_msg = f"Could only generate {successful_count} out of {self.count} requested {self.doctype} records after {attempt} attempts. This may be due to duplicate constraints or validation errors."
				frappe.log_error(error_msg, "F-DocGenesis Generation")
				if not self.config.get('allow_partial', False):
					frappe.throw(error_msg)

			self.stats['end_time'] = datetime.now()
			self.stats['total_created'] = len(self.generated_records)

			return self._build_result()

		except Exception as e:
			frappe.log_error(f"Generation session failed: {str(e)}")
			raise

	def _generate_single_record(self, doctype: str, depth: int = 0,
	                             is_primary: bool = False, field_name: Optional[str] = None,
	                             override_data: Optional[Dict] = None) -> Optional[Any]:
		"""
		Generate a single record with all dependencies

		Args:
			doctype: Doctype to generate
			depth: Current recursion depth
			is_primary: Is this a primary record
			field_name: Field name (for child records)

		Returns:
			Created document or None
		"""
		if depth > self.max_depth:
			return None

		max_retries = 5
		for attempt in range(max_retries):
			try:
				parser = DoctypeFieldParser(doctype)

				# Create new document
				doc = frappe.new_doc(doctype)

				# Set field values
				for field in parser.get_all_fields():
					fieldname = field['fieldname']
					fieldtype = field['fieldtype']
					options = field.get('options')
					reqd = field.get('reqd', False)

					# Apply overrides if present
					if override_data and fieldname in override_data:
						doc.set(fieldname, override_data[fieldname])
						continue

					# Skip certain fields
					if fieldname in ['name', 'naming_series']:
						# Handle naming series
						if fieldtype == 'Select' and options and 'naming_series' in fieldname:
							naming_options = [opt.strip() for opt in options.split('\n') if opt.strip()]
							if naming_options:
								doc.naming_series = naming_options[0]
						continue

					# Handle mandatory 'title' field specially to ensure it's always set
					if fieldname == 'title' and reqd:
						# Generate unique title with timestamp
						title_value = self.faker.generate_value(
							fieldtype='Data',
							field_name=fieldname,
							options=options
						)
						if title_value:
							doc.set(fieldname, title_value)
						else:
							# Fallback to ensure title is set
							doc.set(fieldname, f"{doctype} {frappe.generate_hash(length=8)}")
						continue

					# Handle Link fields
					if fieldtype == 'Link' and options:
						link_value = self._resolve_link_field(
							fieldname=fieldname,
							target_doctype=options,
							reqd=reqd,
							depth=depth
						)
						if link_value:
							doc.set(fieldname, link_value)

					# Handle Table (child table) fields
					elif fieldtype == 'Table' and options:
						self._populate_child_table(doc, fieldname, options, depth)

					# Handle Select fields
					elif fieldtype == 'Select' and options:
						select_value = self.faker.generate_value(
							fieldtype='Select',
							options=options,
							field_name=fieldname
						)
						if select_value:
							doc.set(fieldname, select_value)

					# Handle all other field types
					elif fieldtype not in ['Table', 'Link', 'Section Break', 'Column Break', 'HTML']:
						value = self.faker.generate_value(
							fieldtype=fieldtype,
							field_name=fieldname,
							options=options
						)
						if value is not None:
							doc.set(fieldname, value)

				# Check for duplicate before inserting
				if not self.dry_run and self._is_duplicate(doc, parser):
					frappe.log(f"Skipping duplicate record for {doctype}, retrying with new data...")
					continue  # Retry with new generated values

				# Add generation flag and session tracking
				if hasattr(doc, 'custom_fdocgenesis_flag'):
					doc.custom_fdocgenesis_flag = 1

				# Store session ID for tracking which records were generated together
				if hasattr(doc, 'custom_fdocgenesis_session'):
					doc.custom_fdocgenesis_session = self.session_id

				# Save document
				if not self.dry_run:
					try:
						doc.insert(ignore_permissions=True)
						frappe.db.commit()
					except frappe.exceptions.DuplicateEntryError as e:
						frappe.log(f"Duplicate entry detected for {doctype}: {str(e)}, retrying...")
						continue  # Retry with new data
					except Exception as insert_error:
						# Check if it's an integrity error (duplicate)
						error_msg = str(insert_error).lower()
						if 'duplicate' in error_msg or 'integrity' in error_msg:
							frappe.log(f"Duplicate/Integrity error for {doctype}: {str(insert_error)}, retrying...")
							continue  # Retry with new data
						else:
							# Re-raise if it's not a duplicate error
							raise

					# Apply workflow state if configured
					if not self.dry_run:
						self._apply_workflow_state(doc)

				return doc

			except Exception as e:
				error_msg = str(e).lower()
				# If it's a duplicate/integrity error and we have retries left, continue
				if attempt < max_retries - 1 and ('duplicate' in error_msg or 'integrity' in error_msg):
					frappe.log(f"Error generating {doctype} (attempt {attempt + 1}): {str(e)}, retrying...")
					continue
				else:
					frappe.log_error(f"Error generating {doctype}: {str(e)}")
					return None

		# If all retries failed
		frappe.log_error(f"Failed to generate {doctype} after {max_retries} attempts (duplicate entries)")
		return None

	def _resolve_link_field(self, fieldname: str, target_doctype: str,
	                        reqd: bool = False, depth: int = 0) -> Optional[str]:
		"""
		Resolve a link field value

		Args:
			fieldname: Name of the link field
			target_doctype: Target doctype for the link
			reqd: Is field required
			depth: Current depth

		Returns:
			Document name or None
		"""
		# Get config for this field
		field_config = self.config.get(fieldname, {})
		mode = field_config.get('mode', 'new')
		use_existing = field_config.get('use_existing', False)  # Handle "use existing" checkbox
		existing_count = field_config.get('existing_count', 0)
		new_count = field_config.get('new_count', 1)
		filters = field_config.get('filters', {})

		# If "use_existing" checkbox is checked, override mode to 'existing'
		if use_existing:
			mode = 'existing'

		# Mode: Use Existing
		if mode == 'existing':
			existing_record = self._get_existing_record(target_doctype, filters)
			# If required and no existing record found, create one
			if not existing_record and reqd and depth + 1 <= self.max_depth:
				frappe.log(f"No existing {target_doctype} found for required field {fieldname}, creating new one")
				new_doc = self._generate_single_record(
					target_doctype,
					depth=depth + 1,
					is_primary=False,
					field_name=fieldname
				)
				if new_doc:
					self._increment_stat(target_doctype)
					self.generated_records.append({
						'doctype': target_doctype,
						'docname': new_doc.name if not self.dry_run else f"DRY-RUN-{target_doctype}",
						'is_primary': False,
						'depth': depth + 1,
						'field_name': fieldname
					})
					return new_doc.name if not self.dry_run else f"DRY-RUN-{target_doctype}"
			return existing_record

		# Mode: Generate New
		elif mode == 'new':
			if depth + 1 <= self.max_depth:
				new_doc = self._generate_single_record(
					target_doctype,
					depth=depth + 1,
					is_primary=False,
					field_name=fieldname
				)
				if new_doc:
					self._increment_stat(target_doctype)
					self.generated_records.append({
						'doctype': target_doctype,
						'docname': new_doc.name if not self.dry_run else f"DRY-RUN-{target_doctype}",
						'is_primary': False,
						'depth': depth + 1,
						'field_name': fieldname
					})
					return new_doc.name if not self.dry_run else f"DRY-RUN-{target_doctype}"
			return None

		# Mode: Hybrid
		elif mode == 'hybrid':
			# Decide randomly whether to use existing or generate new
			if random.random() < 0.5 and existing_count > 0:
				return self._get_existing_record(target_doctype, filters)
			else:
				if depth + 1 <= self.max_depth:
					new_doc = self._generate_single_record(
						target_doctype,
						depth=depth + 1,
						is_primary=False,
						field_name=fieldname
					)
					if new_doc:
						self._increment_stat(target_doctype)
						return new_doc.name if not self.dry_run else f"DRY-RUN-{target_doctype}"
			return None

		return None

	def _get_existing_record(self, doctype: str, filters: Optional[Dict] = None) -> Optional[str]:
		"""
		Get an existing record name

		Args:
			doctype: Doctype to query
			filters: Optional filters

		Returns:
			Document name or None
		"""
		# Use cache
		cache_key = f"{doctype}:{json.dumps(filters or {}, sort_keys=True)}"
		if cache_key not in self.link_cache:
			try:
				records = frappe.get_all(
					doctype,
					filters=filters or {},
					fields=['name'],
					limit=50,
					order_by='modified DESC'
				)
				self.link_cache[cache_key] = [r.name for r in records]
			except Exception as e:
				frappe.log_error(f"Error fetching existing {doctype}: {str(e)}")
				self.link_cache[cache_key] = []

		if self.link_cache[cache_key]:
			return random.choice(self.link_cache[cache_key])

		return None

	def _populate_child_table(self, parent_doc, fieldname: str,
	                          child_doctype: str, depth: int):
		"""
		Populate a child table

		Args:
			parent_doc: Parent document
			fieldname: Child table field name
			child_doctype: Child table doctype
			depth: Current depth
		"""
		# Get config for child table
		child_config = self.config.get(fieldname, {})
		rows = child_config.get('rows', random.randint(1, 5))

		child_parser = DoctypeFieldParser(child_doctype)

		for i in range(rows):
			child_row = parent_doc.append(fieldname, {})

			# Populate child row fields
			for field in child_parser.get_all_fields():
				child_fieldname = field['fieldname']
				child_fieldtype = field['fieldtype']
				child_options = field.get('options')

				# Skip system fields
				if child_fieldname in ['name', 'parent', 'parenttype', 'parentfield', 'idx']:
					continue

				# Handle Link fields in child table
				if child_fieldtype == 'Link' and child_options:
					# Get child field config
					child_field_config = child_config.get(child_fieldname, {})
					link_value = self._resolve_link_field(
						fieldname=child_fieldname,
						target_doctype=child_options,
						reqd=field.get('reqd', False),
						depth=depth
					)
					if link_value:
						child_row.set(child_fieldname, link_value)

				# Handle Select fields
				elif child_fieldtype == 'Select' and child_options:
					select_value = self.faker.generate_value(
						fieldtype='Select',
						options=child_options,
						field_name=child_fieldname
					)
					if select_value:
						child_row.set(child_fieldname, select_value)

				# Handle other field types
				elif child_fieldtype not in ['Table', 'Section Break', 'Column Break', 'HTML']:
					value = self.faker.generate_value(
						fieldtype=child_fieldtype,
						field_name=child_fieldname,
						options=child_options
					)
					if value is not None:
						child_row.set(child_fieldname, value)

	def _is_duplicate(self, doc, parser: DoctypeFieldParser) -> bool:
		"""
		Check if document is a duplicate

		Args:
			doc: Document to check
			parser: DoctypeFieldParser for the doctype

		Returns:
			True if duplicate exists, False otherwise
		"""
		try:
			# Get unique fields for this doctype
			unique_fields = parser.get_unique_fields()

			# Check each unique field
			for fieldname in unique_fields:
				field_value = doc.get(fieldname)
				if field_value:
					# Check if a record with this unique value exists
					if frappe.db.exists(doc.doctype, {fieldname: field_value}):
						return True

			# Check primary key (name field) for doctypes with specific naming
			meta = frappe.get_meta(doc.doctype)

			# For doctypes that use title as name or have specific patterns
			if hasattr(doc, 'title') and doc.title:
				# Check if doc with same title exists
				existing = frappe.db.exists(doc.doctype, {'title': doc.title})
				if existing:
					return True

			# For standard naming (check common identifying fields)
			identifying_fields = ['name', 'title', 'subject', 'item_name',
			                      'customer_name', 'supplier_name', 'employee_name']

			for field in identifying_fields:
				if hasattr(doc, field) and doc.get(field):
					field_value = doc.get(field)
					# Check if value looks like it could be a primary key
					try:
						existing = frappe.db.exists(doc.doctype, field_value)
						if existing:
							return True
					except:
						pass

			return False

		except Exception as e:
			# If we can't determine, assume it's not a duplicate
			frappe.log(f"Error checking duplicate for {doc.doctype}: {str(e)}")
			return False

	def _increment_stat(self, doctype: str):
		"""Increment statistics for a doctype"""
		if doctype not in self.stats['by_doctype']:
			self.stats['by_doctype'][doctype] = {'new': 0, 'existing': 0}
		self.stats['by_doctype'][doctype]['new'] += 1

	def _build_result(self) -> Dict[str, Any]:
		"""Build the final result dictionary"""
		duration = (self.stats['end_time'] - self.stats['start_time']).total_seconds()

		return {
			'success': True,
			'session_id': self.session_id,
			'doctype': self.doctype,
			'count': self.count,
			'dry_run': self.dry_run,
			'records_created': len([r for r in self.generated_records if r['is_primary']]),
			'total_documents': len(self.generated_records),
			'generation_time': round(duration, 2),
			'stats': self.stats['by_doctype'],
			'records': [r['docname'] for r in self.generated_records if r['is_primary']]
		}

"""
Template Manager Module for F-DocGenesis
Manages reusable generation configuration templates
"""

import frappe
from typing import Dict, List, Optional, Any
import json
from datetime import datetime


class TemplateManager:
	"""Manages generation templates"""

	def __init__(self):
		self.templates = {}

	def save_template(self, name: str, doctype: str, config: Dict,
	                  description: Optional[str] = None,
	                  profile: str = 'Standard',
	                  is_public: bool = False,
	                  ai_prompt: Optional[str] = None,
	                  ai_provider: Optional[str] = None) -> Dict[str, Any]:
		"""
		Save a generation template

		Args:
			name: Template name
			doctype: Target doctype
			config: Generation configuration
			description: Optional description
			profile: Generation profile (Minimal/Standard/Rich)
			is_public: Make template available to all users
			ai_prompt: Optional AI prompt for AI-powered generation
			ai_provider: AI provider to use (openai/anthropic/local)

		Returns:
			Saved template dictionary
		"""
		try:
			# Check if template exists
			existing = self.get_template(name)

			template_data = {
				'name': name,
				'doctype': doctype,
				'config': config,
				'description': description or '',
				'profile': profile,
				'is_public': is_public,
				'ai_prompt': ai_prompt,
				'ai_provider': ai_provider,
				'created_by': frappe.session.user,
				'modified_at': frappe.utils.now_datetime(),
				'version': existing.get('version', 0) + 1 if existing else 1
			}

			if not existing:
				template_data['created_at'] = frappe.utils.now_datetime()
				template_data['usage_count'] = 0

			# Store in cache
			self.templates[name] = template_data

			# In production, this would save to database
			# For now, we'll use frappe.cache
			cache_key = f"fdocgenesis_template_{name}"
			frappe.cache().set_value(cache_key, json.dumps(template_data))

			return {
				'success': True,
				'template': template_data
			}

		except Exception as e:
			frappe.log_error(f"Template save error: {str(e)}")
			return {
				'success': False,
				'error': str(e)
			}

	def get_template(self, name: str) -> Optional[Dict]:
		"""
		Get a template by name

		Args:
			name: Template name

		Returns:
			Template dictionary or None
		"""
		# Check cache first
		if name in self.templates:
			return self.templates[name]

		# Try frappe cache
		cache_key = f"fdocgenesis_template_{name}"
		cached = frappe.cache().get_value(cache_key)

		if cached:
			try:
				template = json.loads(cached)
				self.templates[name] = template
				return template
			except:
				pass

		return None

	def list_templates(self, doctype: Optional[str] = None,
	                   user: Optional[str] = None,
	                   public_only: bool = False) -> List[Dict]:
		"""
		List available templates

		Args:
			doctype: Filter by doctype
			user: Filter by creator
			public_only: Show only public templates

		Returns:
			List of template summaries
		"""
		templates = []

		# Get all template keys from cache
		cache_pattern = "fdocgenesis_template_*"
		all_keys = frappe.cache().get_keys(cache_pattern)

		for key in all_keys:
			try:
				cached = frappe.cache().get_value(key)
				if not cached:
					continue

				template = json.loads(cached)

				# Apply filters
				if doctype and template.get('doctype') != doctype:
					continue

				if user and template.get('created_by') != user:
					continue

				if public_only and not template.get('is_public'):
					continue

				# Check permissions
				if not template.get('is_public'):
					if template.get('created_by') != frappe.session.user:
						continue

				templates.append({
					'name': template['name'],
					'doctype': template['doctype'],
					'description': template.get('description'),
					'profile': template.get('profile'),
					'created_by': template.get('created_by'),
					'created_at': template.get('created_at'),
					'version': template.get('version'),
					'usage_count': template.get('usage_count', 0),
					'is_public': template.get('is_public', False),
					'ai_prompt': template.get('ai_prompt'),
					'ai_provider': template.get('ai_provider')
				})

			except Exception as e:
				frappe.log_error(f"Error loading template from {key}: {str(e)}")
				continue

		return sorted(templates, key=lambda t: t.get('usage_count', 0), reverse=True)

	def delete_template(self, name: str) -> Dict[str, Any]:
		"""
		Delete a template

		Args:
			name: Template name

		Returns:
			Result dictionary
		"""
		try:
			template = self.get_template(name)

			if not template:
				return {
					'success': False,
					'error': f'Template "{name}" not found'
				}

			# Check permissions
			if template.get('created_by') != frappe.session.user:
				if not frappe.has_permission('System Manager'):
					return {
						'success': False,
						'error': 'Insufficient permissions to delete template'
					}

			# Delete from cache
			cache_key = f"fdocgenesis_template_{name}"
			frappe.cache().delete_value(cache_key)

			# Remove from local cache
			if name in self.templates:
				del self.templates[name]

			return {
				'success': True,
				'message': f'Template "{name}" deleted'
			}

		except Exception as e:
			frappe.log_error(f"Template delete error: {str(e)}")
			return {
				'success': False,
				'error': str(e)
			}

	def export_template(self, name: str) -> Dict[str, Any]:
		"""
		Export template as JSON

		Args:
			name: Template name

		Returns:
			Template data for export
		"""
		template = self.get_template(name)

		if not template:
			return {
				'success': False,
				'error': f'Template "{name}" not found'
			}

		# Create export package
		export_data = {
			'fdocgenesis_template': True,
			'version': '1.0',
			'exported_at': frappe.utils.now_datetime().isoformat(),
			'exported_by': frappe.session.user,
			'template': {
				'name': template['name'],
				'doctype': template['doctype'],
				'config': template['config'],
				'description': template.get('description'),
				'profile': template.get('profile'),
				'version': template.get('version')
			}
		}

		return {
			'success': True,
			'data': export_data
		}

	def import_template(self, data: Dict, overwrite: bool = False) -> Dict[str, Any]:
		"""
		Import template from JSON

		Args:
			data: Imported template data
			overwrite: Overwrite if exists

		Returns:
			Import result
		"""
		try:
			# Validate import data
			if not data.get('fdocgenesis_template'):
				return {
					'success': False,
					'error': 'Invalid template file'
				}

			template_data = data.get('template')
			if not template_data:
				return {
					'success': False,
					'error': 'No template data found'
				}

			name = template_data['name']

			# Check if exists
			existing = self.get_template(name)
			if existing and not overwrite:
				return {
					'success': False,
					'error': f'Template "{name}" already exists. Use overwrite=True to replace.'
				}

			# Save template
			result = self.save_template(
				name=name,
				doctype=template_data['doctype'],
				config=template_data['config'],
				description=template_data.get('description'),
				profile=template_data.get('profile', 'Standard'),
				is_public=False  # Imported templates are private by default
			)

			if result.get('success'):
				return {
					'success': True,
					'message': f'Template "{name}" imported successfully'
				}
			else:
				return result

		except Exception as e:
			frappe.log_error(f"Template import error: {str(e)}")
			return {
				'success': False,
				'error': str(e)
			}

					'success': False,

				}

			# Increment usage count
			self.increment_usage(name)

			# Determine generation count
			generation_count = count if count is not None else 10

			frappe.log_error(f"Error using template: {str(e)}")

			}

"""
Validators Module for F-DocGenesis
Validates generated data against business rules and constraints
"""

import frappe
from typing import Dict, List, Optional, Any, Tuple


class DataValidator:
	"""Validates generated data for correctness and business rules"""

	def __init__(self, doctype: str):
		"""
		Initialize validator

		Args:
			doctype: Target doctype
		"""
		self.doctype = doctype
		self.meta = frappe.get_meta(doctype)
		self.errors = []
		self.warnings = []

	def validate_generation_config(self, config: Dict, count: int) -> Tuple[bool, List[str]]:
		"""
		Validate generation configuration before generating

		Args:
			config: Generation configuration
			count: Number of records to generate

		Returns:
			Tuple of (is_valid, error_messages)
		"""
		self.errors = []
		self.warnings = []

		# Check if doctype exists
		if not frappe.db.exists('DocType', self.doctype):
			self.errors.append(f"Doctype '{self.doctype}' does not exist")
			return False, self.errors

		# Check permissions
		if not frappe.has_permission(self.doctype, 'create'):
			self.errors.append(f"Insufficient permissions to create {self.doctype}")
			return False, self.errors

		# Validate count
		if count <= 0:
			self.errors.append("Count must be greater than 0")
		elif count > 10000:
			self.warnings.append("Large count (>10000) may take significant time")

		# Validate mandatory fields can be generated
		from f_docgenesis.main.field_parser import DoctypeFieldParser
		parser = DoctypeFieldParser(self.doctype)

		for field in parser.get_mandatory_fields():
			fieldtype = field['fieldtype']

			# Check if fieldtype is supported
			if fieldtype in ['Attach', 'Attach Image', 'Signature']:
				self.warnings.append(
					f"Mandatory field '{field['fieldname']}' requires file upload - will use placeholder"
				)

		# Validate link field configuration
		for field in parser.get_link_fields():
			fieldname = field['fieldname']
			target_doctype = field['options']

			if fieldname in config:
				field_config = config[fieldname]
				mode = field_config.get('mode', 'new')

				# If using existing, check if records exist
				if mode in ['existing', 'hybrid']:
					existing_count = frappe.db.count(target_doctype)

					if existing_count == 0:
						self.errors.append(
							f"Cannot use existing {target_doctype} records - none exist"
						)
					elif field_config.get('existing_count', 0) > existing_count:
						self.warnings.append(
							f"Requested {field_config['existing_count']} existing {target_doctype} but only {existing_count} available"
						)

		# Check for circular dependencies
		from f_docgenesis.main.dependency_resolver import check_circular_dependency
		has_circular, path = check_circular_dependency(self.doctype)

		if has_circular:
			self.warnings.append(
				f"Circular dependency detected: {' → '.join(path)}"
			)

		# Check if doctype is submittable
		if self.meta.is_submittable:
			self.warnings.append(
				"Doctype is submittable - records will be created in Draft state"
			)

		return len(self.errors) == 0, self.errors

	def validate_generated_doc(self, doc: Any) -> Tuple[bool, List[str]]:
		"""
		Validate a generated document

		Args:
			doc: Document object to validate

		Returns:
			Tuple of (is_valid, error_messages)
		"""
		errors = []

		try:
			# Check mandatory fields
			for field in self.meta.get_mandatory_fields():
				if not doc.get(field.fieldname):
					errors.append(f"Mandatory field '{field.fieldname}' is empty")

			# Validate against doctype validation rules
			# This calls the doctype's validate() method
			try:
				doc.run_method('validate')
			except Exception as e:
				errors.append(f"Validation error: {str(e)}")

			# Check unique constraints
			for field in self.meta.fields:
				if field.unique and doc.get(field.fieldname):
					# Check if value already exists
					existing = frappe.db.exists(self.doctype, {
						field.fieldname: doc.get(field.fieldname)
					})
					if existing and existing != doc.name:
						errors.append(
							f"Duplicate value for unique field '{field.fieldname}': {doc.get(field.fieldname)}"
						)

		except Exception as e:
			errors.append(f"Validation exception: {str(e)}")

		return len(errors) == 0, errors

	def validate_business_rules(self, doc: Any) -> Tuple[bool, List[str]]:
		"""
		Validate business rules for generated document

		Args:
			doc: Document object

		Returns:
			Tuple of (is_valid, error_messages)
		"""
		errors = []

		# Custom business rule validations can be added here
		# These are application-specific

		# Example: Check date ranges
		if hasattr(doc, 'from_date') and hasattr(doc, 'to_date'):
			if doc.from_date and doc.to_date:
				if doc.from_date > doc.to_date:
					errors.append("'From Date' cannot be after 'To Date'")

		# Example: Check numeric ranges
		for field in self.meta.fields:
			if field.fieldtype in ['Int', 'Float', 'Currency']:
				value = doc.get(field.fieldname)
				if value is not None and value < 0:
					# Check if negative values are allowed
					if 'allow_negative' not in field.options:
						errors.append(
							f"Field '{field.fieldname}' cannot be negative"
						)

		return len(errors) == 0, errors

	def get_validation_summary(self) -> Dict[str, Any]:
		"""
		Get summary of validation results

		Returns:
			Validation summary dictionary
		"""
		return {
			'errors': self.errors,
			'warnings': self.warnings,
			'error_count': len(self.errors),
			'warning_count': len(self.warnings),
			'is_valid': len(self.errors) == 0
		}


def validate_config(doctype: str, config: Dict, count: int) -> Dict[str, Any]:
	"""
	Validate generation configuration

	Args:
		doctype: Target doctype
		config: Generation configuration
		count: Number of records

	Returns:
		Validation result dictionary
	"""
	validator = DataValidator(doctype)
	is_valid, errors = validator.validate_generation_config(config, count)

	return {
		'success': is_valid,
		'errors': errors,
		'warnings': validator.warnings
	}


def validate_doc(doc: Any) -> Dict[str, Any]:
	"""
	Validate a generated document

	Args:
		doc: Document object

	Returns:
		Validation result dictionary
	"""
	validator = DataValidator(doc.doctype)

	# Validate document structure
	is_valid_structure, structure_errors = validator.validate_generated_doc(doc)

	# Validate business rules
	is_valid_business, business_errors = validator.validate_business_rules(doc)

	all_errors = structure_errors + business_errors

	return {
		'success': len(all_errors) == 0,
		'errors': all_errors,
		'warnings': validator.warnings
	}

import frappe
from frappe import _

@frappe.whitelist()
def get_data_relationships(doctype, docname, depth=1):
	"""
	Returns nodes and edges for the relationship graph.
	"""
	nodes = []
	edges = []
	visited = set()
	queue = [(doctype, docname, 0)]

	# Add root node
	nodes.append({
		"data": {
			"id": docname,
			"label": docname,
			"doctype": doctype,
			"color": get_color_for_doctype(doctype)
		}
	})
	visited.add((doctype, docname))

	while queue:
		current_doctype, current_docname, current_depth = queue.pop(0)

		if current_depth >= int(depth):
			continue

		try:
			doc = frappe.get_doc(current_doctype, current_docname)
			meta = frappe.get_meta(current_doctype)

			# 1. Check Link fields in the document
			for field in meta.fields:
				if field.fieldtype == "Link" and field.options:
					value = doc.get(field.fieldname)
					if value:
						target_doctype = field.options
						target_docname = value

						add_node_and_edge(
							nodes, edges, visited, queue,
							source=(current_doctype, current_docname),
							target=(target_doctype, target_docname),
							label=field.label,
							depth=current_depth + 1
						)

			# 2. Check Child Tables
			for field in meta.fields:
				if field.fieldtype == "Table" and field.options:
					child_doctype = field.options
					children = doc.get(field.fieldname) or []

					# We don't add child table rows as nodes, but we check links INSIDE them
					child_meta = frappe.get_meta(child_doctype)
					for child in children:
						for child_field in child_meta.fields:
							if child_field.fieldtype == "Link" and child_field.options:
								value = child.get(child_field.fieldname)
								if value:
									target_doctype = child_field.options
									target_docname = value

									add_node_and_edge(
										nodes, edges, visited, queue,
										source=(current_doctype, current_docname),
										target=(target_doctype, target_docname),
										label=f"{field.label} > {child_field.label}",
										depth=current_depth + 1
									)

		except frappe.DoesNotExistError:
			continue

	return {
		"nodes": nodes,
		"edges": edges
	}

def add_node_and_edge(nodes, edges, visited, queue, source, target, label, depth):
	source_doctype, source_docname = source
	target_doctype, target_docname = target

	if (target_doctype, target_docname) not in visited:
		nodes.append({
			"data": {
				"id": target_docname,
				"label": target_docname,
				"doctype": target_doctype,
				"color": get_color_for_doctype(target_doctype)
			}
		})
		visited.add((target_doctype, target_docname))
		queue.append((target_doctype, target_docname, depth))

	# Add edge (avoid duplicates if possible, but cytoscape handles multi-edges)
	edge_id = f"{source_docname}-{target_docname}-{label}"
	edges.append({
		"data": {
			"id": edge_id,
			"source": source_docname,
			"target": target_docname,
			"label": label
		}
	})

def get_color_for_doctype(doctype):
	# Simple hash to color or predefined map
	colors = [
		"#FF6B6B", "#4ECDC4", "#45B7D1", "#96CEB4", "#FFEEAD",
		"#D4A5A5", "#9B59B6", "#3498DB", "#E67E22", "#2ECC71"
	]
	return colors[hash(doctype) % len(colors)]

import random
import math
from datetime import timedelta, datetime
from frappe.utils import getdate, add_days

def get_date_distribution(start_date, end_date, count, distribution_type="Uniform", params=None):
	"""
	Returns a list of dates based on the specified distribution.
	"""
	start_date = getdate(start_date)
	end_date = getdate(end_date)
	days_diff = (end_date - start_date).days

	if days_diff <= 0:
		return [start_date] * count

	dates = []
	params = params or {}

	if distribution_type == "Uniform":
		for _ in range(count):
			random_days = random.randint(0, days_diff)
			dates.append(add_days(start_date, random_days))

	elif distribution_type == "Linear Growth":
		# Higher probability for later dates
		# We use a triangular distribution or similar weighting
		for _ in range(count):
			# random.triangular(low, high, mode)
			# mode=days_diff means peak at the end
			random_days = int(random.triangular(0, days_diff, days_diff))
			dates.append(add_days(start_date, random_days))

	elif distribution_type == "Exponential Growth":
		# Stronger growth than linear
		# x = r^2 where r is uniform [0, 1], mapped to days
		for _ in range(count):
			r = random.random()
			# Square root to bias towards 1 (end of range)
			# Wait, to bias towards end (1), we want more values near 1.
			# If we do x = r^2, values bunch near 0.
			# We want x = 1 - (1-r)^2 or similar.
			# Let's use power law: x = r^(1/power)
			# If power > 1, biases towards 1.
			power = params.get("growth_factor", 3)
			random_factor = 1 - (random.random() ** power) # Bias towards 0? No.
			# Let's keep it simple:
			# random.random() ** 0.5 biases towards 1
			# random.random() ** 2 biases towards 0

			# We want growth, so more dates at the end.
			# So we want bias towards days_diff.
			# random.random()^(1/3) -> biases towards 1.
			factor = random.random() ** (1.0 / power)
			random_days = int(factor * days_diff)
			dates.append(add_days(start_date, random_days))

	elif distribution_type == "Seasonal":
		# Peaks in specific months
		peak_months = params.get("peak_months", [12]) # Default December
		# Simple rejection sampling or weighted choice
		# Let's use weighted choice for efficiency

		# 1. Assign weights to each day
		day_weights = []
		current = start_date
		total_weight = 0

		# Optimization: Calculate weights for 366 days and map?
		# Or just iterate if range is small. For large range, maybe approximate.
		# Let's iterate, it's usually < 5 years.

		days_list = []
		for i in range(days_diff + 1):
			day = add_days(start_date, i)
			month = day.month
			weight = 1.0
			if month in peak_months:
				weight = 5.0 # 5x more likely in peak months

			# Weekend adjustment?
			if params.get("weekend_boost") and day.weekday() >= 5:
				weight *= 1.5

			day_weights.append(weight)
			days_list.append(day)
			total_weight += weight

		# 2. Select dates
		# Python 3.6+ random.choices is available
		dates = random.choices(days_list, weights=day_weights, k=count)

	else:
		# Fallback to uniform
		for _ in range(count):
			random_days = random.randint(0, days_diff)
			dates.append(add_days(start_date, random_days))

	dates.sort()
	return dates

import frappe

def on_doctype_update(doc, method):
	"""
	Called on DocType update/insert.
	Ensures that the tracking fields exist for the doctype.
	"""
	if doc.custom:
		# Only add to custom doctypes or if specifically configured?
		# User asked for "if this app is installed then a check box should be created for the doctypes"
		# implying ALL doctypes, or at least those where we want to generate data.
		# Let's add to all non-system doctypes to be safe and consistent.
		pass

	# Skip system doctypes
	if doc.module in ['Core', 'Custom', 'Desk', 'Integrations', 'Email', 'Printing', 'Workflow']:
		return

	# Skip single and virtual doctypes (they don't have their own table)
	if doc.issingle or doc.is_virtual:
		return

	ensure_tracking_fields(doc.name)

def ensure_tracking_fields(doctype):
	"""Ensure tracking fields exist for the doctype"""
	fields = [
		{
			'fieldname': 'custom_fdocgenesis_flag',
			'label': 'F-DocGenesis Generated',
			'fieldtype': 'Check',
			'default': '0',
			'read_only': 1,
			'hidden': 1
		},
		{
			'fieldname': 'custom_fdocgenesis_session',
			'label': 'Generation Session ID',
			'fieldtype': 'Data',
			'read_only': 1,
			'hidden': 1
		}
	]

	for field in fields:
		if not frappe.db.has_column(doctype, field['fieldname']):
			# Check if Custom Field exists
			if not frappe.db.exists('Custom Field', {'dt': doctype, 'fieldname': field['fieldname']}):
				try:
					custom_field = frappe.get_doc({
						'doctype': 'Custom Field',
						'dt': doctype,
						'fieldname': field['fieldname'],
						'label': field['label'],
						'fieldtype': field['fieldtype'],
						'default': field.get('default'),
						'read_only': field['read_only'],
						'hidden': field['hidden'],
						'insert_after': 'naming_series' # Try to put it early
					})
					custom_field.insert(ignore_permissions=True)
					# We don't commit here, let the transaction handle it
				except Exception as e:
					frappe.log_error(f"Failed to create custom field {field['fieldname']} for {doctype}: {str(e)}")
